agot_others_cb_tier = {
	categories = {
		agot
	}
	# 10 Aug, SECOND PASS -- KINGDOM IS THE DEFAULT AGAIN, for a reason the first pass had backwards.
	#
	# Duchy tier was made default to stop Winterfell falling first. It did not, because TIER WAS NEVER THE
	# CAUSE: the engine picked the war goal on ai_score and ai_score does not know which castle is in the
	# books. Both tiers went for Winterfell. What fixed it was naming the target ourselves.
	#
	# And on AGOT's map, kingdom is the honest unit. The North's Marches -- Winterfell, the Dreadfort, Last
	# Hearth, Karhold, the Neck, Skagos and the rest -- are KINGDOM titles, eight of them under e_the_north.
	# Duchy tier was cutting below the level the map is drawn in, so every duchy the dead took still read to
	# the player as "they are in the March of Winterfell", because that is the label on the region.
	#
	# With ln_pick_invasion_kingdom_effect naming the goal, kingdom tier gives the shape Thomas asked for:
	# Last Hearth, then Karhold, then the Dreadfort, and Winterfell only once it is surrounded.
	default = others_cb_kingdom

	others_cb_duchy = { }
	others_cb_kingdom = { }
	others_cb_empire = { }
}
