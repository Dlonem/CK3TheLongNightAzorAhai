# AGOT: The Long Night & Azor Ahai -- the two halves of the blessing score that must not cancel out.
#
# WHY THESE ARE SCRIPT VALUES AND NOT JUST MORE `modifier` LINES. An ai_will_do block is a flat sum, so a
# man with four flaws simply goes further and further negative and there is no way to say "enough". Script
# values take `min` and `max` (see zz_ln_flee_values.txt:16), which is the only way to CAP a subset of the
# sum. Two caps, doing opposite jobs:
#
#   ln_blessing_sin_value      min = -30   no pile of flaws can bury a man the fire has marked
#   ln_blessing_destiny_value  min = 0     destiny can never be subtracted, only ignored
#
# Both are read in the CANDIDATE's scope, via `add = scope:recipient.<value>` -- real vanilla syntax, e.g.
# `add = scope:actor.diarchy_regent_succession_score_value`.

# ── THE SINS, CAPPED ───────────────────────────────────────────────────────────────────────────────────
# R'hllor's own sins are content / forgiving / arrogant, read off the_rhllor_religion in
# common/religion/religion_types/00_agot_the_rhllor.txt. cynical and craven are ours: a man who believes in
# nothing is no use to a god, and a coward is not Azor Ahai in any telling.
#
# THE CAP IS THE POINT. Thomas: "a destined man may be badly flawed. That's literally the whole point."
# Before it, craven + content came to -40 and no amount of prophecy could climb back out. Now the worst
# possible character costs 30, which is real -- it beats a duchy -- but survivable by anyone the fire wants.
ln_blessing_sin_value = {
	value = 0
	if = { limit = { has_trait = content }    add = -15 }
	if = { limit = { has_trait = forgiving }  add = -10 }
	if = { limit = { has_trait = arrogant }   add = -15 }
	if = { limit = { has_trait = cynical }    add = -15 }
	if = { limit = { has_trait = craven }     add = -25 }
	min = -30
}

# ── DESTINY, FLOORED ───────────────────────────────────────────────────────────────────────────────────
# Everything here is a sign the world is already pointing at somebody, independent of what they hold.
ln_blessing_destiny_value = {
	value = 0

	# The story has a use for him. See ln_bears_a_destiny_trigger for why this is not hardcoding a name,
	# and why it keeps working for people born decades into a campaign.
	if = { limit = { has_character_flag = canon_status_canon }      add = 35 }
	else_if = { limit = { has_character_flag = canon_status_semicanon }  add = 15 }

	# The sword that was promised, and the swords that were not.
	if = { limit = { ln_carries_lightbringer_trigger = yes }   add = 40 }
	if = { limit = { ln_carries_valyrian_steel_trigger = yes } add = 25 }

	# Blood of the dragon. Deliberately modest -- the dynasty is large and full of the unworthy, and a
	# priest who weighted this heavily would end up attached to Aemond during the Dance. Twenty points is
	# an omen, not a coronation; his sins still have to be paid for out of the same sum.
	if = { limit = { agot_is_dragonblood_character = yes }     add = 20 }
	else_if = { limit = { agot_has_dragonblood_heritage = yes } add = 10 }
	if = { limit = { dynasty ?= dynasty:dynn_Targaryen }       add = 10 }

	# Sight. Both traits verified: dragon_dreams 00_agot_traits.txt:1974, greensight :2005.
	# (The trait is greensight. There is no `greenseer` trait -- that word is only a doctrine parameter.)
	if = { limit = { has_trait = dragon_dreams } add = 20 }
	if = { limit = { has_trait = greensight }    add = 15 }

	# ── THE WANDERER CLAUSE ────────────────────────────────────────────────────────────────────────────
	# The whole reason this file exists. A landless adventurer carrying any sign at all is Daenerys across
	# the Narrow Sea, or Dunk on the road with a borrowed shield. The station ladder gives them nothing;
	# this gives them the equivalent of a duchy for being nobody in an interesting way.
	if = {
		limit = {
			is_landless_adventurer = yes
			ln_bears_a_destiny_trigger = yes
		}
		add = 20
	}

	# The weapon the hour demands. See ln_is_the_weapon_trigger.
	if = { limit = { ln_is_the_weapon_trigger = yes } add = 30 }

	min = 0
}

# ══ 1.0.15 -- THE WORTHINESS SCORE ══════════════════════════════════════════════════════════════════════════
# Who does the story hand the blow to, when more than one hand is available? Used by the score-victory
# cornering (war_for_the_dawn_cb on_victory), the captured-king execution (ln_dawn.11, via ln_nk_captured_effect), and nothing else --
# the open-battle duel (ln_dawn.2) keeps its own inline weights because there the question is survival, not
# selection.
#
# WHAT THE NUMBER MEANS (checklist #5): this is not "who is the best fighter" -- it is "who is the story
# pointing at". Prowess is a tie-breaker at the bottom, not a headline. Thomas's calibration from the spear
# session holds: Chosen +30 over Favored +15, Resurrected +10 ("luck already spent"), and the destiny value
# on top carries Lightbringer (+40), Valyrian steel (+25), dragon blood, sight and canon weight.
#
# INJURIES SUBTRACT (checklist: "weighted based on if you're injured"): a man being carried on a shield does
# not behead the King of Winter. incapable is an absolute bar, enforced here rather than in every caller.
ln_dawn_worthiness_value = {
	value = ln_blessing_destiny_value

	# AUDIT FIX -- AND THE SINS TOO. This file's own header calls these "the two halves of the blessing
	# score that must not cancel out", and both the interaction and the decision read them together. The
	# worthiness score inherited the destiny half and silently dropped the other, so a craven arrogant cynic
	# could be handed the Dawn while every red priest alive was refusing to bless him.
	# The -30 floor inside ln_blessing_sin_value is what keeps Thomas's rule intact -- "a destined man may
	# be badly flawed, that's literally the whole point" -- and it cannot swamp the +1000 bane term.
	add = ln_blessing_sin_value

	if = { limit = { has_trait = chosen_by_rhllor }       add = 30 }
	else_if = { limit = { has_trait = favored_by_rhllor } add = 15 }
	# Wave two under The Cycle: the man who ended the last one is the man the world expects to end this one.
	if = { limit = { has_trait = azor_ahai_reborn }       add = 40 }
	if = { limit = { has_trait = resurrected_by_rhllor }  add = 10 }

	# The tie-breaker, deliberately small: half of prowess, so 20 prowess is worth 10 points -- a third of
	# being Chosen. The story picks the named man with a limp over the nameless athlete, but between two
	# nobodies it picks the one who can swing.
	add = { value = prowess  divide = 2 }

	# What the field has already taken out of him.
	if = { limit = { has_trait = wounded_1 } add = -10 }
	if = { limit = { has_trait = wounded_2 } add = -20 }
	if = { limit = { has_trait = wounded_3 } add = -35 }
	if = { limit = { has_trait = maimed }    add = -25 }
	# An absolute bar, not a weight. Nothing outweighs being unable to stand.
	if = { limit = { has_trait = incapable } add = -1000 }

	# 1.0.15b -- AND THE ONE THING THAT DECIDES IT. A man holding dragonglass outranks the finest knight in
	# Westeros holding castle-forged steel, because the knight cannot hurt him and the man can. Deliberately
	# larger than every other term combined: this is not a preference, it is the difference between a
	# champion and a spectator. See ln_bears_the_bane_trigger.
	if = { limit = { ln_bears_the_bane_trigger = yes } add = 1000 }
}
