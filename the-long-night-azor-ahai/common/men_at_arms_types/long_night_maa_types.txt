# Other MAA

# @ CONSTANTS ARE FILE-SCOPED. @cultural_maa_extra_ai_score is declared at the top of vanilla's
# 00_cultural_maa_types.txt and is not visible from here, so `ai_quality = { value = @cultural_maa_extra_ai_score }`
# below was failing to parse every load. Same value as vanilla's and AGOT's: worth roughly eight extra
# regiments to the AI's scoring, which is what makes the AI actually want the unit.
@cultural_maa_extra_ai_score = 80

other_maa = {
	type = Other_infantry

	# The Long Night+: tougher, harder-hitting dead so the horde grinds down the living instead of folding fast.
	# BALANCE, 10 Aug -- reported by Franz Liszt on the Workshop ("why are wights raised as 45 dmg 189 def
	# ice spiders thats crazy"). He was right, and understated it. Toughness 120 was higher than ANYTHING in
	# vanilla or AGOT: both cap at 100, on accolade_maa_elephantiers (stack 25) and dragons_regiment
	# (stack 1). This unit carries stack = 500. Vanilla's median toughness is 20 and its standard stack 100.
	# His in-game 45/189 is this plus terrain and phase modifiers.
	#
	# 75 puts them at AGOT's giant_regiment tier (50 dmg / 75 tough), which is the right company for a
	# monstrous unit, and leaves them the toughest infantry on the map without being unkillable. Damage stays
	# at 38: the dead should hit hard, they just should not also be immortal.
	damage = 38
	toughness = 75
	pursuit = 70
	screen = 0

	siege_tier = 4
	siege_value = 0.9

	terrain_bonus = {
		forest = { damage = 3 toughness = 3 }
		taiga = { damage = 3 toughness = 3 }
		jungle = { damage = 3 toughness = 3 }
		mountains = { damage = 3 toughness = 3 }
		desert_mountains = { damage = -3 toughness = -3 }
		wetlands = { damage = 3 toughness = 3 }
	}
	# FIX: can_recruit was never closed. Everything below it -- counters, winter_bonus, buy_cost, stack, icon --
	# was being parsed as triggers INSIDE can_recruit and silently discarded; that is where "Unknown trigger:
	# icon" came from. Logic preserved verbatim, only the bracing corrected.
	# NOTE: the Banefort/witch clause looks like copy-paste from another mod. Left as written.
	can_recruit = {
		OR = {
			has_trait = other_trait
			AND = {
				dynasty = dynasty:dynn_Banefort
				any_secret = { secret_type = secret_witch }
			}
		}
	}
	
	# 1.0.16 -- WAS EMPTY. MEN_AT_ARMS_MAX_COUNTER = 0.9 (game/common/defines/00_defines.txt:600), so up to
	# 90% of a damage exchange was being left on the table -- and left on it in BOTH directions, because
	# `Other_infantry` is a custom type name that no vanilla or AGOT unit counters either. The dead were
	# fighting the whole war with the counter system switched off.
	# Ice Spiders are fast, monstrous shock infantry: they break formations and run down what breaks. They do
	# NOT counter cavalry -- a thing that size is not catching a horse. Types verified against
	# game/common/men_at_arms_types/.
	# ONE-WAY, AND DELIBERATELY WATCHED (adversarial review find). Nothing in vanilla, AGOT or this mod
	# lists `Other_infantry` in its own counters block, so this fix runs in one direction only: the dead
	# gain against these four and the living gain nothing back. Closing it properly would mean editing
	# vanilla and AGOT men-at-arms files, which is the file-collision failure mode this project avoids on
	# principle. peasant_militia was 2 in the first draft and is 1 here to keep the ceiling honest while the
	# rest of this release's buffs are unmeasured -- advantage, damage/toughness mults, innovations and
	# traditions all landed in the same version. If the dead feel unstoppable on the boot run, cut this
	# block first: it is the newest and the least load-bearing of the five.
	counters = {
		heavy_infantry = 1
		pikemen = 1
		archers = 1
		peasant_militia = 1
	}

	winter_bonus = {
		normal_winter = { damage = 3 toughness = 3 pursuit = 3 }
		harsh_winter = { damage = 6 toughness = 6 pursuit = 3 }
	}

	buy_cost = { gold = other_maa_recruitment_cost }
	low_maintenance_cost = { gold = other_maa_low_maint_cost }
	high_maintenance_cost = { gold = other_maa_high_maint_cost }
	ai_quality = { value = @cultural_maa_extra_ai_score }
	stack = 500
	hired_stack_size = 1
	# 1.1.0 (B2): was `titan` -- no titan.dds exists in vanilla, AGOT or this mod, so the regiment drew the missing-
	# texture placeholder. Zombie is the mod's own regimenttypes icon, the one ln_wight_horde already uses.
	icon = Zombie
}
