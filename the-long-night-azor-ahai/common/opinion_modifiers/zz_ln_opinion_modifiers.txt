# AGOT: The Long Night & Azor Ahai -- opinion modifiers.
#
# FOURTEEN OF THESE WERE REFERENCED AND NEVER DEFINED, which the 10 Aug run finally proved: fourteen distinct
# "Unknown opinion modifier ln_*" errors in error.log, one per call site that had actually fired. The
# localization existed (zz_ln_opinions_l_english.yml, written to be read in a tooltip at a glance), the call
# sites existed and passed real numbers -- and the definitions did not. So EVERY social consequence in this mod
# silently did nothing: taking the black, forging Lightbringer over a wife's body, deposing a Lord Commander,
# being named by the red priests. All of it logged one error and moved no opinion at all.
#
# The VALUE comes from the call site -- `add_opinion = { ... opinion = N }` overrides the definition -- so what
# these blocks actually own is HOW LONG the world remembers. That is the only interesting decision here, and it
# is chosen per-modifier below rather than copied. `opinion` is still given a sensible default in each, for any
# future call that forgets to pass one.

# -- THE WATCH AND THE REALM ---------------------------------------------------------------------------------
# The Watch remembers who fed it.
ln_fed_the_watch_opinion = {
	opinion = 15
	years = 20
}
ln_turned_the_crow_away_opinion = {
	opinion = -10
	years = 10
}
# Sent men AND grain, not one or the other. Called with +50 -- the largest positive in the mod, and it should
# be, because it costs a lord real levies in a century when nobody believes the Wall matters.
ln_fed_the_wall_opinion = {
	opinion = 50
	years = 20
}
# The Lord Commander's standing letter to the northern lords. Small, NEGATIVE and short on purpose: it stacks
# once per petition, so a Lord Commander who writes every year makes himself a nuisance rather than a cause.
ln_asks_the_north_for_help_opinion = {
	opinion = -5
	years = 10
}
