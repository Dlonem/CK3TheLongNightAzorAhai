	on_owner_death = { 

	# ── 1.1.0 ── THIS TEST USED TO SIT AT THE BOTTOM OF on_owner_death, AND MOVING IT HERE IS THE FIX. ──
	# Reported by Axel on the Workshop, and independently by Jorahm before him as "post-WFTD incoherence"
	# and "the surviving Other". Axel: "Went beyond the wall and won the war... after this, the Others
	# simply remained within their territory and stopped expanding. Everything else returned to normal."
	#
	# He was describing the branch order exactly. war_for_the_dawn_cb's on_victory sets
	# war_for_dawn_victory and THEN kills the Night King, so on_owner_death fires with the war already won.
	# With this test last, the heir branch (or, since 1.0.9, the successor search) crowned a NEW Night King
	# first -- nightking trait, invasion modifiers, the lot -- and only then did end_story run. The result
	# is a destroyed story, so nothing drives the invasion any more ("everything else returned to normal"),
	# and a freshly crowned Other sitting on the conquered territory doing nothing ("remained within their
	# territory and stopped expanding"). Both halves of his report, from one ordering mistake.
	#
	# AND 1.0.9 WIDENED IT, WHICH IS ON US. Before 1.0.9 the no-heir path under war resolution went straight
	# to end_story. The 1.0.9 fix -- correct for the bug it targeted, a war ending inconclusively when the
	# Night King was merely decapitated -- made that path search for a successor and crown one. So the
	# surviving-Other case, which previously needed the Night King to have an heir, started happening when
	# he had none as well. Both paths now ran through a coronation before this line was ever reached.
	#
	# Asked FIRST and as an if/else_if chain, so a won war ends the story and crowns nobody. The ordering is
	# the entire change: every branch below is untouched, and the death-resolution case is unaffected because
	# both this test and the old one require others_war_resolution.
	# ── 1.0.15b -- CAST DOWN, NOT DESTROYED. ──────────────────────────────────────────────────────────
	# Set by ln_death_catalyst_effect when a Night King is killed by something that cannot actually end
	# him (see ln_bears_the_bane_trigger -- dragonglass, Valyrian steel, dragonfire, nothing else). His
	# body stops; the thing behind it does not. Asked FIRST, ahead of every ending branch, so the dark
	# crowns its next king instead of the story quietly closing. The successor search is the same one the
	# no-heir branch below uses, and if there is genuinely no Other left alive the chain falls through to
	# the ordinary handling.
	if = { limit = { has_global_variable = ln_nk_cast_down

					 global_var:ln_nk_cast_down ?= story_owner

					}

		remove_global_variable = ln_nk_cast_down

		story_owner ?= { save_scope_as = ln_dead_nightking }

		if = {
			limit = {
				exists = title:h_the_others.holder
				title:h_the_others.holder = { has_trait = other_trait }
				NOT = { title:h_the_others.holder ?= scope:ln_dead_nightking }
			}
			title:h_the_others.holder = { save_scope_as = ln_next_nightking }
		}
		else_if = {
			limit = { exists = story_owner.primary_heir }
			story_owner.primary_heir = { save_scope_as = ln_next_nightking }
		}
		else = {
			random_ruler = {
				# 1.1.0b -- prefer their own house; see the note at the reconciler's ladder (~:1800).
				limit = {
					has_trait = other_trait
					house ?= house:house_theothers
					NOT = { this ?= scope:ln_dead_nightking }
				}
				alternative_limit = {
					has_trait = other_trait
					NOT = { this ?= scope:ln_dead_nightking }
				}
				save_scope_as = ln_next_nightking
			}
		}

		if = {
			limit = { exists = scope:ln_next_nightking }
			debug_log = "LN_TRACE BANE: the night king was cast down by ordinary steel - another of them takes the crown"
			make_story_owner = scope:ln_next_nightking
			scope:ln_next_nightking = {
				# AUDIT FIX: one crowning, shared -- and this is where the ln_one_of_them stamp arrives.
				ln_crown_the_next_nightking_effect = yes
			}
		}
		else = {
			debug_log = "LN_TRACE BANE: cast down, but there is no Other left alive to take the crown - the story ends"
			end_story = yes
		}
	}

	else_if = { limit = { AND = { has_game_rule = others_war_resolution

							 has_global_variable = war_for_dawn_victory

							}

							}

							debug_log = "LN_TRACE: long_night_story ending - war_for_dawn_victory stood at owner death; no successor crowned"

							end_story = yes

	}


	

	else_if = { limit = { has_game_rule = others_death_resolution }

	

	debug_log = "LN_TRACE: long_night_story ending - night king died under others_death_resolution"

	end_story = yes

	}

	# Under `others_death_resolution` (the default) ending here is correct and intended: that rule promises

		limit = { has_game_rule = others_war_resolution }

				ln_crown_the_next_nightking_effect = yes
			}
		}

		}
	}
	else = {

	debug_log = "LN_TRACE: long_night_story ending - night king died with no heir to promote"

				exists = title:h_the_others.holder

				# the dead. The moment the dead took it the new holder is converted to wight_government
				# (mutate_into_other_effect), so has_government = nights_watch_government became false FOREVER and
				# the Wall could never fall afterwards. On a 7-day pulse the war window was easy to miss entirely.
				# That is the "Wall doesn't collapse after the Night's Watch loses" report.
				OR = {
					# a) the dead have TAKEN Castle Black in the field -- the county is occupied by the
					#    Others' host while the Watch still holds the title. Live-run report (Thomas,
					#    290 AC): the old check was war-declared + at-war, so the "Wall has fallen" card
					#    arrived the week the war STARTED, with the Watch untouched on its battlements.
					#    Occupation is the moment the wall is actually stormed.
					AND = {
						title:c_castle_black.holder = {
							has_government = nights_watch_government
							is_at_war_with = title:h_the_others.holder
						}
						title:c_castle_black = { county_controller = title:h_the_others.holder }
					}
					# b) failsafe: the dead already HOLD Castle Black, so the Wall is breached by definition.
					#    13 Aug (1.0.6), Lyon's report: this card fired while he was CAPTURING the Wall BACK
					#    from the Others -- the dead still held the title, so (b) passed, and "the Wall is
					#    falling" arrived as the living retook it. The NEVER-IF half was missing: the county
					#    controller must ALSO still be of the dead. When the living are storming Castle Black
					#    the controller is the living army, and this stays shut.
					AND = {
						title:c_castle_black.holder = {
							OR = {
								this = title:h_the_others.holder
								has_trait = other_trait
								has_trait = wight_trait
								has_government = wight_government
							}
						}

									this = title:h_the_others.holder

							}
						}
					}
				}

					NOT = { has_variable = agot_wall_has_fallen_middle }
					NOT = { has_variable = agot_wall_has_fallen_right }
				}
			}
			effect = {
				agot_long_night_wall_collapse_effect = yes
			}
		}

					}

			}
		}

				limit = { scope:ln_others_holder = { has_trait = other_trait } }
				make_story_owner = scope:ln_others_holder
				scope:ln_others_holder = {
					# AUDIT FIX: one crowning, shared -- the stamp and the invasion modifiers come with it.
					ln_crown_the_next_nightking_effect = yes
				}
			}
			else = {
				# He is not. Find another Other anywhere in the world and put him on the throne.
				# 1.1.0b -- PREFER THEIR OWN HOUSE. This search is a whole-world scan and it was picking
				# Others who rule their own realms far from h_the_others: ln_raised_as_an_other_effect
				# deliberately makes Others who KEEP their titles (a beaten lord is taken, not killed), so
				# the free-folk lord of Deadwood Grove can be one, and the crown was landing on him.
				# alternative_limit keeps the old behaviour as the fallback, so the ladder can still never
				# come up empty -- it just looks at their own house first.
				random_ruler = {
					limit = {
						has_trait = other_trait
						house ?= house:house_theothers
						NOT = { this = scope:ln_original_nightking }
						NOT = { this = scope:ln_others_holder }

						has_trait = other_trait
						NOT = { this = scope:ln_original_nightking }
						NOT = { this = scope:ln_others_holder }

					make_story_owner = scope:ln_the_great_other
