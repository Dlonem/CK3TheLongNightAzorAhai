# AGOT: The Long Night & Azor Ahai -- the price of taking a side early.

# coming -- levies committed north, upkeep for men you will not see again for years.

	advantage = 3                       # men who have been told what they are marching towards

# Thoros of Myr after Beric's sixth: "he looked like something that had been dug up."

	icon = health_negative
	health = -0.75
	fertility = -0.25
	monthly_piety = 1.0        # the order knows what he did
	same_faith_opinion = 20
}

# THE COUNTERWEIGHT. Thomas: "I don't want shit to just keep going downhill. I want there to actually be some
# benefit." Granted by ln_risen.11 at the end of the Resurrected chain, and permanent.
#
# A man who has already been dead is very hard to threaten, and everybody around him knows it. That is the
# whole modifier: dread that is not cruelty, and men who will hold a line next to him because the worst thing
# that can happen to them has visibly already happened to him and he came back and kept going.
ln_death_holds_no_terror_modifier = {
	icon = martial_positive
	dread_baseline_add = 15
	enemy_hostile_scheme_success_chance_add = -15
	knight_effectiveness_mult = 0.2
	army_maintenance_mult = -0.1
	stress_gain_mult = -0.2
}
