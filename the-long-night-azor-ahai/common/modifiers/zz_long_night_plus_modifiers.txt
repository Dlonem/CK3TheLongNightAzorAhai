# The Long Night+ — visible marker for a noble who fled south ahead of the dead (paired with the dawn_refugee
# character flag, which drives the logic; this modifier is the player-visible layer). Icon: cold (the Long Night
# they fled). Removed if/when they later rally to the War for the Dawn (future feature).
dawn_refugee_modifier = {
	icon = cold_negative
	stress_gain_mult = 0.25      # the lingering trauma of flight
	monthly_prestige = -0.5      # the shame of abandoning their seat and their people
	# WAS `travel_speed`, AND IT BROKE THE WHOLE MODIFIER. travel_speed is a TRAVEL PLAN modifier, not a
	# character one, and one foreign key is enough for CK3 to refuse to classify the modifier as a character
	# modifier at all -- so `remove_character_modifier = dawn_refugee_modifier` errored every time a refugee
	# came home and the refugee state was never actually cleared. The character-scope form is
	# character_travel_speed / character_travel_speed_mult (vanilla's stewardship_modifier uses the flat one).
	character_travel_speed_mult = 0.25   # a desperate flight south — and a wanderer ever after
	advantage = 5                # survivor's grudge: hardened by what they fled (a hook for the War for the Dawn)
}

# Aftermath: a dawn refugee who came back and resettled his OWN ancestral county (fled → survived → returned).
# Permanent badge — the shame of the flight is answered.
ln_restored_ancestral_seat_modifier = {
	icon = prestige_positive
	monthly_prestige = 0.5       # the songs remember the lord who came home
	county_opinion_add = 15      # his returning people rejoice
}

# 13 Aug (1.0.6) -- Jorahm: "I didn't see any modifiers to indicate that the wall had fallen." Now there
# is one, permanent until the Wall is raised again (removed in the re-arm block of
# ln_struggle_monthly_effect alongside the building repair). Deliberately mild mechanically -- the broken
# building carries the real defensive loss -- but it sits on the county tab and NAMES what happened.
ln_wall_broken_modifier = {
	icon = county_modifier_development_negative
	county_opinion_add = -15
	development_growth_factor = -0.2
	supply_limit_mult = -0.15
}

# 1.0.6a -- the brothers of a fallen Watch, standing on your walls instead of the Wall.
ln_black_cloaks_in_the_garrison_modifier = {
	icon = martial_positive
	garrison_size = 0.05
	levy_reinforcement_rate = 0.05
	advantage = 2
}
