# AGOT: The Long Night & Azor Ahai -- EVERY hook into a vanilla on_action, in one place.
#
# WHY THIS FILE EXISTS. Before this, four separate files each declared `yearly_global_pulse = { on_actions = {...} }`
# and two declared `on_game_start_after_lobby`. Whether CK3 merges same-named on_action definitions across files or
# lets the last one parsed win is not something the mod should be betting on -- and if it is last-wins, three of
# the four yearly hooks were dead and nobody would ever have seen an error, because a hook that does not run
# produces silence, not a log line. That is the worst failure mode there is.
#
# THE RULE, from here on: a vanilla on_action is declared in THIS FILE and nowhere else. Feature files define
# their own named sub-lists (ln_*_pulse, ln_*_catalysts) and this file lists them. Adding a hook means adding one
# line here, which is also the only place to look to find out what this mod attaches to the game loop.

on_game_start_after_lobby = {
	# 1.0.8: others_maintenance.1 moved here from common/on_action/AGOT Invasions/invasions_game_start.txt,
	# which was the second declaration this file's header says should not exist. It was NOT dead --
	# on_action sub-lists really do merge across files, and the proof is that AGOT declares this same
	# on_action in two of its OWN files (common/on_action/game_start.txt:2636 and
	# common/on_action/agot_on_actions/agot_game_start.txt:2048) and ships it working. But consolidating
	# is strictly safe-or-better: if merging holds it changes nothing, and if it ever stopped holding,
	# whichever file lost the race was silently dropping either this event or the three hooks below.
	events = {
		others_maintenance.1
	}
	on_actions = {
		ln_prophecy_game_start
		ln_struggle_game_start
		ln_dragon_repair_game_start
		ln_household_game_start
	}
}

# 1.0.12 -- THE THREE THAT HAD NEVER FIRED. All three of these were declared in
# 00_others_on_actions.txt as raw `effect = {}` blocks against on_actions that vanilla AND AGOT also
# declare with an effect. The engine picks a winner by PARSE ORDER within common/on_action/ (top-level
# files in path order first, then subdirectories, last one parsed wins) and says so in error.log at
# jomini_onaction.cpp:161. 00_others_on_actions.txt sorts first, so it lost all three, every run, since
# the fork. Registered here as sub-lists, which merge instead of competing -- the same shape AGOT uses
# for esr_on_vassal_gained. Bodies stay in 00_others_on_actions.txt; the two county ones also had a
# scope:recipient bug fixed on the way through. Full write-up in that file's header.
on_vassal_gained = {
	on_actions = {
		ln_others_vassal_gained
	}
}

# 13 Aug (1.0.6) -- THE WALL TITLE CHURN, Thomas + Jorahm. AGOT's own LC re-election
# (on_nw_lord_commander_death in agot_wall_on_actions.txt) is gated on agot_is_independent_ruler.
# A Lord Commander with ANY liege -- which this mod's fealty content and the post-War-for-the-Dawn
# settlement can produce -- dies WITHOUT an election, so k_the_wall and d_the_wall fall to ordinary
# blood succession. That is how Thomas's debug panel showed d_the_wall passing to the Iron Throne
# holder by "flag inheritance", with AGOT's undismissable "Fall of the Night's Watch" card
# (agot_nw_ranger_events.0108) firing on every such churn.
#
# ONLY IF: the dying man is the Watch (government_is_nw) and actually holds k_the_wall.
# NEVER IF: he is independent -- that is AGOT's own case and their handler already runs; two
# elections in one death would fight each other.
# The effect is AGOT's own agot_nw_elect_lc, not a parallel system.
ln_nw_vassal_lc_death = {
	trigger = {
		government_has_flag = government_is_nw
		this = title:k_the_wall.holder
		NOT = { agot_is_independent_ruler = yes }
	}
	effect = {
		agot_nw_elect_lc = yes
	}
}
