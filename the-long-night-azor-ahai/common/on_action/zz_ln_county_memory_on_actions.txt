# AGOT: The Long Night & Azor Ahai -- county memory stamp, at the moment of conquest. (1.0.10)
#
# Declared here as a named sub-list and listed in zz_ln_00_hooks.txt, per the rule at the top of that
# file. Root = the character who gained the title; scope:title = the title that changed hands (vanilla
# on_title_gain contract, title_on_actions.txt:187). The one_of_them test runs FIRST because this fires
# for every title transfer in the game and must fail in one cheap check for every ordinary human.
ln_county_memory_on_title_gain = {
	effect = {
		if = {
			limit = {
				ln_is_one_of_them_trigger = yes
				scope:title ?= { tier = tier_county }
			}
			scope:title = { ln_stamp_county_memory_effect = yes }
		}
	}
}

# ══ 1.0.15e -- A CAPTURED SEAT DOES NOT KEEP ITS WALLS. ═════════════════════════════════════════════════════
# ln_last_bastion_modifier is addressed only through its holder's capital_county, by the yearly upkeep. The
# moment the county changes hands the old holder's pulse can no longer reach it -- he is dead, landless, or
# his capital is somewhere else now -- so the modifier STRANDS on the county. Before 1.0.15d that was a slow
# curiosity; now that the capital is the war-score objective, the county most likely to change hands is
# exactly the fortified one, and the stranded modifier would hand the dead +3 fort level and +50% garrison
# to hold Winterfell against every living army trying to take it back.
#
# Root = the new holder, scope:title = the gained title -- the same contract ln_county_memory_on_title_gain
# above relies on. The next yearly upkeep re-adds it if the new holder still qualifies, so a living lord
# inheriting a seat loses nothing but a few months of walls.
ln_seat_title_gain = {
	trigger = {
		scope:title ?= {
			tier = tier_county
			has_county_modifier = ln_last_bastion_modifier
		}
	}
	effect = {
		debug_log = "LN_TRACE SEAT: a fortified seat changed hands - the walls come down until its new lord earns them"
		scope:title = { remove_county_modifier = ln_last_bastion_modifier }
	}
}
