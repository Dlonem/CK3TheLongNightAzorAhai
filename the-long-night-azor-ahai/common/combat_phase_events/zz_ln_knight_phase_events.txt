# AGOT: The Long Night & Azor Ahai -- dying in battle, and the priest who has to decide about it.
#
# ══════════════════════════════════════════════════════════════════════════════════════════════════════════════
# THE RESEARCH THOMAS ASKED FOR, because the answer decided the design.
#
# Q: Can we hijack the vanilla battle death chain?
# A: Only by overriding `knight_killed`, and we should not.
#
#    Combat phase events are KEYED ENTRIES IN A GLOBAL DATABASE, one picked per pulse (every COMBAT_EVENT_DAYS
#    = 5 days) weighted by each entry's `chance`. So a mod can ADD entries in its own file and they merge --
#    AGOT does exactly that with knight_dragon_joined_battle in 00_agot_knight_phase_events.txt. That part is
#    clean and it is what this file does.
#
#    But `knight_killed` is the entry that kills, and there is no additive way to intercept it. Overriding it
#    means shipping our own copy, and this is where it falls down:
#
# Q: Does AGOT touch that file?
# A: HEAVILY. Vanilla's knight_killed is 463 lines. AGOT's is 993, with 587 lines of differences:
#       * they add `is_valid = { is_current_dragonrider_warfare = no }` -- see the correction below
#       * prowess "immortality" scaling changed from 30 to 40 in two places
#       * garuda_court_position replaced with household_guard_court_position and bloodrider_court_position
#    Copying their version into this mod means every one of those silently reverts the day AGOT edits it and we
#    do not notice. Dragonriders quietly become killable again. Nothing logs it. That is the exact failure mode
#    this project has already lost the most time to.
#
# Q: Would a CK3 update break it / would I have to patch immediately?
# A: Yes to both, twice over -- once for PDX and once for AGOT, and AGOT is the faster-moving of the two.
#
#    Verified separately: `on_natural_death_second_chance` is the ONLY hook in the entire game that can stop a
#    death, and it is natural-death only. There is no on_battle_death_second_chance. `on_death` is reactive and
#    CK3 cannot un-kill a character.
#
# ── TWO CORRECTIONS, 1.0.16, both raised by the Supernatural session working the same mechanic ──────────
#
# 1. "DRAGONRIDERS CANNOT BE KNIGHT-KILLED" WAS TOO STRONG, and it would mislead anyone reasoning about
#    the pool. is_current_dragonrider_warfare is NOT a permanent property of a dragonrider. Its definition
#    (AGOT 00_agot_dragon_triggers.txt:78) is three clauses:
#        age >= 14
#        is_current_dragonrider = yes
#        NOT = { has_character_flag = agot_not_using_dragon }
#    That third clause is the one that matters: a rider who is NOT using their dragon fails the trigger and
#    is fully back in the knight-kill pool. So the accurate statement is "a rider who is actually in the air
#    is excluded" -- the same man fighting on foot is as killable as anyone. Our own is_valid below carries
#    the identical clause and its comment ("a rider in the air is not in this lottery") was already right;
#    it was this header that overstated it.
#
# 2. THE POOL IS MUCH BIGGER THAN knight_killed, and the numbers change how you read our chance value.
#    Base CK3 ships knight_none at chance base = 2000 (00_knight_phase_events.txt:5) against knight_killed's
#    30. Our base 40 is therefore roughly 2% of a pulse, not 40-vs-30 -- though pulses are every
#    COMBAT_EVENT_DAYS = 5 days, so a long battle rolls it a dozen times and it accumulates. Read our
#    escalation modifiers (:96-121, up to 90 in the Long Night) against 2000, not against 30.
#
# AND THERE IS NO `forbid` FIELD. Checked directly against the engine's own schema, and this is worth
# recording because it is the obvious thing to reach for: game/common/combat_phase_events/
# _combat_phase_events.info documents exactly four fields -- type, is_valid, chance, effect. There is no
# suppression mechanism. Crowding the pool is genuinely the only additive lever.
#
# SO THIS DOES NOT INTERCEPT THE DEATH. It competes for the same slot.
#
#    When a blessed champion is in a combat, this entry sits in the same weighted pool as knight_killed and
#    takes some of the rolls that would otherwise have killed him. He goes down, badly, and does not die in
#    that instant -- and his priest gets the scene and the choice. It is not a lie: a man who took a blow that
#    would have killed anyone else and is lying in the mud not breathing is exactly the story, and it needs no
#    AGOT file, no vanilla file, and nothing to re-patch.
# ══════════════════════════════════════════════════════════════════════════════════════════════════════════════

ln_knight_fell_for_the_dawn = {
	type = knight

	is_valid = {
		# AGOT's own exclusion, respected rather than worked around: a rider in the air is not in this lottery.
		is_current_dragonrider_warfare = no

		is_alive = yes
		# 1.0.15 -- THE TRAIT GATE IS GONE, AND THAT IS THE BERIC TRACK OPENING (Thomas, 24 Aug). The
		# resurrected_by_rhllor check made the battlefield route once-per-lifetime -- and its twin in
		# has_natural_death_second_chance did the same to the bedside route, so the trait shipped four
		# track rungs that NOTHING in the game could reach (the 1.0.15 audit's finding). What limits
		# repeats now is what always limited Beric: the cooldown flag below (three years), and a NEW sponsor
		# each time -- the old one spent himself the first time and ln_sponsor_can_still_pay_trigger will
		# never pass him again.
		NOT = { has_character_flag = ln_had_the_last_kiss }
		NOT = { has_character_flag = ln_fell_for_the_dawn }

		# A priest is standing behind this man, and he has not already spent himself on somebody else.
		ln_has_living_sponsor_trigger = yes
		var:ln_blessing_sponsor ?= { ln_sponsor_can_still_pay_trigger = yes }

		# 1.0.14 -- THE PHASE GATE CAME OUT OF HERE AND BECAME A WEIGHT. It used to demand Cold Winds or the
		# Long Night, and most of a campaign is neither, so a blessed man who fell in 120 AC was not unlucky --
		# this entry could not enter the pool at all. A priest's promise does not have a calendar on it.
		# How closely the fire is listening is now a TIER in `chance` below, not a door.
	}

	# knight_killed opens at base 30 and is then scaled down hard by prowess and by outnumbering. 40 here means
	# a blessed champion is meaningfully harder to lose outright without being untouchable -- he can still be
	# maimed, still be wounded, still be killed on any roll this entry does not win.
	# knight_killed opens at base 30 and is then scaled down HARD by prowess -- factor = (30 - prowess) / 30,
	# floored at 0.1 with vanilla's own comment, "Everyone dies." THIS ENTRY HAS NO PROWESS SCALING, so the
	# ratio between the two moves in the champion's favour the better a fighter he is. At Cold Winds weight 40
	# against a prowess-15 knight's effective 15, the fire takes roughly three falls out of four.
	#
	# ── THE TIERS, AND WHY THEY ARE WEIGHTS AND NOT GATES ────────────────────────────────────────────────
	# Thomas: "it's still possible during the Long Summer, but less likely; pretty standard during Winter is
	# Coming, because we want a champion alive and well when we know the Night King is coming; and our
	# champions get at least an extra chance during the Long Night itself."
	#
	#   Long Summer, the Dawn, or no struggle at all   x0.5   ->  20     the fire is listening, not closely
	#   Cold Winds                                     x1.0   ->  40     standard. Keep the man alive.
	#   The Long Night                                 x1.5   ->  60     it is watching now
	#   and x1.5 again on anyone already named         ->  30 / 60 / 90
	#
	# Every `struggle:` read below is `?=`. That is load-bearing now rather than tidy: is_valid no longer
	# guarantees the struggle exists, and a bare `=` on a missing struggle throws.
	chance = {
		base = 40

		# THE QUIET YEARS. Not Cold Winds and not the Long Night -- so the Long Summer, the Dawn after the
		# dead are beaten, or a game where the struggle has not begun or was ruled off entirely. Halved, not
		# closed: a red priest who has said the words means them in a peaceful summer too.
		modifier = {
			factor = 0.5
			NOR = {
				struggle:ln_long_night_struggle ?= { is_struggle_phase = ln_phase_cold_winds }
				struggle:ln_long_night_struggle ?= { is_struggle_phase = ln_phase_the_long_night }
			}
		}

		# COLD WINDS carries no modifier on purpose. It is the baseline the other two are measured against --
		# the years when everyone can feel it coming and a champion is worth keeping on his feet.

		# THE LONG NIGHT. The dark itself; the fire is paying closer attention.
		modifier = {
			factor = 1.5
			struggle:ln_long_night_struggle ?= { is_struggle_phase = ln_phase_the_long_night }
		}

		# And it watches the ones it has already marked most closely of all.
		modifier = {
			factor = 1.5
			OR = {
				has_trait = favored_by_rhllor
				has_trait = chosen_by_rhllor
				has_trait = azor_ahai_reborn
			}
		}
	}

	effect = {
		debug_log = "LN_TRACE FELL FOR THE DAWN: the fire took the roll instead of knight_killed"
		debug_log_scopes = yes
		# 1.0.15: three years, matching ln_had_the_last_kiss. When this was permanent, a handshake that
		# aborted midway (priest killed on the road, event queue lost) burned the man's ONLY interception
		# forever with nothing paid out. Timed, the worst case is a three-year shadow -- and the flag stays
		# long enough to stop the same battle rolling him twice.
		add_character_flag = { flag = ln_fell_for_the_dawn  years = 3 }
		save_scope_as = ln_the_fallen

		# He is not dead. He is also not getting up on his own.
		# REASON is REQUIRED -- it is a macro parameter, not an optional flavour argument, and calling this
		# bare errors with "Arguments needed for scripted effect". Vanilla always passes a death reason.
		increase_wounds_no_death_effect = { REASON = death_wounds }
		add_character_modifier = {
			modifier = ln_fell_in_the_dark_modifier
			years = 3
		}

		# ── THE HANDSHAKE ──────────────────────────────────────────────────────────────────────────────
		# Thomas: "I don't have it on five speed. Boom, next thing you know, I'm dead. I'm like, what on
		# earth just happened?" So the man who FELL hears about it first, and his priest's answer reaches
		# him afterwards as its own card. Day 1 he goes down, day 3 the priest decides, day 4 he is told.
		#
		# AND BOTH SCOPES ARE STASHED -- defensively, not because the delay eats them. Saved scopes DO
		# survive a delayed trigger_event (see the corrected note in zz_ln_wight_notice_effects.txt).
		# Stashing still buys something real: it survives the referent dying in the three days before
		# the priest answers, and it makes the path debuggable, which it was not before.
		var:ln_blessing_sponsor ?= {
			save_scope_as = ln_the_priest
			set_variable = { name = ln_the_fallen_champion  value = scope:ln_the_fallen }
			trigger_event = { id = ln_prophecy.21  days = 3 }
		}
		if = {
			limit = { exists = scope:ln_the_priest }
			set_variable = { name = ln_the_kneeling_priest  value = scope:ln_the_priest }
			trigger_event = { id = ln_prophecy.22  days = 1 }
		}
	}
}

# ln_commander_fell_for_the_dawn -- 1.0.16. THE OTHER HALF OF WHAT JURING ASKED FOR.
#
# Juring, on the Workshop: "is it possible to make that death in duels and as a battlefield commander also
# count for the resurrection. In the end Beric was a commander when he died the first time and he died the
# for the sixth time in a duel against the Hound."
#
# He was half-served. ln_knight_fell_for_the_dawn above covers a blessed man dying as a KNIGHT, and it has
# shipped since 1.0.14. But this mod's entire combat_phase_events directory contained exactly ONE `type =`
# line and it was `knight` -- so nothing of ours ever entered the COMMANDER pool, and a blessed champion
# leading his own army simply died. Beric-as-commander was the one case the feature was named for and the
# one case it could not reach.
#
# ── WHY THE NUMBERS HERE ARE NOT THE NUMBERS ABOVE ────────────────────────────────────────────────────────
# The knight entry runs at base 40. Dropping 40 into the commander pool would have been wildly overtuned,
# because that pool is a completely different shape. Measured from AGOT's own override, which is the file
# that actually loads (1158310/2962333032/common/combat_phase_events/00_commander_phase_events.txt):
#     commander_none               1000        commander_duel                     30
#     commander_wounded              25        commander_dragon_sum                1
#     commander_maimed               10        commander_dragon_joined_battle 100000
#     commander_killed                5        commander_wild_dragon_joined_battle 0
# commander_killed opens at FIVE, against the knight pool's thirty, and is then scaled down by prowess on
# the same (30 - prowess)/30 curve. Base 8 here, with no prowess scaling, leaves a blessed commander taking
# roughly three kill-rolls in four at Cold Winds -- deliberately the same feel the knight entry describes
# for itself, arrived at with the arithmetic of this pool rather than that one.
#
# AND IT IS RESTRICTED THE WAY commander_killed IS RESTRICTED. Vanilla will not kill a commander outright
# unless he is ALREADY WOUNDED or is bad at both martial and prowess (`00_commander_phase_events.txt`
# is_valid). Our entry mirrors that condition exactly, and the reason is not politeness: without it we would
# be winning rolls that were only ever going to WOUND the man, "saving" a commander who was in no danger and
# quietly suppressing commander_wounded and commander_maimed instead. We take rolls from the entry that
# kills, and from nothing else.
#
# NOTE ON 100000. AGOT's commander_dragon_joined_battle sits at 100000 to make itself unmissable. We do not
# go near that number: two must-win entries in one pool means one of them silently loses, and it would be
# theirs. A dragon joining a battle must keep working exactly as AGOT wrote it.
# ══════════════════════════════════════════════════════════════════════════════════════════════════════════
ln_commander_fell_for_the_dawn = {
	type = commander

	is_valid = {
		# Same AGOT exclusion the knight entry respects: a rider in the air is not in this lottery. And with
		# commander_dragon_joined_battle at 100000, a dragon-joined battle is not our business either.
		is_current_dragonrider_warfare = no

		is_alive = yes

		# commander_killed's OWN gate, mirrored. See the header: we compete with the entry that kills, not
		# with the ones that wound. Verified against AGOT's override, which matches vanilla here.
		OR = {
			has_trait_rank = {
				trait = wounded
				rank >= 1
			}
			AND = {
				martial <= low_skill_rating
				prowess <= low_skill_rating
			}
		}

		# Identical handshake preconditions to the knight entry, including the SHARED ln_fell_for_the_dawn
		# flag -- a man cannot be commander and knight in the same combat, so sharing it is what stops the
		# two entries paying out twice for one battle.
		NOT = { has_character_flag = ln_had_the_last_kiss }
		NOT = { has_character_flag = ln_fell_for_the_dawn }
		ln_has_living_sponsor_trigger = yes
		var:ln_blessing_sponsor ?= { ln_sponsor_can_still_pay_trigger = yes }
	}

	chance = {
		base = 8

		# The same three tiers as the knight entry, for the same reason -- a priest's promise has no calendar
		# on it, but the fire listens harder in the dark. Every struggle read is `?=`: is_valid does not
		# guarantee the struggle exists and a bare `=` on a missing struggle throws.
		modifier = {
			factor = 0.5
			NOR = {
				struggle:ln_long_night_struggle ?= { is_struggle_phase = ln_phase_cold_winds }
				struggle:ln_long_night_struggle ?= { is_struggle_phase = ln_phase_the_long_night }
			}
		}
		modifier = {
			factor = 1.5
			struggle:ln_long_night_struggle ?= { is_struggle_phase = ln_phase_the_long_night }
		}
		modifier = {
			factor = 1.5
			OR = {
				has_trait = favored_by_rhllor
				has_trait = chosen_by_rhllor
				has_trait = azor_ahai_reborn
			}
		}
	}

	effect = {
		debug_log = "LN_TRACE FELL FOR THE DAWN: the fire took the roll instead of commander_killed"
		debug_log_scopes = yes
		add_character_flag = { flag = ln_fell_for_the_dawn  years = 3 }
		save_scope_as = ln_the_fallen

		increase_wounds_no_death_effect = { REASON = death_wounds }
		add_character_modifier = {
			modifier = ln_fell_in_the_dark_modifier
			years = 3
		}

		# The same handshake, into the same four cards: day 1 he goes down, day 3 the priest decides,
		# day 4 he is told. Nothing downstream needs to know which pool he fell out of.
		var:ln_blessing_sponsor ?= {
			save_scope_as = ln_the_priest
			set_variable = { name = ln_the_fallen_champion  value = scope:ln_the_fallen }
			trigger_event = { id = ln_prophecy.21  days = 3 }
		}
		if = {
			limit = { exists = scope:ln_the_priest }
			set_variable = { name = ln_the_kneeling_priest  value = scope:ln_the_priest }
			trigger_event = { id = ln_prophecy.22  days = 1 }
		}
	}
}

# THE DEAD DO NOT FALL TO A LUCKY SPEAR (1.1.0, 1 Sep 2026)
# ══════════════════════════════════════════════════════════════════════════════════════════════════════════════
# WHERE THIS CAME FROM. RedSaintNino, Workshop: "I killed a micro army of the night king, a 2000 man force,
# and it triggered the end of the night king." The 1.1.0 death-reason gate already stops that ending the Long
# Night -- he is cast down and another of them is crowned instead. But Thomas's read is that the gate treats
# the symptom:
#
#   "I'm not talking about rerouting death. I'm saying let's make it a statistic in probability of ever
#    rolling the death roll... push that percentage of actually dying in combat from 10% to 0.001%."
#
# HE IS DESCRIBING THIS FILE'S OWN MECHANISM, and it is already proven directly above: there is no `forbid`
# field and no way to intercept knight_killed, but the pool is weighted, so an entry that competes for the
# same slot takes rolls away from it. ln_knight_fell_for_the_dawn has done exactly that for blessed champions
# since 1.0.14. This is the same trick pointed at the other side of the field.
#
# ── THE ARITHMETIC, because the numbers are not intuitive ────────────────────────────────────────────────
#   knight_none      base 2000   (game/common/combat_phase_events/00_knight_phase_events.txt:5)
#   knight_killed    base 30, then scaled by factor = (40 - prowess) / 40, floored at 0.1
#
# 1.1.3 CORRECTION (1.1.2 open item 20): the numbers above are AGOT's, because AGOT's copy of this file is
# what loads (same path, "#AGOT Modified": value = 40 / divide = 40 where vanilla has 30 / 30). This header
# used to compute against vanilla's 30/30. The result is the same either way, and the 1.1.3 review corrected
# this note's first correction: he is created at prowess 30 (agot_long_night_scripted_effects.txt:233), but
# `prowess` here is his total, and his own traits add far more (other_trait and nightking in
# common/traits/invasions_traits.txt, giant +6, sadistic +2), so he is well past 40 and the factor sits on
# the 0.1 floor -- knight_killed enters the pool at an effective 3 against roughly 2000, ~0.15% per pulse.
# COMBAT_EVENT_DAYS = 5, so a long war rolls it hundreds of times, and RedSaintNino's 2,000 men found it.
# Compounding is the whole problem.
#
# At base 50000 the pool becomes ~52000 and his death roll is ~0.006% a pulse -- about twenty-five times
# safer, which is the order of magnitude Thomas asked for. He is still not immortal, and that is deliberate:
# the songs are allowed the one impossible spear.
#
# ── WHAT IT DOES NOT TOUCH ───────────────────────────────────────────────────────────────────────────────
# Nothing in the mod's own confrontations. ln_dawn.2's duel, ln_dawn.11, ln_dawn.12 and the Last Hour melee
# all kill him with an explicit `death` effect through AGOT's multi-duel engine -- none of them goes anywhere
# near the knight pool. This makes the field battle a worse road to his death without making the scenes one
# jot harder, which is the entire point: "a lot of people who try to take him on who aren't worthy end up
# dying, possibly, instead of having to replace the Night King outright."
#
# WALKERS: ASKED FOR AND ANSWERED, 1 Sep. The note that stood here said an ordinary Other dying to a lucky
# knight roll was arguably too cheap but that it was a balance call nobody had asked for. Thomas jotted it
# and then said to take the whole list this pass, so ln_knight_the_walkers_hold below is that entry. The two
# are deliberately an order of magnitude apart -- see its own arithmetic.
ln_knight_the_dead_do_not_fall = {
	type = knight

	is_valid = {
		is_alive = yes
		has_trait = nightking
	}

	chance = {
		base = 50000
	}

	effect = {
		# Nothing happens, and nothing happening is the point -- this entry exists to spend the pulse that
		# knight_killed would otherwise have had. Same shape as vanilla's own knight_none.
		debug_log = "LN_TRACE NIGHT KING: a battle pulse spent on him and he is still standing"
	}
}


# ═════════════════════════════════════════════════════════════════════════════════════════════════════════════════
# AND THE REST OF THEM HOLD TOO (1.1.0)
# ═════════════════════════════════════════════════════════════════════════════════════════════════════════════════
# Same mechanism as the Night King's entry above, one order of magnitude weaker, and the arithmetic is the
# reason for both numbers rather than a feel.
#
#   An Other is created at prowess 16 (other_template's spawn sites; 00_long_night_other_templates.txt).
#   knight_killed's own scaling is factor = (30 - prowess) / 30, so his is (30-16)/30 = 0.467 and the entry
#   enters the pool at an effective ~14 against knight_none's 2000 -- about 0.7% a pulse, five times the
#   Night King's own rate before his entry was added, and rolled every COMBAT_EVENT_DAYS = 5 days for as
#   long as the war lasts. Over a Long Night that is most of them, dying to spear levies.
#
#   At base 15000 the pool becomes ~17000 and the roll falls to ~0.08% -- roughly eight times safer, and
#   still an order of magnitude short of the Night King's ~0.006%. That gap is the point: he is the one who
#   cannot be taken by a lucky spear, and they are merely very hard.
#
# WHAT THIS DOES NOT TOUCH, and it is the same list as his. Every place the mod kills a Walker on purpose --
# the Last Hour melee, the encounter melee, ln_dawn's confrontations -- does it with an explicit `death`
# effect through AGOT's multi-duel engine and never enters this pool. So the reputation track still works
# exactly as it did: ln_walkers_slain is earned by standing in front of one, which is what it was always
# supposed to mean. All this removes is the version where a levy archer collects one by accident.
#
# WIGHTS ARE NOT IN HERE ON PURPOSE. They are what the dead spend. A wight that cannot be cut down in the
# line is not a horror, it is a chore.
ln_knight_the_walkers_hold = {
	type = knight

	is_valid = {
		is_alive = yes
		has_trait = other_trait
		# His own entry already spends the pulse for him; two must-win entries in one pool is the mistake
		# this file warns about in the commander section.
		NOT = { has_trait = nightking }
	}

	chance = {
		base = 15000
	}

	effect = {
		# Nothing happens, and nothing happening is the point -- the pulse is spent, and knight_killed did
		# not get it. Same shape as vanilla's knight_none.
		debug_log = "LN_TRACE WALKER: a battle pulse spent on one of them and it is still standing"
	}
}


# ═════════════════════════════════════════════════════════════════════════════════════════════════════════════════
# 1.1.2 -- THE COMMANDER POOL, WHICH WAS NEVER SHIELDED AT ALL
# ═════════════════════════════════════════════════════════════════════════════════════════════════════════════════
# THE BUG THIS FIXES, in players' own words. mikekcallaghan1124, 9 Sep: "I have now killed the night king on
# two play throughs marked and one was with lightbringer but the war just kept going on both times."
# Skyblaze137, 11 Sep: "i play Night kings kill and Ather stell life just." Both are describing a cast-down --
# he died to something that was not one of the four scenes, another of them was crowned, and the Long Night
# carried on. Thomas's ruling, 12 Sep: "it's pretty lore-breaking for people when it's like, oh I killed the
# Night King and then his cronies became the new Night King... I want the fix that we have that does that to
# be incredibly rare. That might mean fixing the root cause."
#
# ── THE ROOT CAUSE, AND WHY THE 1.1.0 SHIELD DID NOT CATCH IT ────────────────────────────────────────────
# ln_knight_the_dead_do_not_fall above shields the KNIGHT pool. The Night King is assign_commander'd to his
# largest army every ~15 days (00_others_heartbeat_events.txt), and a man cannot be commander and knight in
# the same combat -- so in his own battles he has spent two versions sitting in a pool nothing was crowding.
#
# THE COMMANDER POOL AS IT ACTUALLY LOADS (AGOT's copy wins -- it ships 00_commander_phase_events.txt at
# vanilla's own path, 977 lines against vanilla's 417, so AGOT's numbers are the live ones):
#   commander_none                  base 1000   (+2000 for extreme-conqueror AI)
#   commander_wounded               base 25, factor (50 - prowess)/50  -> 10 at prowess 30
#   commander_maimed                base 10, factor (40 - prowess)/40  -> 2.5
#   commander_killed                base 5,  factor -- SEE BELOW       -> 200
#   commander_dragon_sum            base 1
#   commander_dragon_joined_battle  base 100000
#
# ── THE TWO-STEP THAT WAS KILLING HIM ────────────────────────────────────────────────────────────────────
# commander_killed's own is_valid will not touch a hale commander: it needs `wounded rank >= 1` OR both
# martial and prowess at or below low_skill_rating. He is martial 25 / prowess 30, so only the wound opens
# the door. commander_wounded calls increase_wounds_effect, whose rank-3 branch is an outright
# `death = { death_reason = death_fight }`. So: wound him once, and every later pulse is a 200-weight roll
# at killing him outright in a pool of ~1,200. That is ~16% per pulse, every five days, for the rest of the
# battle.
#
# ⚠ AND 200 IS AN UPSTREAM TYPO, NOT A DESIGN. AGOT's commander_killed rescales vanilla's factor and the
# edit lost a word:
#       value = 40          # correct -- "value at which immortality happens"
#       subtract = prowess
#       value = 40          # <- this was meant to be `divide = 40`
# Script values execute in order (game/common/script_values/_script_values.info), so the second `value`
# discards the subtraction and the factor is a flat 40 for EVERY commander in an AGOT game, prowess
# irrelevant. Their two neighbouring entries made the same rescale correctly (`divide = 50`, `divide = 40`),
# which is what makes it a typo rather than intent. Registered in UPSTREAM-WATCH; it is theirs to fix and we
# are not shipping a copy of their file over it.
#
# ── SO THIS IS THE SAME TRICK, POINTED AT THE OTHER POOL, PLUS ONE THING MORE ────────────────────────────
# The entry spends the pulse the way its knight-side siblings do. It also CLOSES THE DOOR BEHIND IT: the
# effect strips any wound he has picked up, because a wound is the single precondition commander_killed
# needs. The trait's `opposites` list already names wounded_1/2/3, and opposites are known in this project
# to block `add_trait` -- but `change_trait_rank`, which is what increase_wounds_no_death_effect actually
# calls, addresses the trait GROUP and has never been tested against opposites here. Rather than bet a
# player's campaign on the answer, the entry removes the wound whether or not it should have been possible
# to apply. If opposites already hold, these three lines are no-ops that cost nothing.
#
# ── THE ARITHMETIC ───────────────────────────────────────────────────────────────────────────────────────
# At base 50000 the commander pool becomes ~51,000 and his chance of being wounded at all falls to ~0.02%
# a pulse. Conditional on a wound landing, he then has to draw commander_killed (200) before he draws this
# entry (50000) -- about 0.4%. The two together are ~0.00008% per pulse. He is still not immortal, which is
# deliberate and matches the knight-side entry: the songs are allowed the one impossible spear.
#
# 50000 is also the number chosen for a reason the file states above and is worth repeating: AGOT's
# commander_dragon_joined_battle sits at 100000 and MUST keep winning its own slot. Nothing in this file may
# go near that number.
#
# ── WHAT THIS DOES NOT TOUCH ─────────────────────────────────────────────────────────────────────────────
# Every sanctioned kill. ln_dawn.2's duel, ln_dawn.11, ln_dawn.12 and the Last Hour melee all kill him with
# an explicit `death` effect and never enter either pool. The field battle gets worse as a road to his death;
# the scenes do not get one jot harder.
ln_commander_the_dead_do_not_fall = {
	type = commander

	is_valid = {
		# Same AGOT exclusion every other entry in this file respects: a rider in the air is not in this
		# lottery, and with commander_dragon_joined_battle at 100000 a dragon-joined battle is not our
		# business either.
		is_current_dragonrider_warfare = no

		is_alive = yes
		has_trait = nightking
	}

	chance = {
		base = 50000
	}

	effect = {
		debug_log = "LN_TRACE NIGHT KING: a commander pulse spent on him and he is still standing"
		# THE DOOR COMMANDER_KILLED COMES THROUGH. Explicitly, one rank at a time, because the three concrete
		# traits are what the nightking trait's own `opposites` block names.
		if = { limit = { has_trait = wounded_3 } remove_trait = wounded_3 }
		if = { limit = { has_trait = wounded_2 } remove_trait = wounded_2 }
		if = { limit = { has_trait = wounded_1 } remove_trait = wounded_1 }
	}
}


# ── AND THE REST OF THEM HOLD IN COMMAND TOO ─────────────────────────────────────────────────────────────
# The commander-side twin of ln_knight_the_walkers_hold, at the same one-order-of-magnitude gap and for the
# same reason. Walkers lead the dead's smaller hosts constantly; before this they carried the full commander
# risk while their knight-side selves were shielded. Their wounds are NOT stripped -- that mercy is the
# Night King's alone, and a Walker who has been hurt in the field is a Walker the living can finish.
ln_commander_the_walkers_hold = {
	type = commander

	is_valid = {
		is_current_dragonrider_warfare = no
		is_alive = yes
		has_trait = other_trait
		# His own entry already spends the pulse for him; two must-win entries in one pool is the mistake
		# this file warns about above.
		NOT = { has_trait = nightking }
	}

	chance = {
		base = 15000
	}

	effect = {
		debug_log = "LN_TRACE WALKER: a commander pulse spent on one of them and it is still standing"
	}
}
