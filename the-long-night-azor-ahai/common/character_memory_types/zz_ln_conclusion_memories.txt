# AGOT: The Long Night & Azor Ahai -- what the men who ended it remember.
#
# Thomas: "Are there character flags that get added that, like, hey -- this person did this thing?" There
# were flags. Flags are for script. These are for the PLAYER: dated entries in the character's own memories
# list, visible every time anyone opens him, for the rest of his life and in his descendants' second- and
# third-person retellings. Shape copied from wight_raised_memory above (this mod's own, proven in play);
# categories are arbitrary grouping strings per vanilla's _character_memories.info.

# Created once, in ln_create_lightbringer_effect. The forger keeps this whatever else happens -- a man who
# paid that price and then watched somebody else end the dark with a borrowed axe REMEMBERS the price.

	categories = { major martial }
	icon = "martial.dds"

	}

		random_valid = { desc = ln_forged_lightbringer_memory_desc_third_perspective }

# ══ 1.1.0b -- THE TWO ENDINGS NOBODY REMEMBERED ═══════════════════════════════════════════════════════════
# Thomas: "do we have memories added for the long night, killing the Night King? Do we have anything else?"
# We had the slayer and his three knights, and that was all four people in the world. Every king who fought
# the whole war from the first snow, every player who was not the champion, came out of the Long Night with a
# character page identical to one from a save where nothing had happened.
#
# NO PARTICIPANTS ON EITHER, and that is deliberate rather than lazy. A participant tag that is not supplied
# at the firing site logs an error every time the page is drawn (game/common/character_memory_types/
# _character_memories.info), and both of these fire on EVERY living player at once -- including the runs where
# a wild dragon ended it and there is no slayer to name, and the run where the dark simply won and there is
# nobody at all. These are memories about the world, not about a man, and the text is written that way.

# ══ 1.1.0 (A3) -- THE BERIC RECORD ═══════════════════════════════════════════════════════════════════════
# The resurrection track is the one mechanic in this mod that repeats and escalates -- a man dies, a named
# red priest burns himself out dragging him back, the man comes back less -- and until now NOTHING wrote
# any of it down on either side. Two types, one per side of the bargain, each with the other party as its
# participant, so a twice-risen champion's page reads as a LIST: two dates, two priests. Shape copied from
# ln_paid_in_full_memory above. Both participant tags are supplied at every firing site, guarded on the
# scope being in hand (an unsupplied tag errors every time the page is drawn).
#
# ln_came_back_memory: on the man, once per return. Fired from ln_the_fire_takes_something_effect
# (zz_ln_returned_effects.txt) -- the one road BOTH resurrection routes run through (the bedside last
# kiss and the battlefield interception via ln_prophecy.23), so it cannot miss a return and cannot double.
ln_came_back_memory = {
	categories = { major }
	icon = "religious.dds"   # memory_types/ has no faith.dds; religious.dds is vanilla's (11 uses)

	description = {
		random_valid = { desc = ln_came_back_memory_desc }
	}
	second_perspective_description = {
		random_valid = { desc = ln_came_back_memory_desc_second_perspective }
	}
	third_perspective_description = {
		random_valid = { desc = ln_came_back_memory_desc_third_perspective }
	}

	participants = { ln_the_priest }
	duration = { years = 1000 }
}
