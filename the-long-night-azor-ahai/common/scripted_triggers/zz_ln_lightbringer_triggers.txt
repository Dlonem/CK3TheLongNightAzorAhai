# The four rungs. resurrected_by_rhllor is not part of the rhllor_champion group (it is its own trait), so the
# list is written out rather than checked by group.
ln_is_a_champion_of_rhllor_trigger = {
	OR = {
		has_trait = favored_by_rhllor
		has_trait = chosen_by_rhllor
		has_trait = azor_ahai_reborn
		has_trait = resurrected_by_rhllor
	}
}

# THE ONLY MAN WHO CAN PAY WITH HIMSELF.
#
# Thomas, on the Resurrected chain: "I don't want shit to just keep going downhill... The way Jon Snow was
# resurrected, that shit actually worked out for him really well."
#
# A man who has already died once, and come back, can drive the sword through his own heart and survive it.
# Nobody else can -- the tale is explicit that the third tempering kills whoever the blade goes into, and the
# only way around that is to be someone death has already had a turn at. It costs him: the wound never closes.
# But he does not have to kill his wife, and he gets the full sword.
#
# This is the payoff the Resurrected chain did not have, and it is only reachable by having died.
ln_can_temper_in_own_heart_trigger = {
	has_trait = resurrected_by_rhllor
}

# Somebody has to be standing at the anvil who knows what he is doing. A court smith is AGOT's and vanilla's
# own position (court_smith_court_position) -- no new one is created here, per the standing rule.
ln_has_a_forge_trigger = {
	is_landed = yes
	OR = {
		cp:councillor_court_chaplain ?= { has_trait = red_priest }
		court_position:court_smith_court_position ?= { is_available_adult = yes }
		ln_has_red_priest_trigger = yes
	}
}

# Is the sword already in the world? A global rather than an artifact search, because the artifact can be
# looted, vaulted, inherited or buried with a man, and none of that means it can be forged again.
ln_lightbringer_exists_trigger = {
	has_global_variable = ln_lightbringer_forged
}

# Carrying it. Used by the reveal, by the deed count and by the Night King's death.
ln_carries_lightbringer_trigger = {
	any_equipped_character_artifact = {
		has_variable = ln_is_lightbringer
	}
}
