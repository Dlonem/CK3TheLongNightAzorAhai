# AGOT: The Long Night & Azor Ahai -- hooking into AGOT's dragon system rather than inventing one.
#
# THE LINK, verified in AGOT's own script and not guessed at:
#     rider  -> dragon : var:current_dragon   (734 uses across AGOT)
#     dragon -> rider  : var:current_rider    (336 uses)
# Both are ordinary character variables on ordinary characters -- AGOT's dragons ARE characters, with
# has_trait = dragon as the load-bearing marker. So we can read the bond without touching a single AGOT file.
#
# THE EXEMPTION THOMAS ASKED FOR. He is right that the Wall's refusal cannot apply to the Night King's mount:
# agot_spawn_adult_ice_dragon_effect creates dragon_skroth with `dynasty_house = house:house_theothers` and
# trait = immortal. That house is the marker. A dead thing riding a dead thing has no trouble with a ward built
# to keep fire out, and having it balk at the Wall would be the single most immersion-breaking line in the mod.

ln_rides_a_living_dragon_trigger = {
	# 1.1.0 -- trigger_if, not a bare exists-then-read pair. The pair is logically correct, but CK3's
	# TOOLTIP build evaluates every line independently, so an option gated on this trigger logged
	# "Failed to fetch variable for 'current_dragon'" once per FRAME while its card was open -- 1,424
	# lines in one sitting of ln_dawn.2, and measurable FPS. trigger_if short-circuits the tooltip too.
	trigger_if = {
		limit = { exists = var:current_dragon }
		var:current_dragon = {
			is_alive = yes
			has_trait = dragon
			# Not one of theirs. See the note above.
			NOT = { house ?= house:house_theothers }
		}
	}
	trigger_else = {
		always = no
	}

	# And the rider is still among the living too.
	NOT = { has_trait = nightking }
	NOT = { has_trait = other_trait }
	NOT = { has_trait = wight_trait }
}

# ── IS THIS THING A DRAGON ──────────────────────────────────────────────────────────────────────────────────
# 1.0.12, from a Workshop report by Keir Starmer: "Minor bug with the dragonglass decision, it gives your
# dragon the modifier too."
#
# He is right, and the cause is structural rather than careless. AGOT's dragons ARE characters -- they sit in
# your court, they can be knights, and every_courtier and every_knight find them like anyone else. So any
# effect that arms, equips, blesses or curses "everyone in the hall" arms the dragon too unless it says
# otherwise. There is no engine-level "is this a person" we can lean on: is_human is used all over AGOT but
# nothing proves it excludes a dragon, and a guess that is wrong here fails silently.
#
# has_trait = dragon is the marker AGOT actually loads its whole dragon system on, and this mod already
# treats it that way (see the note in zz_ln_catalyst_effects.txt: "a dragon without it is pruned as an
# ordinary courtier"). One trigger so there is one place to widen it.
#
# THE WIDER PRINCIPLE, for whoever picks up the Supernatural compatibility patch: dragons are magic, so magic
# does not land on them. Werewolf and vampire conversions, blood offerings, ritual targets -- all of it wants
# this trigger, and all of it currently has the same hole.
ln_is_a_dragon_trigger = {
	has_trait = dragon
}


# ── IS SHE ACTUALLY OVER THIS FIELD ────────────────────────────────────────────────────────────────────────
# 1.1.0 PLAYTEST FIX, 1 Sep. ln_rides_a_living_dragon_trigger answers "does this man have a living dragon",
# which is a question about the BOND. Several cards were using it to answer a different question -- "is there
# a dragon in the sky above this scene" -- and beyond the Wall those two answers come apart, because of the
# mod's own founding piece of lore: Silverwing refused the Wall three times and Alysanne never got her over
# (see zz_ln_dragon_wall_effects.txt). Thomas, watching ln_dawn.2 fire its dragon card on the Desolate Field:
# "it's basically all well and good except for that one little bit about crossing the wall."
#
# The North is NOT excluded -- the North is south of the Wall and she flies there happily; that is why the
# refusal event uses the North as its catchment. Only the Wall itself and what lies past it are closed to her.
# No exemption is needed for the Night King's mount: house_theothers is already excluded upstream, and this
# trigger is only ever asked about a living man's living dragon.
# The ward is a thing in a structure, and the structure can fail. Latched every pulse by
# ln_refresh_beyond_the_wall_latch_effect (zz_ln_invasion_target_effects.txt) rather than asked as a map
# sweep, because ln_bears_the_bane_trigger reads it and that trigger is rebuilt every frame a tooltip is open.
ln_the_wall_is_breached_trigger = {
	has_global_variable = ln_wall_is_breached
}

ln_dragon_is_over_the_field_trigger = {
	ln_rides_a_living_dragon_trigger = yes
	# She will not cross an intact Wall. Once the dead hold a piece of it, there is a hole and the ward that
	# turned Silverwing three times is not there any more -- Thomas's call, 1 Sep: "maybe you can fly your
	# dragon up there after the wall falls." So this clause only binds while the Wall is whole.
	# 1.1.0: TWO WAYS SHE CAN BE UP THERE. The Wall is broken, or her rider forced her over it anyway
	# (ln_omen.10's fourth option). The second is the expensive one -- see ln_spear_hit_chance_value, where
	# forcing an intact ward makes the Night King's spear far likelier to find her.
	trigger_if = {
		limit = {
			ln_the_wall_is_breached_trigger = no
			NOT = { has_character_flag = ln_forced_her_over }
		}
		NOT = {
			location ?= {
				OR = {
					geographical_region = world_westeros_the_wall
					geographical_region = world_westeros_beyond_the_wall
				}
			}
		}
	}
}
