# AGOT: The Long Night & Azor Ahai -- CONSECRATION: the second road to Lightbringer.
#
# THE PROMISE THIS KEEPS. Shipped in the 1.0.10b change notes as "Next update" and carried in every work order
# since: "An alternative to forging Lightbringer -- consecrating a Valyrian steel blade the way Azor Ahai is
# said to have made the first one -- and a special case for Dawn."
#
# WHY IT IS A STAMP AND NOT A NEW SWORD. THE CK3 ARTIFACT API HAS NO RENAME. The complete list of effects that
# can touch an artifact after creation is add/remove_artifact_modifier, add/remove_house_artifact_claim,
# add/remove_personal_artifact_claim, copy_artifact_modifiers, destroy_artifact, destroy_owned_artifact,
# equip_artifact_to_owner, unequip_artifact_from_owner and set_artifact_rarity. set_title_name is the only
# rename that exists anywhere in the engine and it takes a title.
#
# That constraint turned out to be the better design. Consecration does not replace the blade and does not
# rename it: it adds a modifier, lifts the rarity and stamps ln_is_lightbringer on the artifact itself. So
# Longclaw stays Longclaw, keeps its own modifier, its history and its house -- and is ALSO the red sword.
# A destroy-and-recreate would have thrown away the artifact's whole history entry and its house claim, which
# is most of what makes a named Valyrian blade interesting in AGOT.
#
# ONE LIGHTBRINGER PER WORLD (Thomas's call, 20 Aug). Consecration sets the same ln_lightbringer_forged global
# the forge sets, so whichever road arrives first closes the other, and the consecrated blade answers
# ln_carries_lightbringer_trigger everywhere -- the reveal, the deed count, and the Night King's death.

# ── THE BLADE ───────────────────────────────────────────────────────────────────────────────────────────────
# AGOT's own test, verified rather than guessed: common/scripted_triggers/00_agot_artifact_triggers.txt:44
#     agot_is_valyrian_steel_artifact = { has_variable = valyrian_steel }
#
# ── 1.1.3: A BLADE IS A WEAPON. ONE DEFINITION, USED EVERYWHERE THE MOD ASKS. ──────────────────────────────
# Xia (Discord, 19 Sep): Consecrate the Blade refused Dawn and accepted a Valyrian Steel Goblet -- "how am I
# supposed to kill the Night King with a goblet?" AGOT's variable is set on its Valyrian steel ARMOUR too
# (00_agot_artifact_armor_effects.txt:231), and the Valyrian Steel submod (2962713441) sets it on a goblet
# of type miscellaneous_when_not_court (its 00_basic_creation_effects.txt:377-411). Everything in this mod
# that asked "is this Valyrian steel?" read the bare variable, so a goblet or a breastplate could be
# consecrated, counted as the prophecy deed, and bear the bane against the Night King. Meanwhile the
# destiny trigger was filtered the other way, to `artifact_type = sword` only, and dropped AGOT's own
# Valyrian axes, spear and daggers (all in 00_agot_artifact_vs_sword_effects.txt; the hammers and maces in
# 00_agot_artifact_weapon_effects.txt carry no valyrian_steel -- corrected in the 1.1.3 review). AGOT's one
# Valyrian crossbow (Lizard-Lion) is excluded by the bow clause below.
#
# Artifact scope. The slot is the test: every weapon type vanilla ships (axe, hammer, mace, spear, sword,
# dagger, artifacts/types/00_types.txt:119-186) sits in primary_armament, and so do the four bow types
# (:442-457), which are named out -- a Valyrian bowstring is not a blade. artifact_slot_type is vanilla's
# own trigger (script_values/04_ep2_tournament_values.txt:1313; customizable_localization/
# 00_artifact_custom_loc.txt:773). Armour, goblets, rings and everything else fall outside the slot.
ln_is_valyrian_blade_trigger = {
	agot_is_valyrian_steel_artifact = yes
	artifact_slot_type = primary_armament
	NOR = {
		artifact_type = bow
		artifact_type = longbow
		artifact_type = composite
		artifact_type = crossbow
	}
}

# Artifact scope. The NOT clause stops a blade being consecrated twice and stops the forged sword being
# offered to itself (it is not marked valyrian_steel, but a future route might hand one over that is).
ln_blade_can_be_consecrated_trigger = {
	ln_is_valyrian_blade_trigger = yes
	NOT = { has_variable = ln_is_lightbringer }
}

# Character scope. Does he carry something the rite could take hold of? Equipped only -- a sword in a vault is
# not in his hand at the fire.
ln_carries_a_blade_to_consecrate_trigger = {
	any_equipped_character_artifact = {
		ln_blade_can_be_consecrated_trigger = yes
	}
}

# ── DAWN, AND WHY IT IS ITS OWN CASE ────────────────────────────────────────────────────────────────────────
# Dawn is not Valyrian steel in the books -- it is the heart of a fallen star, forged at Starfall -- but AGOT
# stamps it valyrian_steel = yes all the same (00_agot_artifact_weapon_effects.txt:30, inside
# agot_create_artifact_vs_dawn_effect), so the general test above already finds it. What it needs is a
# different PRICE, not a different route.
#
# THE PRICE, SETTLED 1.0.11: no third tempering. A sword already quenched in a falling star does not need a
# life put through it -- but the bearer has to be the one the star was falling for, which is
# ln_deed_bleeding_star and nothing else.
#
# IDENTIFYING IT. There is no has_artifact_template trigger in this engine, and AGOT sets no dawn-specific
# variable. has_artifact_modifier IS vanilla (00_court_positions.txt:236, 90_minor_decisions.txt:2504) and
# vs_dawn_modifier is unique to Dawn, so the modifier is the handle. IF AGOT EVER REUSES vs_dawn_modifier ON
# A SECOND ARTIFACT THIS TEST GOES WRONG SILENTLY -- re-check it on their updates.
ln_blade_is_dawn_trigger = {
	has_artifact_modifier = vs_dawn_modifier
}

ln_carries_dawn_trigger = {
	any_equipped_character_artifact = {
		ln_blade_is_dawn_trigger = yes
		NOT = { has_variable = ln_is_lightbringer }
	}
}

# ── THE PRICE FOR EVERY OTHER BLADE ─────────────────────────────────────────────────────────────────────────
# Nissa Nissa was his wife, but the mod's argument has never been "your wife" -- it is that the price has to
# actually cost you. Thomas's call, 20 Aug: spouse, lover, soulmate, best friend, or an adult child.
#
# ADULTS ONLY, IN EVERY BRANCH INCLUDING CHILDREN. ln_lightbringer.3's volunteer already carries the same
# is_adult gate and this is the same rite by another road.
#
# WHY A CHILD IS ON THE LIST AT ALL: a champion of R'hllor who is celibate, widowed, or sworn to the Watch has
# no spouse and no lover, and locking those characters out would have made this whole path unreachable for
# exactly the people the prophecy layer is most about.
#
# The relation forms are vanilla's own -- `any_relation = { type = lover }` and friends appear 418 / 108 / 106
# times across the base game -- and the selection itself is done branch by branch in ln_consecrate.1's
# immediate, the same shape ln_lightbringer.3 uses to pick Nissa Nissa. A parameterised "is this person
# beloved" trigger was tried first and thrown away: it needed `prev` inside an OR inside a scope change, which
# is exactly the kind of construct that fails quietly rather than loudly.
ln_has_a_beloved_to_give_trigger = {
	OR = {
		primary_spouse ?= { is_alive = yes  is_adult = yes }
		any_relation = { type = lover        is_alive = yes  is_adult = yes }
		any_relation = { type = soulmate     is_alive = yes  is_adult = yes }
		any_relation = { type = best_friend  is_alive = yes  is_adult = yes }
		any_child     = { is_alive = yes  is_adult = yes }
	}
}

# ── CAN HE HOLD THE RITE AT ALL ─────────────────────────────────────────────────────────────────────────────
# Consecration is a rite in front of a fire, not smithing, so it does NOT want ln_has_a_forge_trigger's court
# smith. It wants somebody who can say the words. ln_has_red_priest_trigger is this mod's own existing test
# and is reused rather than duplicated.
# 1.1.3 -- the requirement text (ln_consecrate_needs_a_voice) used to say "or my own faith in the Lord of
# Light"; the code has never accepted faith alone. It is a red priest in the chaplain (or camp priest) seat,
# or being one. The text now says exactly that (1.1.3 review: "in my hall" read as any courtier, which the
# decision's new Hear the Red Priest face does mean -- so Melisandre as Spymaster opens that door but not
# this one; widening this to ln_red_priest_in_hall_trigger is Thomas's call, not made here).
ln_can_consecrate_trigger = {
	is_adult = yes
	is_alive = yes
	ln_is_a_champion_of_rhllor_trigger = yes
	NOT = { ln_lightbringer_exists_trigger = yes }
	OR = {
		ln_has_red_priest_trigger = yes
		cp:councillor_court_chaplain ?= { has_trait = red_priest }
		has_trait = red_priest
	}
}
