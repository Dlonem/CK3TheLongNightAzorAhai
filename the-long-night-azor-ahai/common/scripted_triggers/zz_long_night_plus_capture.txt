# The Long Night+ — override of desirable_for_capture_trigger (copy of AGOT's body, itself == vanilla, + a Night King clause).
# Safety net for the rare window where the Night King is caught in a siege WITHOUT an army (so the vanilla
# siege_0001_can_be_captured shield doesn't apply). siege.0001 routes "desirable" characters to CAPTURE and the rest
# to a random KILL (death_siege). Forcing the Night King to be desirable means: if ever taken in a siege, he is
# IMPRISONED, never randomly slain — so the only "passive" death possible becomes a DELIBERATE execution. His
# glorious death in open battle is unaffected (that goes through combat-phase events, not this capture/kill split).
# Resync note: if AGOT changes desirable_for_capture_trigger, recopy their body and re-add the nightking clause.
desirable_for_capture_trigger = {
	OR = {
		has_trait = nightking   # The Long Night+ — the Night King is always worth capturing, never randomly killed in a siege.
		is_playable_character = yes
		any_parent = { always = yes }
		any_spouse = { is_playable_character = yes }
		any_close_family_member = { is_playable_character = yes }
		diplomacy >= very_high_skill_rating
		martial >= very_high_skill_rating
		stewardship >= very_high_skill_rating
		intrigue >= very_high_skill_rating
		learning >= very_high_skill_rating
		has_trait = lifestyle_physician
		has_trait = lifestyle_mystic
	}
}
