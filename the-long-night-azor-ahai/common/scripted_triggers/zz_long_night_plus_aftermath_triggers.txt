# The Long Night+ — AFTERMATH (resettling) shared triggers.
# Design doc: dev/the_long_night_plus_aftermath.md (sibling, not shipped).

# The "Crown" of the aftermath: the King on the Iron Throne if one exists, else an independent ruler of the
# North. NB: in AGOT 1.19 the North's lord-paramount title is e_the_north (EMPIRE tier) — k_the_north does
# NOT exist.
#
# v0.3.0 — THIRD BRANCH ADDED. The original hardcoded exactly two titles, so in any world where the Iron Throne
# had fallen and nobody happened to hold e_the_north, NOBODY qualified as the Crown — and ln_pose_question_of_
# the_north_effect fell straight through to its `else`, auto-firing the wildling path and silently deciding the
# fate of the North with no one asked. A player who conquered Westeros under a title of their own founding — a
# Golden Throne, a restored Kingdom of the North that never took e_the_north — was locked out of the biggest
# decision in the aftermath of their own war.
# The fallback is deliberately narrow so the aftermath decisions (which gate on this same trigger) cannot end up
# shown to several rulers at once: independent, empire tier or higher, holding land in Westeros, and only when
# the Iron Throne genuinely does not exist.
ln_aftermath_crown_trigger = {
	OR = {
		AND = {
			exists = title:h_the_iron_throne.holder
			this = title:h_the_iron_throne.holder
		}
		AND = {
			has_title = title:e_the_north
			is_independent_ruler = yes
		}
		ln_aftermath_fallback_crown_trigger = yes
	}
}

# Whoever actually rules Westeros when the two canon thrones are gone.
ln_aftermath_fallback_crown_trigger = {
	NOT = { exists = title:h_the_iron_throne.holder }
	NOT = { exists = title:e_the_north.holder }
	is_alive = yes
	is_independent_ruler = yes
	highest_held_title_tier >= tier_empire
	NOT = { has_faith = faith:cold_gods }
	NOT = { has_trait = wight_trait }
	NOT = { has_trait = other_trait }
	any_sub_realm_county = {
		title_province = { geographical_region = world_westeros_seven_kingdoms }
	}
}

# The Long Night is over — the Others' hegemony sits empty.
ln_long_night_over_trigger = {
	NOT = { exists = title:h_the_others.holder }
}

# County scope: unheld wilderness, open for colonization.
ln_wilderness_county_trigger = {
	holder ?= { government_has_flag = government_is_wilderness }
}

# County scope: belongs to the wildling settlement zone = the de jure k_the_wall (the Wall + both Gifts) plus
# every NORTHERN duchy directly adjacent to that zone (the mountain clans, the Last Lands... computed
# DYNAMICALLY — map adjacency is not readable from title files, and a hard list would rot). "Northern" =
# de jure e_the_north, which excludes beyond-the-Wall land ("south of the Wall" per design).
ln_wildling_zone_county_trigger = {
	OR = {
		target_is_de_jure_liege_or_above = title:k_the_wall
		AND = {
			target_is_de_jure_liege_or_above = title:e_the_north
			duchy = {
				any_de_jure_county = {
					any_neighboring_county = { target_is_de_jure_liege_or_above = title:k_the_wall }
				}
			}
		}
	}
}

# County scope: sellable to an Essosi colonial charter — coastal northern wilderness. (No east-coast-only
# geographical region exists; "coastal North" is the accepted approximation, cf. design doc.)
ln_essos_target_county_trigger = {
	ln_wilderness_county_trigger = yes
	is_coastal_county = yes
	title_province = { geographical_region = world_westeros_the_north_mainland }
}
