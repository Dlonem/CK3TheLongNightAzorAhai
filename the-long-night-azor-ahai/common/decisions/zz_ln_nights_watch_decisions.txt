# AGOT: The Long Night & Azor Ahai -- taking the black.
#
# THE FULL POST-MORTEM lives in common/scripted_triggers/zz_ln_nights_watch_triggers.txt. The short version:
# the first cut of this decision granted a Wall county by hand and then called change_government, and Thomas's
# 111 AC save came back with "Fief of The Nightfort -- Feudal Realm", no Night's Watch government and no liege.
# nights_watch_government.can_get_government wants a nightswatch trait AND a liege who holds k_the_wall, and a
# scripted title grant provides neither. AGOT already ships eighteen keepership court positions that do the
# whole thing correctly, so this decision now walks a man through their door instead of cutting its own.

# ══ SEND THE DUNGEONS TO THE WALL (1.0.16a) ══════════════════════════════════════════════════════════════
# Thomas asked for a MASS send-to-the-Night's-Watch beside the prisoner window's mass ransom / release /
# execute. That is not buildable, and the reason is worth writing down so nobody spends a night on it again:
# those three buttons are hardcoded C++ calls -- gui/window_court.gui:841-891,
# `[CourtWindow.MassPrisonerAction('ransom')]` and its two siblings. There is no fourth string, no
# character_interaction field that opts an interaction into mass mode, and Mass Release runs with ALL
# sixteen of its send_options switched OFF (that is what "without demanding anything in return" means in
# MASS_PRISONER_ACTION_RELEASE_TT), so AGOT's own Send to the Wall can never fire through it.
#
# A decision is the only batch mechanism the engine leaves open to script -- and the wheel already exists in
# this mod: ln_lc.10 option a (events/zz_ln_lord_commander_events.txt:240-277) is this exact every_prisoner
# loop, from the Watch's side. This is that loop from the other side of the Wall, made a player's door.
#
# AND IT ANSWERS THE REAL QUESTION. Thomas: "I honestly never know who I can send." The count is in the
# tooltip before you click, and the three conditions that block every prisoner at once each get their own
# line. See the block comment on ln_prisoner_fit_for_the_wall_trigger for why AGOT gives you none of that.
#
# IT IS AN ORDER, NOT AN OFFER, and that is a deliberate departure. AGOT's send_option routes through a
# negotiated release the prisoner can refuse at -75 base acceptance (00_prison_interactions.txt:6793). A
# man in your dungeon being marched to the Wall does not get a vote, and a mass action that silently
# half-worked would be worse than no mass action.
# ⚠ 1.1.0 (§10.3): the DISPLAYED name is now "Send the Prisoners to the Wall". The internal key below
# deliberately still says "dungeons" -- renaming a decision key breaks saves and every reference. Change
# only loc. Do not "fix" the mismatch.
ln_send_the_dungeons_to_the_wall_decision = {
	picture = { reference = "gfx/interface/illustrations/decisions/decision_prison.dds" }

	# 1.1.0 (§12): the base desc already speaks the south's truth out loud ("the south keeps its hands
	# clean, and the north gets bodies"). The North gets its own, because a Stark emptying his cells
	# northward is not dumping -- he is manning his own shield.
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					OR = {
						culture = culture:northman
						capital_province ?= { geographical_region = world_westeros_the_north }
					}
				}
				desc = ln_send_the_dungeons_to_the_wall_decision_desc_north
			}
			desc = ln_send_the_dungeons_to_the_wall_decision_desc
		}
	}
	selection_tooltip = ln_send_the_dungeons_to_the_wall_decision_tt
	confirm_text = ln_send_the_dungeons_to_the_wall_decision_confirm

	decision_group_type = major
	ai_check_interval = 0        # never AI. AGOT's own per-prisoner option already covers the AI's needs.

	is_shown = {
		is_ai = no
		# Shown whenever you are holding anyone, EVEN IF NOBODY IS ELIGIBLE -- that is the whole point. A
		# hidden decision teaches a player nothing; a visible one with a reason teaches him the rule once.
		any_prisoner = { is_alive = yes }
	}

	is_valid = {
		# 1.1.0 (§8.2, Send a Raven): was is_available, whose in_army = no is what Thomas kept
		# hitting. This decision can be taken from the field -- your castellan empties the cells on your written order. The vanilla ladder's
		# commanding rung (00_available_for_events_triggers.txt:540); everything else it checks is kept.
		is_available_adult_or_is_commanding = yes

		custom_description = {
			text = ln_wall_you_are_of_the_free_folk
			culture = { agot_is_wildling_culture = no }
		}
		custom_description = {
			text = ln_wall_your_seat_is_not_in_the_seven_kingdoms
			primary_title ?= {
				title_capital_county.title_province = {
					geographical_region = world_westeros_seven_kingdoms
				}
			}
		}
		custom_description = {
			text = ln_wall_there_is_no_watch_left
			ln_a_watch_still_stands_trigger = yes
		}
		custom_description = {
			text = ln_wall_none_of_them_will_serve
			# 1.1.0 (§10): may_be_sent, not fit -- fit never asked who the prisoner IS to you. See the
			# trigger's block comment for the captured-king defect this closes.
			any_prisoner = { ln_prisoner_may_be_sent_to_the_wall_trigger = yes }
		}
	}

	effect = {
		save_scope_as = ln_the_gaoler
		custom_tooltip = ln_send_the_dungeons_to_the_wall_effect_tt

		every_prisoner = {
			limit = { ln_prisoner_may_be_sent_to_the_wall_trigger = yes }
			save_scope_as = ln_wall_recruit

			# ══ AGOT HAS THE FINAL SAY. ═══════════════════════════════════════════════════════════════════
			# The limit above is OUR mirror, which exists so the tooltip can name a reason. This is AGOT's
			# own trigger, called the way every AGOT caller calls it, and it is what actually authorises the
			# sending. If their rules ever diverge from our mirror we simply send fewer men than we listed --
			# never a man they would have refused. That is the safe direction.
			if = {
				limit = { agot_nw_can_banish = { BANISHER = scope:ln_the_gaoler  BANISHEE = scope:ln_wall_recruit } }

				release_from_prison = yes
				agot_add_to_nightswatch_effect = yes

				# The court move, exactly as ln_lc.10 does it and for the same reason brindatm found in 1.0.6:
				# agot_add_to_nightswatch_effect grants the trait and strips claims but has NEVER moved anybody.
				# AGOT's own control case pairs it with add_courtier on the Lord Commander, so we do too --
				# MINUS their per-man agot_nights_watch.0100 notification, which would card a player Lord
				# Commander once per prisoner. From his court the yearly ln_allocate_brothers_effect sweep
				# posts them on to whichever company is thinnest.
				if = {
					limit = {
						exists = title:k_the_wall.holder
						NOT = { this = title:k_the_wall.holder }
					}
					title:k_the_wall.holder = { add_courtier = scope:ln_wall_recruit }
				}
			}
		}

		add_piety = 100
		# 1.0.15d's lesson, carried: k_the_wall can stand without a holder, and an unguarded reach through it
		# threw "Undefined event target 'title'" ten times in one run.
		#
		# 1.0.17 -- AND THE SECOND HALF OF THAT LESSON, WHICH THIS SITE WAS SHIPPED WITHOUT. is_shown here
		# is `is_ai = no` plus `any_prisoner`, and nothing in it excludes the Lord Commander -- a player LC
		# holding prisoners is an ordinary case, not an exotic one. When he takes this decision the giver
		# and the receiver are the same man, so reverse_add_opinion errors with "trying to add an opinion
		# of themself" and the Wall's gratitude is silently dropped after the men have already been sent.
		# Confirmed live in the same shape at events/zz_ln_watch_events.txt:255 on 30 Aug.
		if = {
			limit = {
				exists = title:k_the_wall.holder
				NOT = { this = title:k_the_wall.holder }
			}
			reverse_add_opinion = {
				modifier = ln_fed_the_watch_opinion
				target = title:k_the_wall.holder
			}
		}
	}
}

ln_man_an_empty_castle_decision = {
	picture = { reference = "gfx/interface/illustrations/decisions/decision_castle_view.dds" }

	desc = ln_man_an_empty_castle_decision_desc
	selection_tooltip = ln_man_an_empty_castle_decision_tt
	confirm_text = ln_man_an_empty_castle_decision_confirm

	decision_group_type = major
	ai_check_interval = 0        # never AI. The Watch recruits its own; this is a player's door.

	is_shown = {
		is_ai = no
		# YOU HAVE TO ALREADY BE ONE OF THEM. See ln_is_an_unplaced_brother_trigger for why this changed --
		# the Wall is not a place you walk up to and are handed a county.
		ln_is_an_unplaced_brother_trigger = yes
		title:k_the_wall.holder ?= { has_government = nights_watch_government }
	}

	is_valid = {
		# 1.1.0 (§8.2, Send a Raven): was is_available, whose in_army = no is what Thomas kept
		# hitting. This decision can be taken from the field -- an order to a garrison. The vanilla ladder's
		# commanding rung (00_available_for_events_triggers.txt:540); everything else it checks is kept.
		is_available_adult_or_is_commanding = yes

		# A CASTLE HAS TO BE AVAILABLE -- which is not the same as standing empty, and the difference matters
		# more than I realised until I counted. AGOT's own history fills fifteen of sixteen Wall baronies by
		# 200 AC, so "must be empty" would have made this a greyed-out decision in the bookmarks most people
		# play. See common/scripted_triggers/zz_ln_relieve_triggers.txt for the numbers.
		#
		# So: an empty castle, OR a brother who should not be holding one.
		custom_description = {
			text = ln_black_needs_a_castle
			OR = {
				ln_wall_has_a_free_keepership_trigger = yes
				ln_wall_has_a_relievable_keeper_trigger = yes
			}
		}

		# YOU HAVE TO BE WORTH A CASTLE. See ln_worth_a_castle_trigger -- the Watch takes anyone, the Lord
		# Commander does not give anyone a keep.
		custom_description = {
			text = ln_black_needs_to_be_somebody
			ln_worth_a_castle_trigger = yes
		}

		# THE ONE THING THAT CANNOT BE ALLOWED. A ruler with no heir and no liege has nobody to hand the realm
		# to, and the transfer would drop his titles into nothing. Landless men are never caught by this.
		custom_description = {
			text = ln_black_needs_somewhere_for_the_realm_to_go
			OR = {
				is_landed = no
				exists = primary_heir
				exists = liege
			}
		}

		# AGOT's base_court_position_validity_trigger refuses both of these, and a decision that offers a door
		# the engine then slams is worse than no door. Checked here so the player is told why.
		custom_description = {
			text = ln_black_not_at_war_with_the_watch
			NOT = { is_at_war_with = title:k_the_wall.holder }
		}
		is_imprisoned = no
	}

	effect = {
		# No ceremony left to perform -- he took the black already, and everything that goes with it happened
		# then. All that is left is the appointment.
		ln_grant_a_keepership_effect = yes
		trigger_event = { id = ln_watch.1  days = { 5 14 } }
	}

	# AGOT's own keeper appointments are what fill the Wall, so this is never AI.
	ai_will_do = { base = 0 }
}

# ══════════════════════════════════════════════════════════════════════════════════════════════════════════════
# THE OTHER WAY IN, FOR A WALL THAT IS ALREADY FULL
#
# By 200 AC exactly one of the sixteen Wall baronies is unheld. A decision that requires an empty castle is a
# decision that does not exist in most bookmarks. This is the version for the rest of them: you take the black
# knowing there is no room, and the Lord Commander makes room.
#
# It takes longer -- a rider has to go and tell a man he is finished -- which is why it is a separate door
# rather than a silent fallback inside the first one. The player should know which of the two happened.
# ══════════════════════════════════════════════════════════════════════════════════════════════════════════════
ln_take_the_black_and_wait_decision = {
	picture = { reference = "gfx/interface/illustrations/decisions/decision_castle_view.dds" }

	desc = ln_take_the_black_and_wait_decision_desc
	selection_tooltip = ln_take_the_black_and_wait_decision_tt
	confirm_text = ln_take_the_black_and_wait_decision_confirm

	decision_group_type = major
	ai_check_interval = 0

	is_shown = {
		is_ai = no
		# ALSO INTERNAL NOW. A brother of the Watch asking his Lord Commander to take a keep off a man who is
		# not using it -- which is a conversation between two members of an order, not a way in from outside.
		OR = {
			ln_is_an_unplaced_brother_trigger = yes
			has_government = ln_ranger_government
		}
		title:k_the_wall.holder ?= { has_government = nights_watch_government }
		# Only when the simple door is genuinely shut. Two decisions offering the same thing is worse than one
		# decision that sometimes cannot be taken.
		NOT = { ln_wall_has_a_free_keepership_trigger = yes }
	}

	is_valid = {
		is_available = yes
		is_imprisoned = no

		custom_description = {
			text = ln_black_needs_someone_to_relieve
			ln_wall_has_a_relievable_keeper_trigger = yes
		}
		custom_description = {
			text = ln_black_needs_to_be_somebody
			ln_worth_a_castle_trigger = yes
		}
	}

	effect = {
		# The rider goes out today. AGOT's on_court_position_revoked returns the castle to the Lord Commander
		# a day later, so the handover lands a week after that -- long enough that it reads as a journey rather
		# than a menu.
		ln_relieve_a_brother_effect = yes
		trigger_event = { id = ln_watch.9  days = 7 }
	}

	ai_will_do = { base = 0 }
}

# REBUILDING WHAT THEY GAVE YOU
#
# Thomas: "say I'm given the ruin for the Nightfort. Obviously I'm not gonna be getting any income, but perhaps
# there's an opportunity to turn it from a ruin into the actual building it's supposed to be. Maybe it's just a
# matter of getting the funds."
#
# Without this the black is a dead end: a ruin generates no tax, no levies and no buildings, so a man who took
# the black would hold a castle he could never do anything with. That is not a hard start, it is a broken one.
#
# set_holding_type is the engine's own route -- 14 vanilla uses for castle_holding alone.
#
# TWO PATHS, because after the rewrite above a man can be either. The Nightfort, the Shadow Tower and Eastwatch
# keeperships grant a COUNTY, so their holder may have several ruined baronies under him. The other fifteen
# grant a single BARONY, and a baron's ruin is not reachable through any_held_county at all -- which the first
# version of this decision would have missed entirely.
ln_rebuild_the_castle_decision = {
	picture = { reference = "gfx/interface/illustrations/decisions/decision_castle_view.dds" }

	desc = ln_rebuild_the_castle_decision_desc
	selection_tooltip = ln_rebuild_the_castle_decision_tt
	confirm_text = ln_rebuild_the_castle_decision_confirm

	decision_group_type = realm
	cooldown = { years = 5 }
	ai_check_interval = 60

	is_shown = {
		is_landed = yes
		has_government = nights_watch_government
		OR = {
			any_held_county = {
				any_county_province = { has_holding_type = ruin_holding }
			}
			any_held_title = {
				tier = tier_barony
				title_province = { has_holding_type = ruin_holding }
			}
		}
	}

	is_valid = {
		# 1.1.0 (§8.2, Send a Raven): was is_available, whose in_army = no is what Thomas kept
		# hitting. This decision can be taken from the field -- you are paying, not laying stone. The vanilla ladder's
		# commanding rung (00_available_for_events_triggers.txt:540); everything else it checks is kept.
		is_available_adult_or_is_commanding = yes
		custom_description = {
			text = ln_rebuild_needs_the_coin
			gold >= 400
		}
	}

	effect = {
		remove_short_term_gold = 400

		if = {
			limit = {
				any_held_county = { any_county_province = { has_holding_type = ruin_holding } }
			}
			random_held_county = {
				limit = { any_county_province = { has_holding_type = ruin_holding } }
				save_scope_as = ln_rebuilt_county
				random_county_province = {
					limit = { has_holding_type = ruin_holding }
					ln_raise_the_walls_effect = yes
				}
				# The county was empty for generations. It does not fill up because a wall went back up.
				change_county_control = 25
			}
		}
		else = {
			random_held_title = {
				limit = {
					tier = tier_barony
					title_province = { has_holding_type = ruin_holding }
				}
				save_scope_as = ln_rebuilt_county
				title_province = {
					ln_raise_the_walls_effect = yes
					county = { change_county_control = 15 }
				}
			}
		}

		add_prestige = 150
		trigger_event = { id = ln_watch.5  days = { 10 30 } }
	}

	ai_will_do = {
		base = 40
		modifier = { factor = 0  gold < 800 }
	}
}
