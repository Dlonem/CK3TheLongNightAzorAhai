# AGOT: The Long Night & Azor Ahai -- the dragonglass.
#
# The single most important thing a player can do with the warning years, and the reason the intro card moved to
# the Long Summer's midpoint. It is expensive, it is available only once the cold has started, and it expires --
# so a lord who mines too early has wasted the gold, and one who waits too long does not finish in time.
#
# Available to anyone in the struggle region. You do not have to be a northerner to see sense.

ln_mine_dragonglass_decision = {
	picture = { reference = "gfx/interface/illustrations/decisions/decision_oily_black_stone.dds" }

	desc = ln_mine_dragonglass_decision_desc
	selection_tooltip = ln_mine_dragonglass_decision_tt
	confirm_text = ln_mine_dragonglass_decision_confirm

	decision_group_type = realm
	# 1.0.15d -- WAS 60 MONTHS. Thomas's 130 AC run: ln_prophecy.31 fired ZERO times in a full Long Night
	# with the dead across half the North. Five years between AI evaluations, against a decision that only
	# becomes available once Cold Winds arrive, means most lords were looked at once or twice in the entire
	# window -- and each look was a weighted roll they usually lost. Twelve months, so a lord who cannot
	# afford it in the first winter can still buy it in the third.
	# This matters far more since 1.0.15b than it did before: dragonglass is now one of only three things
	# that can end the Night King, so an AI realm that never buys any is an AI realm that cannot win.
	ai_check_interval = 12

	is_shown = {
		exists = struggle:ln_long_night_struggle
		struggle:ln_long_night_struggle = {
			OR = {
				is_struggle_phase = ln_phase_cold_winds
				is_struggle_phase = ln_phase_the_long_night
			}
		}
		# 1.0.16 -- WAS `is_landed = yes`, AND THAT WAS THE BUG. Since 1.0.15b dragonglass is one of only
		# three things that can end the Night King, so this gate was not "who may mine" -- it was "who may
		# ever win." It locked out two populations the mod exists to serve:
		#   * the landless adventurer. Corvo beat the Long Night from a camp, with no dragons, and reported
		#     it; under the bane rule that run is no longer winnable.
		#   * THE NIGHT'S WATCH RANGER -- ln_ranger_government carries government_is_custom_landless, so the
		#     sworn brother who actually rides north was shut out of every dragonglass path in the mod,
		#     including the Fist of the First Men expedition written for him.
		# is_ruler is the honest test: it still excludes plain courtiers and knights, who correctly receive
		# ln_dragonglass_knight_modifier from their lord instead (zz_ln_preparation_effects.txt:98-117).
		is_ruler = yes
		NOT = { has_character_modifier = ln_dragonglass_stores_modifier }

		# THE KNOWLEDGE GATE, and it is the best hook in the mod because it costs nothing to build.
		# You cannot mine dragonglass if nobody in your hall has ever heard it does anything. The intro card
		# offers exactly that choice -- "Send to the Citadel. Ask what the records actually say" sets
		# ln_heard_the_warning; "The world is full of old stories" does not. So the decision you made forty
		# years into the Long Summer, which looked like pure flavour at the time, is what determines whether
		# you can arm against the dead when it arrives.
		OR = {
			has_character_flag = ln_heard_the_warning
			government_has_flag = government_is_nw        # a sworn brother's founding purpose IS the warning
			has_character_flag = ln_looked_north          # you sent men after the lost ranging
			has_character_flag = ln_named_it_first        # you believed the man who saw one
			ln_has_deed_greensight_trigger = yes          # you did not need telling
			ln_participates_in_the_dawn_trigger = yes     # or you swore yourself to this war and meant it
			struggle:ln_long_night_struggle ?= { is_struggle_phase = ln_phase_the_long_night }  # or it is here
		}
	}

	is_valid = {
		is_available = yes
		gold >= ln_dragonglass_cost_value
	}

	effect = {
		remove_short_term_gold = ln_dragonglass_cost_value
		add_character_modifier = {
			modifier = ln_dragonglass_stores_modifier
			years = 30
		}
		add_character_flag = ln_mined_the_glass
		trigger_event = ln_prophecy.31
	}

	# The AI needs a reason not to bankrupt itself on black glass, and a reason to do it anyway when the dead
	# are actually coming. Thomas's note: "make sure you have a lot of modifiers that impact an AI's decision
	# to now spend that kind of money."
	ai_will_do = {
		base = 25

		# It is here. Buy the glass.
		modifier = {
			add = 75
			struggle:ln_long_night_struggle ?= { is_struggle_phase = ln_phase_the_long_night }
		}
		# The North believes it first and always has.
		modifier = {
			add = 30
			OR = {
				culture = culture:northman
				culture = culture:frostborn
			}
		}
		# So does anyone who has actually seen something.
		modifier = {
			add = 40
			OR = {
				has_character_flag = ln_named_it_first
				has_character_flag = ln_looked_north
			}
		}
		# A cautious ruler prepares; a rash one spends it on a war.
		modifier = {
			add = 20
			ai_boldness <= -20
		}

		# ── 1.0.15d -- DOES THIS MAN BELIEVE IN IT? (Thomas: "properly weighted based on the type of
		# character, whether or not they believe in that kind of stuff.") Buying a mountain of black glass
		# on the word of a story is a matter of faith before it is a matter of strategy, and the mod
		# already knows exactly who believes.
		#
		# THE WATCH does not need convincing -- it is the reason they exist.
		modifier = {
			add = 60
			government_has_flag = government_is_nw
		}
		# A man who has SEEN it, or dreamed it, is not guessing.
		modifier = {
			add = 45
			OR = {
				has_trait = greensight
				has_trait = dragon_dreams
			}
		}
		# The red faith has been shouting about the long night for a century, and this is its hour.
		modifier = {
			add = 35
			faith = { has_doctrine_parameter = can_heal_via_flames }
		}
		# Zeal buys what prudence argues about.
		modifier = { add = 25  has_trait = zealous }
		modifier = { add = 20  has_trait = paranoid }
		# The named and the blessed have every reason to arm themselves.
		modifier = { add = 30  ln_is_a_champion_of_rhllor_trigger = yes }
		# And the ones who will not have it. A cynic buys nothing on the strength of a prophecy, and an
		# arrogant man does not believe anything can reach him. Weighted DOWN, never to zero: the glass is
		# still bought when the dead are at the gate, because by then it has stopped being a belief.
		modifier = { add = -35  has_trait = cynical }
		modifier = { add = -15  has_trait = arrogant }
		# Do not do this while broke.
		modifier = {
			factor = 0
			gold < ln_dragonglass_cost_value
		}
		# NOTE: `yearly_income` could not be verified as a trigger anywhere in vanilla, so this uses gold
		# reserves instead -- an AI that cannot comfortably absorb the cost twice over should not be buying.
		# 1.0.15d -- PHASE-AWARE. A 0.15 brake is right while this is insurance against a rumour. It is the
		# wrong number when the dead are over the Wall: a lord merely short of a comfortable reserve should
		# still be buying glass, and before this he almost never did.
		modifier = {
			factor = 0.15
			gold < ln_dragonglass_comfortable_value
			NOT = { struggle:ln_long_night_struggle ?= { is_struggle_phase = ln_phase_the_long_night } }
		}
		modifier = {
			factor = 0.5
			gold < ln_dragonglass_comfortable_value
			struggle:ln_long_night_struggle ?= { is_struggle_phase = ln_phase_the_long_night }
		}
	}
}
