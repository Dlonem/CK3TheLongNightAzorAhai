						set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "MISSING_TITLE_LOC" GENDER = male }
						set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "MISSING_TITLE_LOC" GENDER = female }
					}
				}

				### FALLBACK WARNING ###
				else = {
					set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "MISSING_TITLE_LOC" GENDER = male }
					set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "MISSING_TITLE_LOC" GENDER = female }
				}
			}

			### HIDDEN TITLED CHARACTERS ###
			else_if = {
				limit = { is_title_hidden_character = yes }

				if = {
					limit = { scope:high_septon ?= yes }
					if = {
						limit = { $SCOPED_TITLE$ ?= title:k_the_most_devout }

						set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "HIGH_SEPTON" GENDER = male }
					}

					### FALLBACK WARNING ###
					else = {
						set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "MISSING_TITLE_LOC" GENDER = male }
						set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "MISSING_TITLE_LOC" GENDER = female }
					}
				}
				else = {
					set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "BLANK_TITLE_LOC" GENDER = male }
					set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "BLANK_TITLE_LOC" GENDER = female }
				}
			}

			### FALLBACK WARNING ###
			else = {
				$SCOPE$ ?= {
					set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "MISSING_TITLE_LOC" GENDER = male }
					set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "MISSING_TITLE_LOC" GENDER = female }
				}
			}
		}
	}
	else = {
		if = { # Septon/Septa
			limit = { scope:septon ?= yes }

			set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "SEPTON" GENDER = male }
			set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "SEPTA" GENDER = female }
		}
		else_if = { # Maester/Archmaester/Grandmaester
			limit = { scope:maester ?= yes }

			if = {
				limit = {
					has_inactive_trait = grandmaester
					NOT = {
						has_trait = archmaester
					}
				}
				set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "grand_maester" GENDER = male }
			}
			else_if = {
				limit = {
					has_trait = archmaester
					NOT = {
						has_inactive_trait = grandmaester
					}
				}
				set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "ArchaesterTitle" GENDER = male }
			}
			else_if = {
				limit = {
					NOR = {
						has_trait = archmaester
						has_inactive_trait = grandmaester
					}
					OR = {
						AND = {
							has_trait = maester
							has_trait_xp = {
								trait = maester
								value = 100
							}
						}
						has_trait = maester_historical
					}
				}
				set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "MaesterTitle" GENDER = male }
			}
			else_if = {
				limit = {
					NOT = { has_trait = archmaester }
					has_trait = maester
					has_trait_xp = {
						trait = maester
						value = 0
					}
				}
				set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "MaesterTitle" GENDER = male }
			}
			else = {
				set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "AcolyteTitle" GENDER = male }
			}
		}
		else_if = { # Silent Sister
			limit = { scope:silent_sister ?= yes }

			set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "SILENT_SISTER" GENDER = female }
		}
		else_if = { # Ser
			limit = { scope:knighthood ?= yes }

			set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "SER" GENDER = ALL }
		}
		else_if = { # Councillor
			limit = { scope:councillor ?= yes }

			set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "lord_non_paramount" GENDER = male }
			set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "lady_non_paramount" GENDER = female }
		}
		else_if = { # Lord/Lady (Scion Honorific)
			limit = { scope:scion_status ?= flag:active }

			set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "LORD_SCION" GENDER = male }
			set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "LADY_SCION" GENDER = female }
		}

		### FALLBACK -- A NAME, NOT A WARNING (1.0.12) ###
		# Workshop report, Buggyriot: "i dont get why this mod is giving missing title loc for all rulers added
		# by essos expanded eastern expansion." Thomas's instruction was to fix it for EVERY mod, not that one.
		#
		# STRUCTURAL, not ours. apply_character_title_data is a scripted effect, so it overrides BY NAME and
		# exactly one definition survives in a playset. It is called from
		# common/customizable_localization/00_agot_title_holder_custom_loc.txt at STRING-RENDER time, not from an
		# on_action, so there is no "after" hook and no way to chain to the copy we replaced. Any submod that adds
		# a government has the same problem in the other direction.
		#
		# What we can stop is the end of the chain printing a developer's breadcrumb into a player's UI.
		# lord_non_paramount / lady_non_paramount are AGOT's own keys, used a dozen times in this same switch.
		# An unknown ruler reads as "Lord X": right far more often than wrong, and never ugly.
		else = {
			set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "lord_non_paramount" GENDER = male }
			set_title_localization_effect = { SCOPE = $SCOPE$ KEY = "lady_non_paramount" GENDER = female }
		}
	}
	}
