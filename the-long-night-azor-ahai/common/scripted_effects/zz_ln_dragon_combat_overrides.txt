# AGOT: The Long Night & Azor Ahai -- Door Two: convert, never clone.
#
# THIS FILE OVERRIDES ONE AGOT SCRIPTED EFFECT BY KEY: `dragon_combat_result_grab`, from
# common/scripted_effects/00_agot_dragon_combat_effects.txt (line 1930 as of AGOT 2026-08). Scripted
# effects are keyed singletons and the LAST loaded definition wins -- the same mechanism Thomas's own
# Supernatural+AGOT compat patch uses for the secret effects (zzz_spn_agot_secret_effects.txt states
# it plainly: "AGOT loads first, so AGOT's versions were being discarded entirely"). This mod loads
# after AGOT, so this definition replaces theirs. database_conflicts.log will log the duplicate key
# on every boot: that line is EXPECTED and is this file working.
#
# WHY THIS EFFECT. The funnel check (13 Aug, 1.0.7): every dragon death that actually EXECUTES in
# dragon-vs-dragon combat lives inside this one effect. The four `death = { }` sites in
# dragon_combat_move_chomp_effect are wrapped in show_as_tooltip -- display only, never run; the real
# chomp kill is this effect's flag:chomp branch. dragon_combat_result_grab has exactly one caller
# (finalise_dragon_combat_results_effect), and there is no competing-entry slot in dragon combat the
# way there is in the knight lottery: the deaths are direct weighted random_list branches. So the
# smallest possible intervention is this: one effect, re-stated, with six dragon-death sites wrapped.
#
# WHAT CHANGED, EXACTLY -- and nothing else changed, byte for byte:
#   Six scope:dsc_loser_dragon death sites (sudden_strike 95, dragonfire 95, driven_to_ground,
#   disemboweled, eviscerated, chomp) now ask ln_the_dead_claim_this_trigger first. If the dead can
#   claim this kill (Long Night active, victim a living dragon and not already theirs, victor one of
#   them or the ground theirs, and the game rule not set to deny it), the death is REPLACED by
#   ln_the_dead_claim_dragon_effect -- the dragon survives as itself, converted. Otherwise the
#   original death runs untouched.
#
#   1.1.2 -- AND THE SIX RIDER DEATHS NOW GO THROUGH ln_dragon_duel_rider_lost_effect, defined
#   immediately below. For every rider in the world that effect is AGOT's death, verbatim and
#   unchanged. For the Night King it is a fall. See its own header for why.
#
# ON EVERY AGOT UPDATE: diff their dragon_combat_result_grab against the body below (minus the six
# wraps) and fold in any change. The wraps themselves are mechanical; the anchors are the six
# death_reason lines.

# ── THE RIDER WHO LOST, AND THE ONE RIDER WHO DOES NOT DIE OF IT (1.1.2) ─────────────────────────────
# Thomas, 12 Sep: "I would like to get rid of that and instead make it so that the only way for the Night
# King to die is by, you know, Valyrian steel, dragonglass, having that trial of the seven duel or
# whatever. I don't want there to be any one-offs that would kill him."
#
# A dragon duel was one of those one-offs, and a loud one: the disemboweled branch alone kills the losing
# rider on 94 of 100 rolls. The Others are first in line for a claimed dragon
# (ln_the_dead_claim_dragon_effect hands it to title:h_the_others.holder), so the Night King is the rider
# most likely to be in the sky and the one whose death there costs the most -- an unstamped death is a
# cast-down, another of them is crowned, and the player who just shot him out of the air is told the Long
# Night goes on.
#
# WHAT THIS IS. AGOT's own death, for everyone. The Night King instead comes off his mount and out of the
# fight -- which is also the scene the show gave us, and it costs the living nothing: the dragon itself is
# still resolved by the branch above this call, so shooting the ice dragon down still WORKS. What it stops
# is the rider dying as a side effect of it.
#
# NO WOUND IS APPLIED, and that is deliberate rather than mercy: a wound is the one precondition
# commander_killed needs, so wounding him here would re-open the door the commander-pool shield in
# common/combat_phase_events/zz_ln_knight_phase_events.txt was added to close.
#
# agot_remove_from_combat is AGOT's own effect and is already used ten times in this file's body, so this
# adds no new symbol -- it is the same exit the 3/2/1 branches of the same random_lists already take.
ln_dragon_duel_rider_lost_effect = {
	if = {
		limit = { has_trait = nightking }
		debug_log = "LN_TRACE NIGHT KING: thrown from a dragon and not killed by it"
		debug_log_scopes = yes
		agot_remove_from_combat = yes
	}
	else = {
		death = {
			killer = scope:dsc_victor
			death_reason = death_dragon_duel
		}
	}
}

dragon_combat_result_grab = {
	scope:dsc_loser = {
		if = {
			limit = {
				scope:dragon_death_rattle = flag:sudden_strike
			}
			random_list = {
				5 = {
					scope:dsc_loser_dragon = {
						wound_dragon = yes
						agot_remove_from_combat = yes
					}
				}
				95 = {
					scope:dsc_loser_dragon = {
						if = {
							limit = { ln_the_dead_claim_this_trigger = yes }
							ln_the_dead_claim_dragon_effect = yes
						}
						else = {
							death = {
								killer = scope:dsc_victor
								death_reason = death_dragon_duel
							}
						}
					}
				}

			ln_dragon_duel_rider_lost_effect = yes
		}

			limit = {
				scope:dragon_death_rattle = flag:dragonfire
			}
			random_list = {

						agot_remove_from_combat = yes

					}
				}
				95 = {

					}

			}

		}

			limit = {
				scope:dragon_death_rattle = flag:driven_to_ground
			}
			random_list = {

					agot_remove_from_combat = yes
				}
				1 = {
					add_trait = wounded_1
					agot_remove_from_combat = yes
				}
				2 = {
					add_trait = wounded_2
					agot_remove_from_combat = yes
				}
				3 = {
					add_trait = wounded_3
					agot_remove_from_combat = yes
				}
				90 = {
					modifier = {
						add = {
							value = prowess
							multiply = -1

					}
					# 1.1.2 -- the rider death, routed. See ln_dragon_duel_rider_lost_effect.
					ln_dragon_duel_rider_lost_effect = yes
				}
			}
			scope:dsc_loser_dragon = {
				if = {
					limit = { ln_the_dead_claim_this_trigger = yes }
					ln_the_dead_claim_dragon_effect = yes
				}
				else = {
					death = {
						killer = scope:dsc_victor
						death_reason = death_dragon_duel
					}

			}
		}

				scope:dragon_death_rattle = flag:disemboweled
			}

				if = {

					ln_the_dead_claim_dragon_effect = yes
				}
				else = {
