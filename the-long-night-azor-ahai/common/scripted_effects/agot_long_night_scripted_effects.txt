# scouring_lands_effect). All three were inherited from the ancestor mods, called by nothing in this mod,
# and unknown to AGOT -- verified by name against the whole playset. scouring_lands_effect had a totally
# EMPTY body. The two skroth effects were earlier drafts of the Others' ice dragon, superseded by
# agot_spawn_adult_ice_dragon_effect (still here, still live -- it is what the legacy
# agot_long_night_dragon_on rule calls) and by ln_raise_fallen_dragon_effect on the modern default.
# Three near-identically-named dragon spawners with only one live path is precisely the trap that had
# already produced one stale-duplicate bug in this mod, so the dead two are gone rather than annotated.
# The live scouring chain is untouched: longnight_create_other_for_scouring, longnight_spawn_army_for_
# scouring and scouring_cb are all still here.




# The Long Night+: when the Others fall, the lands the Night King held are left scoured. Ordinary holdings

# collapse into wilderness (secondary holdings razed, development wiped); the great de jure seats are instead

# left as ruins, scaled to their standing (kingdom seat -> a renowned large ruin, duchy seat -> a medium ruin).

# The living who return south inherit a ravaged, depopulated waste to reclaim through AGOT's colonization and

# ruin systems. Must run in the_long_night on_end BEFORE the wights die, so the dead are still scopeable holders.

agot_long_night_scour_others_lands_effect = {

	# 1.0.10 -- arm the post-Dawn county-memory sweep: counties this walk cannot reach (reconquered by
	# the living mid-war, or converted land the dead no longer held) get repaired by the yearly
	# aftermath pass instead. Cleared by the sweep itself after one clean pass.
	set_global_variable = { name = ln_county_memory_pending  value = yes }
	if = {
		limit = { has_global_variable = ln_county_memory_done_1 }
		remove_global_variable = ln_county_memory_done_1
	}

	# 13 Aug (1.0.6a) -- THE SURVIVORS OF THE WAR FOR THE DAWN. Thomas, post-victory: characters named
	# "Lord Other The Others" holding "The Army of the Dead of The Nightfort" as INDEPENDENT FEUDAL REALMS,
	# settling the former Wall lands with ordinary human portraits. He checked: they had no wight trait.
	# They were of house_theothers, and this limit only ever asked about the two traits, so the scour
	# walked straight past them and destroy_others_kingdom_titles (which only touches the Night King's own
	# titles, and only above county tier) never saw them either. Between the two holes, every dead lord
	# who was not personally trait-marked kept his county forever.
	#
	# ln_is_one_of_them_trigger is the mod's existing answer to "is this one of the dead" -- traits,
	# the Night King, the house, and now wight_government. Ruins are excluded because they are already
	# the thing this effect turns land into.
	#
	# AND THE LAND BECOMES RUINS, NOT A GRANT. Thomas's ruling: "those lands should just become ruins...
	# that way the wall can retake them over. Because if you tried to manually do it, you'd end up giving
	# all of the lands the Long Night conquered to the Night's Watch, and I definitely want to avoid that."
	# So nothing is handed to anybody -- the Watch, the North and the Crown retake it on their own clock
	# through the reconquest and colonisation systems that already exist.
	every_county = {

		limit = {

			exists = holder

			holder = {

				ln_is_one_of_them_trigger = yes

				NOT = { has_government = ruins_government }

			}

		}

		save_scope_as = scour_county

		# 1.0.10 -- give the land its memory back before it becomes a ruin. Culture and faith survive
		# ruination, so what this writes is what the colonists find. Stamped counties restore exactly;
		# unstamped converted ones fall back to the de jure kingdom's living holder; beyond the Wall
		# nothing resolves and the winter stays. See zz_ln_county_memory_effects.txt.
		ln_restore_county_memory_effect = yes



		if = {

			# De jure seat of a kingdom -> a renowned great ruin (e.g. Winterfell, Storm's End).

			limit = {

				de_jure_liege ?= {

					de_jure_liege ?= {

						tier = tier_kingdom

						title_capital_county = scope:scour_county

					}

				}

			}

			agot_ruin_county_effect = yes

			title_province = {

				remove_building = medium_ruin_01

				add_building = large_ruin_01

				add_building = ruin_renowned_01

			}

			ln_mark_scoured_ruin_effect = yes

		}

		else_if = {

			# De jure seat of a duchy -> a sizeable ruin.

			limit = {

				de_jure_liege ?= {

					tier = tier_duchy

					title_capital_county = scope:scour_county

				}

			}

			agot_ruin_county_effect = yes

			ln_mark_scoured_ruin_effect = yes

		}

		else = {

			# Everything else simply returns to the wild.

			make_settlement_county_wilderness = { COUNTY = this }

		}

	}

}

# for all of them including him, and only the fealty card is withheld from the one man who cannot swear to

# the scope change itself optional, so it is correct whatever order the engine picks. It also moved the

# `exists = this.capital_barony` beside a bare `this.capital_barony.title_province` deref; that is the
# self-refuting shape this release corrected everywhere it appeared.)

# Landed variant (used by the Flee South decision): a lord abandons their seat ahead of the dead.

# Root = a domicile. Add a tier-2 camp building, putting its tier-1 in first if it is missing.
#
# 10 Aug, off the live log: flee_noble_south_effect force-added six tier-2 buildings directly and the engine
# refused every one -- "Cannot construct an upgrade when previous building has not been built", ~151 per run.
# add_domicile_building ignores cost and construction time but it does NOT ignore the prerequisite chain, and
# each of the six declares one in common/domiciles/00_camp_buildings.txt:
#   supply_tent_reserve_provisions <- supply_tent_01      mess_tent_cooks              <- mess_tent_01
#   camp_fire_local_hangers_on     <- camp_fire_01        baggage_train_porters        <- baggage_train_01
#   proving_grounds_training_circle<- proving_grounds_01  barber_tent_surgeons_tools   <- barber_tent_01
#
# So the fled lord's camp was getting NOTHING, silently, in every game -- the errors were the only trace of a
# feature that never ran. Guard-then-add is vanilla's own idiom (00_administrative_effects.txt:430).
# has_domicile_building_or_higher rather than has_domicile_building: the plain form misses upgraded tiers and
# would re-add a base the lord has already outgrown.
# 1.0.12 -- TWO MORE WAYS TO FAIL, BOTH MEASURED, 14 PER RUN. The 10 Aug guard below (re-test the base
# rather than trust the add) was right and stays. It just did not cover everything add_domicile_building
# refuses to do:
#
#   7x  Domicile owner failed to meet triggered requirements: 'camp_fire_local_hangers_on'
#   5x  Domicile owner failed to meet triggered requirements: 'supply_tent_reserve_provisions'
#   2x  Domicile has no free building slots: 'barber_tent_...'
#
# THE FIRST TWO ARE THE UPGRADE'S OWN can_construct, AND IT IS EXACTLY TWO. Of the six pairs this effect is
# called with, only these two carry a can_construct at all, and both carry the same one -- vanilla
# common/domiciles/buildings/00_camp_buildings.txt, camp_fire_local_hangers_on:4185 and
# supply_tent_reserve_provisions: a custom_tooltip wrapping `has_realm_law_flag = unlocks_<key>`, i.e. the
# camp's PURPOSE has to have unlocked that building. mess_tent_cooks, baggage_train_porters,
# proving_grounds_training_circle and barber_tent_surgeons_tools have no can_construct block, and no
# unlocks_ flag exists for them anywhere in common/laws -- so a generic `has_realm_law_flag =
# unlocks_$UPGRADE$` would have been worse than the bug, silently blocking the four that were working.
# The gate is therefore passed IN, one trigger per call site, defaulting to `always = yes`.
# It is checked on `owner` because that is who the engine tested -- "Domicile OWNER failed to meet".
#
# THE THIRD IS SLOTS, AND ONLY THE BASE NEEDS ONE. An upgrade replaces its base in the slot it already
# occupies (previous_building = camp_fire_01 and friends), so it needs no free slot of its own; a base does.
# domicile_building_has_free_internal_slot takes the MAIN building of the domicile, not the building being
# added -- vanilla's own idiom in add_random_internal_japanese_manor_building
# (common/scripted_effects/10_dlc_tgp_japan_scripted_effects.txt:429), which passes japanese_manor_main_01
# even for a manor sitting at main_03. The camp equivalent is camp_main_01.
#
# WHEN AGOT OR VANILLA TOUCH 00_camp_buildings.txt, RE-CHECK WHICH OF THE SIX CARRY A can_construct. A new
# gate upstream shows up here as this error coming back, not as a build failure.
ln_camp_add_upgrade_effect = {
	if = {
		limit = {
			NOT = { has_domicile_building_or_higher = $BASE$ }
			domicile_building_has_free_internal_slot = camp_main_01
		}
		add_domicile_building = $BASE$
	}
	# 10 Aug -- ONLY REACH FOR THE UPGRADE IF THE BASE ACTUALLY LANDED. add_domicile_building force-adds but it
	# SILENTLY NO-OPS when the camp has no free slot, and the upgrade behind it then throws "Cannot construct
	# an upgrade when previous building has not been built". Both halves were in the live log
	# (barber_tent_01 / barber_tent_surgeons_tools, camp_fire_01 / camp_fire_local_hangers_on). Testing the
	# base again rather than trusting the add above is the whole fix.
	if = {
		limit = {
			has_domicile_building_or_higher = $BASE$
			NOT = { has_domicile_building_or_higher = $UPGRADE$ }
			owner = { $GATE$ = $GATE_VALUE$ }
		}
		add_domicile_building = $UPGRADE$
	}
}

		}
