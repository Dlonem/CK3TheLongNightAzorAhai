# AGOT: The Long Night & Azor Ahai -- 1.0.15a
# SAME-NAME OVERRIDE of AGOT's set_title_localization_effect (00_agot_character_data_effects.txt).
# TWIN FILE: an identical copy sits in replace/ so ours parses last under either db-ordering theory.
#
# =============================== WHAT THIS REPLACED, AND WHY ===============================
#
# Until 1.0.15 this mod overrode apply_character_title_data instead -- AGOT's 2,548-line
# government/tier title switch -- with our three governments dispatched first and AGOT's whole body
# copied verbatim below an else. It worked, and it was wrong, for a reason nothing we measured could
# see: scripted effects override BY NAME, exactly one definition survives a playset, and OURS parsed
# last. Every other submod that writes into that same switch had its work silently discarded.
#
# Measured 26 Aug 2026 against the installed mods, not from theory:
#   AGOT Nobility of Westeros 1.2.8   +512 lines, 36 title keys (lord_paramount, king_the_neck,
#                                     landed_knight_master, lord_commander_of_the_city_watch, ...)
#   Essos Expanded: Further East 4.0  +278 lines, 70 title keys AND a whole government, oligarchic
#                                     (ee_god_emperor_male, ee_khal_of_khals_male, ee_yellow_emperor,
#                                     king_asshai_male, hegemon_valyrian_male, ...)
#   Legacy of Valyria - AGOT 0.5.1    32 title keys, same oligarchic government -- the copy EE:FE
#                                     builds on
# All of it thrown away by us. EE:FE's rulers hold oligarchic, AGOT's chain has no branch for it, so
# they fell to the chain's terminal else and rendered "#warning Missing Title Loc#!". Reported by
# MAGNUM on the Workshop 25 Aug 2026, and diagnosed correctly by EE:FE's own developer.
#
# =================================== WHAT THIS DOES NOW ====================================
#
# apply_character_title_data writes exactly ONE thing: scope:get_title, and only through
# set_title_localization_effect -- verified as the sole writer of that scope value in all of AGOT, and
# AGOT is the only mod that defines it (Nobility of Westeros, EE:FE and LoV each ship a copy of
# apply_character_title_data; none of them redefine this).
#
# So we stopped fighting for the switch and took the pen instead. Whoever owns
# apply_character_title_data -- AGOT, EE:FE, Nobility of Westeros, anyone -- runs their own chain with
# every branch intact, and each key they choose passes through here. If the character is one of OURS we
# write our name instead. If not, AGOT's original body runs untouched, verbatim, below the else.
#
#   * every submod's title branches survive; no bridge patch needed from anyone
#   * our wight, ranger and wilderness rulers keep their names in every playset
#   * we shadow 25 lines of AGOT instead of 2,548 -- the stale-fork risk of session 27c is gone
#
# ====================================== MAINTENANCE ========================================
#
# RE-DIFF THE else BLOCK AGAINST AGOT ON EVERY AGOT UPDATE. It is a verbatim copy of AGOT's
# set_title_localization_effect and nothing else. If AGOT ever changes how get_title is written, this
# is the one place that has to follow.
#
# If a future mod redefines set_title_localization_effect as well, ours still parses last (the replace/
# twin) and theirs would be the one lost. Watch that entry in logs/database_conflicts.log.

set_title_localization_effect = {
	### THE RISEN ###
	if = {
		limit = {
			exists = $SCOPE$
			$SCOPE$ ?= { has_government = wight_government }
		}
		if = {
			limit = { $SCOPE$ ?= { ln_named_title_tier_trigger = { TIER = tier_barony } } }
			ln_stamp_title_effect = { SCOPE = $SCOPE$ MALE = baron_wight_male FEMALE = baron_wight_female }
		}
		else_if = {
			limit = { $SCOPE$ ?= { ln_named_title_tier_trigger = { TIER = tier_county } } }
			ln_stamp_title_effect = { SCOPE = $SCOPE$ MALE = count_wight_male FEMALE = count_wight_female }
		}
		else_if = {
			limit = { $SCOPE$ ?= { ln_named_title_tier_trigger = { TIER = tier_duchy } } }
			ln_stamp_title_effect = { SCOPE = $SCOPE$ MALE = duke_wight_male FEMALE = duke_wight_female }
		}
		else_if = {
			limit = { $SCOPE$ ?= { ln_named_title_tier_trigger = { TIER = tier_kingdom } } }
			ln_stamp_title_effect = { SCOPE = $SCOPE$ MALE = king_wight_male FEMALE = king_wight_female }
		}
		else_if = {
			limit = { $SCOPE$ ?= { ln_named_title_tier_trigger = { TIER = tier_empire } } }
			ln_stamp_title_effect = { SCOPE = $SCOPE$ MALE = emperor_wight_male FEMALE = emperor_wight_female }
		}
		else_if = {
			limit = { $SCOPE$ ?= { ln_named_title_tier_trigger = { TIER = tier_hegemony } } }
			ln_stamp_title_effect = { SCOPE = $SCOPE$ MALE = emperor_wight_male FEMALE = emperor_wight_female }
		}
		else = {
			# No title of any tier to read -- landless, or being named through scope:dummy_gender.
			# AGOT's own generic honorific, not a rank this character does not hold.
			ln_stamp_title_effect = { SCOPE = $SCOPE$ MALE = lord_non_paramount FEMALE = lady_non_paramount }
		}
	}

	### THE RANGERS ###
	else_if = {
		limit = {
			exists = $SCOPE$
			$SCOPE$ ?= { has_government = ln_ranger_government }
		}
		ln_stamp_title_effect = { SCOPE = $SCOPE$ MALE = duke_ln_ranger_male FEMALE = duke_ln_ranger_female }
	}

	### THE WILDERNESS ###
	# AGOT's own placeholder government, with no naming branch anywhere in AGOT -- so seventeen counties
	# in a normal game, and the Night King's respawned court in a second Long Night, rendered the warning
	# string. All ten keys already ship in AGOT's agot_government_l_english.yml, valued "Warden"; they
	# were simply unreachable.
	else_if = {
		limit = {
			exists = $SCOPE$
			$SCOPE$ ?= { agot_localization_government_trigger = { GOV = wilderness SPOUSE_SCOPE = scope:spouse_scope } }
		}
		if = {
			limit = { $SCOPE$ ?= { ln_named_title_tier_trigger = { TIER = tier_hegemony } } }
			ln_stamp_title_effect = { SCOPE = $SCOPE$ MALE = hegemon FEMALE = hegemon_female }
		}
		else_if = {
			limit = { $SCOPE$ ?= { ln_named_title_tier_trigger = { TIER = tier_empire } } }
			ln_stamp_title_effect = { SCOPE = $SCOPE$ MALE = emperor_wilderness_male FEMALE = emperor_wilderness_female }
		}
		else_if = {
			limit = { $SCOPE$ ?= { ln_named_title_tier_trigger = { TIER = tier_kingdom } } }
			ln_stamp_title_effect = { SCOPE = $SCOPE$ MALE = king_wilderness_male FEMALE = king_wilderness_female }
		}
		else_if = {
			limit = { $SCOPE$ ?= { ln_named_title_tier_trigger = { TIER = tier_duchy } } }
			ln_stamp_title_effect = { SCOPE = $SCOPE$ MALE = duke_wilderness_male FEMALE = duke_wilderness_female }
		}
		else_if = {
			limit = { $SCOPE$ ?= { ln_named_title_tier_trigger = { TIER = tier_county } } }
			ln_stamp_title_effect = { SCOPE = $SCOPE$ MALE = count_wilderness_male FEMALE = count_wilderness_female }
		}
		else_if = {
			limit = { $SCOPE$ ?= { ln_named_title_tier_trigger = { TIER = tier_barony } } }
			ln_stamp_title_effect = { SCOPE = $SCOPE$ MALE = baron_wilderness_male FEMALE = baron_wilderness_female }
		}
		else = {
			ln_stamp_title_effect = { SCOPE = $SCOPE$ MALE = lord_non_paramount FEMALE = lady_non_paramount }
		}
	}

	### EVERYONE ELSE -- AGOT'S ORIGINAL BODY, VERBATIM ###
	else = {
		if = {
			limit = {
				exists = flag:$KEY$
				switch = {
					trigger = flag:$GENDER$
					flag:male ?= { $SCOPE$ ?= { is_male = yes } }
					flag:female ?= { $SCOPE$ ?= { is_female = yes } }
					fallback = { always = yes }
				}
			}
			if = {
				limit = { $SCOPE$ ?= { is_male = yes } }
				save_scope_value_as = {
					name = get_title
					value = flag:$KEY$
				}
			}
			else_if = {
				limit = { $SCOPE$ ?= { is_female = yes } }
				save_scope_value_as = {
					name = get_title
					value = flag:$KEY$
				}
			}
		}
	}
}
