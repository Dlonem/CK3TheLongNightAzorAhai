# AGOT: The Long Night & Azor Ahai -- 1.0.15, THE ENDING FUNNEL.
#
# Thomas's design rule, 24 Aug: "the war for the dawn only ends if the night king dies." Before this file,
# the mod had four ways for the Long Night to end and only ONE of them (the open-battle killing blow) actually
# ran the ending. The other three -- a war-score victory, killing him mid-war, imprisoning him -- either
# invalidated the war, crowned a replacement, or stalled forever. Every road now funnels into the same place:
# the Night King dies, on screen, by a named hand, and the catalyst chain (zz_ln_catalyst_effects.txt) does
# what it has always done from there. This file adds NO second resolution path -- it only delivers candidates
# to the death that already knows how to resolve itself.
#
# THE ORDERING RULE APPLIES THROUGHOUT: nothing irreversible happens until every value that depends on the
# pre-change state has been read and banked. The death is always the LAST line of whichever chain runs.

# ── THE CORNERED KING ───────────────────────────────────────────────────────────────────────────────────────
# Called from war_for_the_dawn_cb's on_victory, and only when the defender is still alive -- i.e. the war was
# won on score, not by a blow. scope:war, scope:attacker and scope:defender are the CB's own handles (the
# on_defeat block three inches down the same file already uses scope:war, so the shape is proven).
#
# WHO IS IN THE ROOM (checklist #2): not just the war's rulers. Every attacker AND every attacker's knights --
# Jon Snow sworn to somebody else's banner is exactly the man this exists for.
ln_wfd_cornered_effect = {
	scope:defender = { save_scope_as = ln_night_king }

	scope:war ?= {
		every_war_attacker = {
			# 1.0.16 -- DRAGON GUARD. Mirrors the fix 1.0.12 already shipped for the analogous preparation
			# loops (zz_ln_preparation_effects.txt:98-117), which was never carried over to the funnel.
			# Without it a high-prowess dragon can win the worthiness ordering, be handed ln_dawn.11, pick
			# option C ("We have nothing that can end him"), tear down ln_dawn_cornered_king at :655 and
			# LEAVE THE LONG NIGHT UNENDABLE. Thomas: "ideally, maybe dragons can't be eligible Dawn
			# Champions... it'd be nice if whoever kills the Night King had a dragon, that that dragon gets
			# honorable mention." The honorable mention already exists and is untouched -- the dragon is
			# named in the permanent record by ln_last_dawn_dragon (zz_ln_reveal_effects.txt:88-99).
			# The animal is credited; it just does not hold the sword.
			limit = { is_alive = yes  is_adult = yes  NOT = { ln_is_a_dragon_trigger = yes } }
			add_to_temporary_list = ln_dawn_candidates
		}
	}

	# 1.0.16 -- THE KNIGHTS ARE A SIBLING LOOP NOW, NOT A NESTED ONE, AND THAT IS THE BUG THIS FIXES.
	# every_knight used to sit INSIDE the limited body above. `limit` in an every_ loop decides which
	# iterations run the body at all -- so a war attacker who was dead, a child, or a dragon did not merely
	# fail to enter the list himself, HE TOOK HIS ENTIRE KNIGHT ROSTER WITH HIM. That is the precise
	# opposite of what the header of this effect promises: "not just the war's rulers... Jon Snow sworn to
	# somebody else's banner is exactly the man this exists for." Jon Snow sworn to a nine-year-old lord, or
	# to a lord who died in the last battle of the war, was silently not a candidate.
	# The 1.0.12 pattern this file's dragon guard cites uses sibling loops for exactly this reason --
	# zz_ln_preparation_effects.txt:98-116 runs every_knight and every_courtier as two separate loops, each
	# carrying its own limit. Same shape here: the attacker is filtered on his own merits, his knights on
	# theirs, and neither decides for the other.
	# STRICTLY WIDENS the candidate pool. Nobody who qualified before stops qualifying.
	scope:war ?= {
		every_war_attacker = {
			every_knight = {
				limit = { is_alive = yes  is_adult = yes  NOT = { ln_is_a_dragon_trigger = yes } }
				add_to_temporary_list = ln_dawn_candidates
			}
		}
	}

	ordered_in_list = {
		list = ln_dawn_candidates
		order_by = ln_dawn_worthiness_value
		position = 0
		save_scope_as = ln_dawn_champion
	}
	ordered_in_list = {
		list = ln_dawn_candidates
		order_by = ln_dawn_worthiness_value
		position = 1
		save_scope_as = ln_dawn_second
	}

	if = {
		limit = { exists = scope:ln_dawn_champion }
		debug_log = "LN_TRACE ENDGAME: war won on score - the destiny scene picks the hand"
		debug_log_scopes = yes
		# Variables, not just scopes -- the handshake pattern. The event is days away and the variable
		# survives anything short of the referent dying, which the event's own trigger re-checks.
		set_global_variable = { name = ln_dawn_cornered_king  value = scope:ln_night_king }
		set_global_variable = { name = ln_dawn_final_champion  value = scope:ln_dawn_champion }
		if = {
			limit = { exists = scope:ln_dawn_second  NOT = { scope:ln_dawn_second = scope:ln_dawn_champion } }
			set_global_variable = { name = ln_dawn_second_champion  value = scope:ln_dawn_second }
		}
		scope:ln_dawn_champion = { trigger_event = { id = ln_dawn.11  days = 2 } }
		# THE FAILSAFE IS PART OF THE PROMISE. If the scene is lost -- the champion dies on the road, an
		# event queue hiccup, anything -- the king does not get to walk away from a war he lost. Day 14,
		# hidden, idempotent -- and queued at BOTH the leader and the champion (review find: a leader who
		# also dies inside the fortnight used to take the failsafe to his grave). Whoever still breathes
		# finishes it; the second copy no-ops on the guards.
		# ── 1.0.16 -- THE DAYS WERE THE WRONG WAY ROUND, AND IT COST THE RECORD. ────────────────────────
		# The war leader's copy was day 14 and the CHAMPION'S was day 15, so the leader's always landed
		# first -- and ln_dawn.12's own teardown removes ln_dawn_cornered_king, which disarms the
		# champion's copy through its trigger. Result: whenever the champion's ln_dawn.11 was lost in the
		# intervening fortnight, ln_dawn.12 fired at scope:attacker and `killer = root` recorded THE LIEGE
		# WHO DECLARED THE WAR -- in the reveal card, in the roll-call, and in the permanent
		# ln_last_dawn_champion global that a Cycle run reads a century later.
		# Thomas asked exactly this: "you have root in a lot of the killer for the Night King, and I wanna
		# make sure that applies to the person who's actually killing the Night King versus just me."
		# The whole funnel exists to find the man the story picked -- "Jon Snow sworn to somebody else's
		# banner is exactly the man this exists for" (:20) -- and the failsafe was quietly overruling it.
		# Swapped: the champion gets day 14, the war leader day 16. The leader is still the backstop for a
		# champion who dies inside the fortnight; he is no longer the default answer.
		scope:ln_dawn_champion = { trigger_event = { id = ln_dawn.12  days = 14 } }
		scope:attacker = { trigger_event = { id = ln_dawn.12  days = 16 } }
	}
	else = {
		# Nobody fit to swing. The old resolution: the war leader takes the kill.
		# (Known and accepted: this death fires the catalyst sweep DURING on_victory, and the sweep may
		# touch this same concluding war. Every guard on that path is idempotent -- the global is already
		# set, on_victory's living-defender gate fails -- so the second pass is a no-op. Reviewed 1.0.15.)
		#
		# ── 1.1.3: THIS KILL OBEYS THE BANE RULE AND CARRIES THE STAMP, LIKE EVERY OTHER SANCTIONED ONE. ──
		# 1.1.2 open item 8. This was `death = { death_reason = death_battle  killer = scope:attacker }` with
		# no ln_mark_the_sanctioned_kill_effect, and since 1.1.0b's flag gate every unstamped death casts him
		# down -- so a War for the Dawn WON on score, with nobody eligible to face him (every attacker dead,
		# a child or a dragon, no adult knights), crowned another Other instead of ending the dark. Mirrors
		# ln_dawn.12 (zz_ln_dawn_events.txt): the leader's hand is tested; if it bears the bane the stamp goes
		# on and the blow lands as the ending it is; if not, he is not killed at all -- the victory global
		# comes down and the king is left standing, exactly as the failsafe's baneless branch leaves him.
		if = {
			limit = { scope:attacker = { ln_bears_the_bane_trigger = yes } }
			debug_log = "LN_TRACE ENDGAME: war won on score but no candidate found - the war leader bears the bane and resolves it"
			scope:ln_night_king = {
				ln_mark_the_sanctioned_kill_effect = yes
				death = {
					death_reason = death_the_dawn
					killer = scope:attacker
				}
			}
		}
		else = {
			debug_log = "LN_TRACE ENDGAME: war won on score, no candidate, and no bane in the leader's hand - the king is left standing"
			if = { limit = { has_global_variable = war_for_dawn_victory } remove_global_variable = war_for_dawn_victory }
		}
	}
}

# ── THE KING IN CHAINS ──────────────────────────────────────────────────────────────────────────────────────
# Called from the on_imprison hook when the prisoner is the Night King. Root = the Night King (the prisoner),
# scope:imprisoner = the captor. Candidates: the captor and the swords standing in his hall -- his knights,
# his courtiers, his guests. The realm that put the King of Winter in a cell gets to decide which of its own
# ends him; that is the execution chain Thomas picked, with the destiny layer choosing the hand.
ln_nk_captured_effect = {
	save_scope_as = ln_night_king
	# He does not get to ask for trial by combat, ransom himself, or charm a jailer. AGOT's own prison
	# interactions all hide for a prisoner carrying this flag; we use their door rather than shadowing
	# five interaction files.
	add_character_flag = keep_imprisoned

	scope:imprisoner = {
		save_scope_as = ln_nk_captor
		# 1.0.16 -- THE CAPTOR WAS THE ONE CANDIDATE WITH NO TEST ON HIM AT ALL. Every knight and courtier
		# below is filtered; he walked straight in. Two consequences, and the second is the quiet one:
		#   * a dragon captor could win the worthiness ordering and be handed ln_dawn.11 -- the single
		#     failure this file's dragon guard exists to prevent, left open at the one entry that skipped it;
		#   * because he was ALWAYS added, ln_dawn_candidates was never empty, ordered_in_list always found
		#     a position 0, and the else-branch below ("a landless wretch with no knights") was dead code
		#     that could not run. It has a real job again now.
		#
		# is_adult IS DELIBERATELY NOT IN THIS LIMIT, unlike the knights' and courtiers'. A child captor
		# must stay eligible: he is the last resort in his own cell, and ln_dawn.11 has no is_adult in its
		# trigger (zz_ln_dawn_events.txt:614-626), so handing him the card ends the Long Night. The
		# worthiness ordering will still prefer any adult standing in his hall. The dragon is the only
		# exclusion that is load-bearing here, because a dragon can break off and leave the dark unendable;
		# a nine-year-old lord cannot.
		if = {
			limit = { is_alive = yes  NOT = { ln_is_a_dragon_trigger = yes } }
			add_to_temporary_list = ln_dawn_candidates
		}
		# 1.0.16 -- same dragon guard as ln_wfd_cornered_effect above; see the note there. A captor's hall
		# can contain a bonded dragon, and a dragon cannot be handed the executioner's card.
		every_knight = {
			limit = { is_alive = yes  is_adult = yes  NOT = { ln_is_a_dragon_trigger = yes } }
			add_to_temporary_list = ln_dawn_candidates
		}
		every_courtier_or_guest = {
			limit = { is_alive = yes  is_adult = yes  is_imprisoned = no  NOT = { ln_is_a_dragon_trigger = yes } }
			add_to_temporary_list = ln_dawn_candidates
		}
	}

	ordered_in_list = {
		list = ln_dawn_candidates
		order_by = ln_dawn_worthiness_value
		position = 0
		save_scope_as = ln_dawn_champion
	}
	ordered_in_list = {
		list = ln_dawn_candidates
		order_by = ln_dawn_worthiness_value
		position = 1
		save_scope_as = ln_dawn_second
	}

	debug_log = "LN_TRACE ENDGAME: the night king is in chains - execution chain begins"
	debug_log_scopes = yes
	set_global_variable = { name = ln_dawn_cornered_king  value = scope:ln_night_king }
	set_global_variable = { name = ln_dawn_in_chains  value = yes }
	if = {
		limit = { exists = scope:ln_dawn_champion }
		set_global_variable = { name = ln_dawn_final_champion  value = scope:ln_dawn_champion }
		if = {
			limit = { exists = scope:ln_dawn_second  NOT = { scope:ln_dawn_second = scope:ln_dawn_champion } }
			set_global_variable = { name = ln_dawn_second_champion  value = scope:ln_dawn_second }
		}
		scope:ln_dawn_champion = { trigger_event = { id = ln_dawn.11  days = 3 } }
		# Review find: the failsafe rides both men, so one death cannot orphan the chain.
		# 1.0.16 -- day 15 -> 14, ahead of the captor's copy below. Same defect and same reasoning as the
		# score-victory route: the captor's day-14 copy used to land first and record HIM as the killer,
		# overruling the destiny ordering that had already named someone in his hall.
		scope:ln_dawn_champion = { trigger_event = { id = ln_dawn.12  days = 14 } }
	}
	else = {
		# A landless wretch with no knights captured the King of Winter. His cell, his blade.
		# 1.0.16 -- REACHABLE FOR THE FIRST TIME (see the captor's limit above), so it now has to be right.
		# Guarded with the same test, because the only ways to arrive here are an empty hall -- in which
		# case the captor is fine and this is the intended ending -- or a captor who failed that test, in
		# which case handing him the card is precisely the failure the test was added to stop.
		if = {
			limit = { scope:ln_nk_captor = { is_alive = yes  NOT = { ln_is_a_dragon_trigger = yes } } }
			set_global_variable = { name = ln_dawn_final_champion  value = scope:ln_nk_captor }
			scope:ln_nk_captor = { trigger_event = { id = ln_dawn.11  days = 3 } }
		}
		else = {
			# Nobody in the world can hold the sword. Not a stall: ln_dawn_cornered_king and
			# ln_dawn_in_chains are both set above, the king carries keep_imprisoned, and the day-16
			# ln_dawn.12 failsafe at the foot of this effect still runs -- so the chain resolves through the
			# backstop rather than through a champion. Logged loudly because if this line ever appears in a
			# real run, the reachability reasoning above was wrong and wants revisiting.
			debug_log = "LN_TRACE ENDGAME: the night king is in chains and NOBODY is eligible to end him - falling through to the day-16 failsafe"
			debug_log_scopes = yes
		}
	}
	# 1.0.16 -- day 14 -> 16. In the else-branch above the captor IS the champion and this is his own
	# failsafe; in the if-branch he is only the backstop for a champion who died in the fortnight.
	scope:ln_nk_captor = { trigger_event = { id = ln_dawn.12  days = 16 } }
}

# ── THE COLD THING IN YOUR CELLS ────────────────────────────────────────────────────────────────────────────
# A captured Other who is NOT the Night King. Root = the prisoner, scope:imprisoner = the captor.
# breakfastenjoyer's report made the case: "I had a court full of immortal others that served as my knights."
# The LotR Realms in Exile pattern he pointed at is the right one -- a thing of the cold does not sit in a
# cell awaiting ransom, and it certainly does not take up a knight's banner afterwards.
ln_other_captured_effect = {
	save_scope_as = ln_captured_other
	add_character_flag = keep_imprisoned
	debug_log = "LN_TRACE: an Other has been imprisoned - the captor is told what he is holding"
	scope:imprisoner = { trigger_event = { id = ln_dawn.20  days = 2 } }
}

# ── THE CAST-DOWN BACKSTOP ──────────────────────────────────────────────────────────────────────────────────
# 1.0.15c. The bane rule means a Night King killed by ordinary steel is cast down, not destroyed, and the
# dark is supposed to crown another. Two handlers can do that -- the story cycle's on_owner_death (which
# reads global_var:ln_nk_cast_down) and this. Which of them gets there first is an ENGINE ORDERING QUESTION
# WE HAVE NOT TESTED, and under others_death_resolution -- the default rule -- losing that race destroys the
# story outright and stops the invasion with no dawn to show for it.
#
# So this does not care who won the race. Once a year, if a cast-down was recorded and the world no longer
# has a Night King in it while the struggle still says the Long Night is running, the crown is put back on
# somebody and the invasion resumes. Root = anything; called from ln_struggle_yearly_work_effect.
#
# COST WHEN NOTHING IS WRONG: one global read a year. The character scan behind it only ever runs in the
# window between a cast-down and a successor existing, which on the healthy path is zero.
# NEVER RUNS IF: no cast-down was recorded, the struggle is past the Long Night, or a Night King is already
# standing -- which is the normal case the instant either handler does its job.
ln_cast_down_backstop_effect = {
	if = {
		limit = {
			has_global_variable = ln_nk_cast_down
			struggle:ln_long_night_struggle ?= { is_struggle_phase = ln_phase_the_long_night }
			NOT = { any_living_character = { has_trait = nightking } }
		}
		# 1.1.3 -- read the fallen king before the handle comes down; the card below names him.
		global_var:ln_nk_cast_down ?= { save_scope_as = ln_dead_nightking }
		remove_global_variable = ln_nk_cast_down

		# The same successor ladder the story cycle uses: the holder of the Others' realm, else any Other
		# alive anywhere. Nothing is invented here.
		if = {
			limit = {
				exists = title:h_the_others.holder
				title:h_the_others.holder = { has_trait = other_trait }
			}
			title:h_the_others.holder = { save_scope_as = ln_next_nightking }
		}
		else = {
			random_living_character = {
				# 1.1.0b -- their own house first, anyone of them second. Same reason as the story cycle's
				# two ladders: a whole-world scan was crowning Others who rule realms of their own.
				limit = {
					has_trait = other_trait
					house ?= house:house_theothers
				}
				alternative_limit = { has_trait = other_trait }
				save_scope_as = ln_next_nightking
			}
		}

		if = {
			limit = { exists = scope:ln_next_nightking }
			debug_log = "LN_TRACE BANE BACKSTOP: cast down and nobody wearing the crown - putting it back on"
			debug_log_scopes = yes
			scope:ln_next_nightking = {
				# AUDIT FIX: one crowning, shared. See ln_crown_the_next_nightking_effect.
				ln_crown_the_next_nightking_effect = yes
				# And a story to drive him, if the race was lost and the old one was destroyed. AGOT's own
				# type name, used exactly as agot_long_night_scripted_effects.txt:777 uses it.
				if = {
					# AUDIT FIX -- THE SEARCH HAS TO BE GLOBAL. any_owned_story is correct, but run inside
					# scope:ln_next_nightking it only asks whether THIS character owns a Long Night story.
					# A story still owned by anybody else was invisible to it, and the backstop would have
					# created a SECOND long_night_story alongside the live one -- two invasions driving the
					# same world. The in-tree form it was modelled on (zz_ln_independence_war_shadow.txt:118)
					# wraps it in any_ruler for exactly this reason, and that is now copied properly.
					limit = {
						NOT = {
							any_ruler = {
								any_owned_story = { story_type = long_night_story }
							}
						}
					}
					debug_log = "LN_TRACE BANE BACKSTOP: the story was destroyed in the race - starting a new one"
					create_story = { type = long_night_story }
				}
			}
			# 1.1.3 -- and every player is told. See ln_send_the_cast_down_card_effect below.
			ln_send_the_cast_down_card_effect = yes
		}
		else = {
			debug_log = "LN_TRACE BANE BACKSTOP: cast down, and there is no Other left alive to take the crown"
		}
	}
}

# ── 1.1.3: THE CARD THAT SAYS HE IS NOT DESTROYED ──────────────────────────────────────────────────────────
# Root = anything (the story, or the yearly pulse). scope:ln_next_nightking = the man just crowned (must
# exist); scope:ln_dead_nightking = the one who fell (may be absent on a strange road). Since he rises again
# in the same face (ln_the_night_king_rises_again_effect) the card shows only the risen king; the fallen one
# is still banked in ln_dawn30_fallen for any later text that wants him. Called from the two places every cast-down crowning goes
# through: the story's cast-down branch (the_long_night.txt) and the backstop above. The three handles go
# into globals because ln_dawn.30 is queued a day out and a queued event carries no scopes; the killer was
# banked by the catalyst in on_death (ln_nk_cast_down_by) and is consumed here. every_player with a one-day
# delay is the idiom the invasion opening uses (zz_ln_invasion_opening_effects.txt:106).
ln_send_the_cast_down_card_effect = {
	if = {
		limit = { exists = scope:ln_dead_nightking }
		set_global_variable = { name = ln_dawn30_fallen  value = scope:ln_dead_nightking }
	}
	else_if = {
		limit = { has_global_variable = ln_dawn30_fallen }
		remove_global_variable = ln_dawn30_fallen
	}
	set_global_variable = { name = ln_dawn30_new  value = scope:ln_next_nightking }
	if = {
		limit = { exists = global_var:ln_nk_cast_down_by }
		set_global_variable = { name = ln_dawn30_killer  value = global_var:ln_nk_cast_down_by }
		remove_global_variable = ln_nk_cast_down_by
	}
	else_if = {
		limit = { has_global_variable = ln_dawn30_killer }
		remove_global_variable = ln_dawn30_killer
	}
	debug_log = "LN_TRACE BANE: the cast-down card goes out to every player"
	every_player = {
		trigger_event = { id = ln_dawn.30  days = 1 }
	}
}
