# The Long Night+ — AFTERMATH (resettling) effects.
# Design doc: dev/the_long_night_plus_aftermath.md (sibling, not shipped).
# Everything rides AGOT's own colonization engine (ai_colonization_effect) — we only generate the colonists,
# fund them, and swear them to the Crown.

# ------------------------------------------------------------------------------------------------------------
# ENGINE WRAPPER — generate a lord of the given culture+faith, settle him on a wilderness county, vassalize him.
#   WILDERNESS  = county title scope (must be wilderness)
#   CULTURE     = culture:x  |  FAITH = faith:y  |  GOVERNMENT = government key
#   LIEGE       = character scope to swear fealty to (a non-existing scope skips vassalization → independent)
# Leaves the new lord saved as scope:ln_new_colonist for the caller.
ln_settle_generated_lord_effect = {
	create_character = {
		age = { 25 45 }
		gender = male
		culture = $CULTURE$
		faith = $FAITH$
		location = $WILDERNESS$.title_province
		save_scope_as = ln_new_colonist
	}
	# Fund the colonization BEFORE the engine call — ai_colonization_effect strips colonize_cost_val from the
	# new holder with no balance check, and a freshly generated lord owns nothing (audited 2026-07-02).
	scope:ln_new_colonist = { add_gold = ln_colonist_grant_value }
	ai_colonization_effect = { WILDERNESS = $WILDERNESS$ ROOT_SCOPE = scope:ln_new_colonist }
	scope:ln_new_colonist = {
		change_government = $GOVERNMENT$
		correct_wilderness_tracker = yes
		# 1.1.3 -- the same permafrost strip AGOT runs on its own feudal settlers (see
		# ln_refugee_settle_here_effect). The charters land in the North mainland, so this is hygiene.
		if = {
			limit = { government_has_flag = government_is_feudal }
			every_held_county = { agot_remove_permafrost_modifiers = yes }
		}
	}
	$WILDERNESS$ = {
		# Wilderness sits at development 0 — give the settlement a start, like the native settle decision does.
		random_list = {
			33 = { change_development_level = 2 }
			33 = { change_development_level = 3 }
			33 = { change_development_level = 4 }
		}
		if = {
			limit = { scope:ln_new_colonist = { has_government = tribal_government } }
			title_province = { set_holding_type = tribal_holding }
		}
		else = {
			title_province = { set_holding_type = castle_holding }
		}
	}
	# Vassalize to the Crown — ai_colonization_effect does NOT set a liege (audited 2026-07-02: change_title_holder
	# type returned, no liege change), the lord would stay independent. CK3's AI then redistributes him to the
	# de jure liege (the North's lord paramount) on its own; a player king does whatever he wants with him.
	if = {
		limit = {
			exists = $LIEGE$
			NOT = { scope:ln_new_colonist = $LIEGE$ }
		}
		create_title_and_vassal_change = { type = swear_fealty save_scope_as = ln_vassal_change add_claim_on_loss = no }
		scope:ln_new_colonist = { change_liege = { liege = $LIEGE$ change = scope:ln_vassal_change } }
		resolve_title_and_vassal_change = scope:ln_vassal_change
	}
}

# ------------------------------------------------------------------------------------------------------------
# Resolve the current Crown (if any) into scope:ln_aftermath_crown. No Crown → scope stays unset.
ln_save_aftermath_crown_effect = {
	if = {
		limit = { exists = title:h_the_iron_throne.holder }
		title:h_the_iron_throne.holder = { save_scope_as = ln_aftermath_crown }
	}
	else_if = {
		limit = {
			exists = title:e_the_north.holder
			title:e_the_north.holder = { is_independent_ruler = yes }
		}
		title:e_the_north.holder = { save_scope_as = ln_aftermath_crown }
	}
}

# ------------------------------------------------------------------------------------------------------------
# THE QUESTION OF THE NORTH — pose the mandatory Watch-or-wildlings dilemma to the Crown (event within 30-180
# days). No Crown at all → nobody holds the authority to keep the Wall: the free folk pour through unbidden.
# ln_question_posed (auto-expiring, 400 days) re-arms the yearly safety net if the chosen crown dies before
# the event is delivered.
ln_pose_question_of_the_north_effect = {
	set_global_variable = { name = ln_question_posed value = yes days = 400 }
	# ── 1.1.3 (Thomas's call D-c): A PLAYER IN THE NORTH FIRST, under the default of the new rule ──────────
	# (common/game_rules/zz_ln_aftermath_rules.txt; Actalo's report is in its header). Four rungs, deepest
	# claim first, and exactly one player is picked: the Iron Throne, then e_the_north vassal or not, then
	# the highest-tier player with land in the North mainland (kingdom, duchy, county -- rungs, so no
	# weights are needed), then the AI ladder below exactly as before. random_player with a limit is
	# vanilla's (15 uses across vanilla and AGOT); there is no ordered_player anywhere, hence the rungs.
	# In multiplayer every other player still gets ln_aftermath.3/.4 as now, and an AFK chooser is
	# auto-picked by ai_chance after MULTIPLAYER_EVENT_TIME_OUT, so the question cannot stall; the yearly
	# re-pose still covers a chooser who dies before the card lands. A vassal player who restores the Watch
	# does so on his liege's behalf -- that is the point of the rule, and the card's text says who is asked.
	if = {
		limit = {
			has_game_rule = ln_question_player_in_the_north_first
			any_player = { ln_question_of_the_north_player_trigger = yes }
		}
		if = {
			limit = { title:h_the_iron_throne.holder ?= { ln_question_of_the_north_player_trigger = yes } }
			debug_log = "LN_TRACE AFTERMATH: the Question of the North goes to the player on the Iron Throne"
			title:h_the_iron_throne.holder = { trigger_event = { id = ln_aftermath.1 days = { 30 180 } } }
		}
		else_if = {
			limit = { title:e_the_north.holder ?= { ln_question_of_the_north_player_trigger = yes } }
			debug_log = "LN_TRACE AFTERMATH: the Question of the North goes to the player holding the North"
			title:e_the_north.holder = { trigger_event = { id = ln_aftermath.1 days = { 30 180 } } }
		}
		else_if = {
			limit = {
				any_player = {
					ln_question_of_the_north_player_trigger = yes
					highest_held_title_tier >= tier_kingdom
				}
			}
			debug_log = "LN_TRACE AFTERMATH: the Question of the North goes to a player king with land in the North"
			random_player = {
				limit = {
					ln_question_of_the_north_player_trigger = yes
					highest_held_title_tier >= tier_kingdom
				}
				trigger_event = { id = ln_aftermath.1 days = { 30 180 } }
			}
		}
		else_if = {
			limit = {
				any_player = {
					ln_question_of_the_north_player_trigger = yes
					highest_held_title_tier >= tier_duchy
				}
			}
			debug_log = "LN_TRACE AFTERMATH: the Question of the North goes to a player duke with land in the North"
			random_player = {
				limit = {
					ln_question_of_the_north_player_trigger = yes
					highest_held_title_tier >= tier_duchy
				}
				trigger_event = { id = ln_aftermath.1 days = { 30 180 } }
			}
		}
		else = {
			debug_log = "LN_TRACE AFTERMATH: the Question of the North goes to a player lord with land in the North"
			random_player = {
				limit = { ln_question_of_the_north_player_trigger = yes }
				trigger_event = { id = ln_aftermath.1 days = { 30 180 } }
			}
		}
	}
	else_if = {
		limit = { exists = title:h_the_iron_throne.holder }
		title:h_the_iron_throne.holder = { trigger_event = { id = ln_aftermath.1 days = { 30 180 } } }
	}
	else_if = {
		limit = {
			exists = title:e_the_north.holder
			title:e_the_north.holder = { is_independent_ruler = yes }
		}
		title:e_the_north.holder = { trigger_event = { id = ln_aftermath.1 days = { 30 180 } } }
	}
	# v0.3.0: before giving up and unleashing the free folk by default, look for whoever actually rules Westeros.
	# Without this, a world whose Iron Throne had fallen and whose e_the_north was unheld decided the fate of the
	# North by falling through an `else` -- no crown asked, no player consulted.
	#
	# WAS ordered_ruler, WHICH DOES NOT EXIST. Caught by /tmp/vocabcheck.py, which checks every identifier this
	# mod uses against the 231,889 that appear anywhere in the base game or AGOT: `ordered_ruler` appears in
	# ZERO of roughly 21,000 files. It is the second keyword I have invented in this project -- the first was
	# is_within_diplo_range_of -- and both would have failed silently, because CK3 does not reliably error on an
	# unknown list builder; it just iterates nothing. This whole branch was dead, and the fate of the North was
	# quietly falling through to the else anyway.
	#
	# random_ruler is verified (it is used across the playset) and the weight does the same job order_by was
	# meant to: an emperor is picked over a king, a king over anyone else, and exactly one is picked either way.
	else_if = {
		limit = { any_ruler = { ln_aftermath_fallback_crown_trigger = yes } }
		random_ruler = {
			limit = { ln_aftermath_fallback_crown_trigger = yes }
			weight = {
				base = 1
				modifier = { add = 20  highest_held_title_tier >= tier_empire }
				modifier = { add = 10  highest_held_title_tier >= tier_kingdom }
				modifier = { add = 4   highest_held_title_tier >= tier_duchy }
			}
			trigger_event = { id = ln_aftermath.1 days = { 30 180 } }
		}
	}
	else = {
		ln_wildlings_unleashed_effect = yes
	}
}

# v0.2.0 -- THE WATCH PATH. Previously the Watch branch of ln_aftermath.1 set its globals inline and told nobody.
# The wildling branch broadcast to every player; this one did not. In multiplayer that meant a crown could refound
# the Night's Watch and every other human player simply never learned the world had changed. Mirrored here so both
# outcomes are announced the same way.
ln_nights_watch_restored_effect = {
	set_global_variable = { name = ln_nw_dilemma_done value = yes }
	set_global_variable = { name = ln_nights_watch_restored value = yes }
	# 1.0.7: the delayed notification card needs the restorer's face, and a saved scope does not
	# survive a queued event -- durable global at the source, rebuilt in the card's immediate.
	scope:ln_watch_chooser ?= { set_global_variable = { name = ln_watch_restorer  value = this } }
	# ══ 1.0.16a -- REFOUND ONLY WHAT IS ACTUALLY GONE. ══════════════════════════════════════════════════════
	# longnight_reform_nightswatch_effect creates a NEW Lord Commander and takes c_castle_black, c_the_nightfort,
	# c_the_shadow_tower, c_eastwatch, d_the_wall and k_the_wall from whoever holds them, with no guard of any
	# kind. Fixing ln_the_watch_survived_trigger (zz_ln_nights_watch_triggers.txt) makes the CARD tell the
	# truth; this makes the BUTTON honest. If the Wall still has a legitimate living holder -- a surviving
	# Watch, or the wildlings who took it and held it -- nothing is taken and no stranger is installed.
	# Thomas's call, 31 Aug, over the alternative of splitting option A into reinforce-vs-refound: that is a
	# design change and this is a hotfix. The two labels the card already carries (ln_aftermath.1.a.held vs
	# ln_aftermath.1.a) are still there if he wants that split later.
	if = {
		limit = { ln_the_watch_survived_trigger = no }
		longnight_reform_nightswatch_effect = yes
	}
	# Notify every player except the crown who just chose (scope:ln_watch_chooser, saved by the event option).
	# ?= keeps the comparison safe when the scope is unset.
	every_player = {
		limit = {
			is_alive = yes
			NOT = { this ?= scope:ln_watch_chooser }
		}
		# 1.0.7: spaced off the moment of choosing (session 18 audit: it landed stacked on the Dawn's
		# own cards). Three weeks for a raven to make the news everyone's news.
		trigger_event = { id = ln_aftermath.4  days = { 20 40 } }
	}
}

# The wildling path is taken (chosen by the Crown, or unleashed by the absence of one). One-shot by design:
# ln_nw_dilemma_done closes the dilemma, ln_wildlings_welcomed drives the yearly migration pulse.
ln_wildlings_unleashed_effect = {
	set_global_variable = { name = ln_nw_dilemma_done value = yes }
	set_global_variable = { name = ln_wildlings_welcomed value = yes }
	# The realm identity if a clan ever unifies the zone: k_the_wall stops carrying the Watch's name.
	# (Its de jure duchies keep their geographic names; d_the_wall stays "the Wall" — the place is still the Wall.)
	title:k_the_wall = { set_title_name = ln_giftlands }
	ln_wildling_wave_effect = yes
	# Notify every player — except the crown who just made the choice (scope:ln_unleash_chooser, saved by the
	# event option; unset on the no-Crown path, where everyone gets the news). ?= keeps the comparison safe
	# when the scope is unset or root is not a character (story on_end).
	every_player = {
		limit = {
			is_alive = yes
			NOT = { this ?= scope:ln_unleash_chooser }
		}
		trigger_event = ln_aftermath.3
	}
}

# ------------------------------------------------------------------------------------------------------------
# One migration wave: up to TWO clans arrive and each settles one duchy of the zone. The first wave fires with
# the choice itself, the yearly pulse keeps them coming while zone land remains — the migration takes a few
# years, leaving the race open to Essosi charters, returning refugees and native lord colonization.
ln_wildling_wave_effect = {
	ln_save_aftermath_crown_effect = yes
	# Candidate counties: the Wall's own de jure (the Gifts are the lore heart of the settlement)...
	title:k_the_wall = {
		every_de_jure_county = {
			limit = { ln_wilderness_county_trigger = yes }
			add_to_list = ln_wave_counties
		}
	}
	# ...plus the adjacent northern duchies (dynamic zone).
	title:e_the_north = {
		every_de_jure_county = {
			limit = {
				ln_wilderness_county_trigger = yes
				ln_wildling_zone_county_trigger = yes
				NOT = { is_in_list = ln_wave_counties }
			}
			add_to_list = ln_wave_counties
		}
	}
	if = {
		limit = { any_in_list = { list = ln_wave_counties always = yes } }
		ln_wildling_settle_one_duchy_effect = yes
		ln_wildling_settle_one_duchy_effect = yes
	}
	else = {
		# Zone fully settled — the migration is over, stop the yearly pulse for good.
		set_global_variable = { name = ln_wildling_migration_done value = yes }
	}
}

# One clan, one duchy. Expects the ln_wave_counties list; silently no-ops when it has run dry.
ln_wildling_settle_one_duchy_effect = {
	if = {
		limit = { any_in_list = { list = ln_wave_counties always = yes } }
		random_in_list = {
			list = ln_wave_counties
			save_scope_as = ln_wave_seed
		}
		scope:ln_wave_seed.duchy = { save_scope_as = ln_wave_duchy }
		# One generated clan chief per duchy — mostly generic free folk, with a sprinkle of the named clans.
		random_list = {
			60 = { ln_spawn_wildling_clan_effect = { COUNTY = scope:ln_wave_seed CULTURE = culture:first_clan } }
			10 = { ln_spawn_wildling_clan_effect = { COUNTY = scope:ln_wave_seed CULTURE = culture:thenn } }
			10 = { ln_spawn_wildling_clan_effect = { COUNTY = scope:ln_wave_seed CULTURE = culture:hornfoot } }
			10 = { ln_spawn_wildling_clan_effect = { COUNTY = scope:ln_wave_seed CULTURE = culture:forestman } }
			10 = { ln_spawn_wildling_clan_effect = { COUNTY = scope:ln_wave_seed CULTURE = culture:nightrunner } }
		}
		# The chief's clan spreads over the duchy's remaining wilderness (mirrors the native settle_duchy loop).
		scope:ln_wave_duchy = {
			every_de_jure_county = {
				limit = {
					ln_wilderness_county_trigger = yes
					NOT = { this = scope:ln_wave_seed }
				}
				save_scope_as = ln_wave_extra
				ai_colonization_effect = { WILDERNESS = scope:ln_wave_extra ROOT_SCOPE = scope:ln_new_colonist }
				scope:ln_wave_extra = {
					random_list = {
						33 = { change_development_level = 2 }
						33 = { change_development_level = 3 }
						33 = { change_development_level = 4 }
					}
					title_province = { set_holding_type = tribal_holding }
				}
			}
		}
		# Hand him the duchy title when he ends up holding half of it — a proper clan High Chieftain.
		if = {
			limit = {
				NOT = { exists = scope:ln_wave_duchy.holder }
				scope:ln_wave_duchy = {
					any_de_jure_county = {
						percent >= 0.5
						holder = scope:ln_new_colonist
					}
				}
			}
			scope:ln_new_colonist = { get_title = scope:ln_wave_duchy }
		}
		# Purge the settled duchy from the wave list so the next call picks a different one.
		scope:ln_wave_duchy = {
			every_de_jure_county = {
				limit = { is_in_list = ln_wave_counties }
				remove_from_list = ln_wave_counties
			}
		}
	}
}

# random_list branches cannot parameterize a macro argument — tiny per-culture forwarder instead.
ln_spawn_wildling_clan_effect = {
	ln_settle_generated_lord_effect = {
		WILDERNESS = $COUNTY$
		CULTURE = $CULTURE$
		FAITH = faith:old_gods_btw_fnf
		GOVERNMENT = tribal_government
		LIEGE = scope:ln_aftermath_crown
	}
}

# ------------------------------------------------------------------------------------------------------------
# REFUGEE PULL — target picking. Sets scope:ln_return_target (county) when a destination exists.
# Priority: his own lost seat (ln_lost_seat, stored at flight) → an empty county near that seat → any
# wilderness left in the North.
ln_refugee_pick_target_effect = {
	if = {
		limit = {
			has_variable = ln_lost_seat
			var:ln_lost_seat ?= { ln_wilderness_county_trigger = yes }
		}
		var:ln_lost_seat = { save_scope_as = ln_return_target }
	}
	else_if = {
		limit = {
			has_variable = ln_lost_seat
			exists = var:ln_lost_seat
		}
		var:ln_lost_seat = { ln_find_wilderness_neighbor_effect = yes }
		if = {
			limit = { NOT = { exists = scope:ln_return_target } }
			random_county_in_region = {
				region = world_westeros_the_north_mainland
				limit = { ln_wilderness_county_trigger = yes }
				save_scope_as = ln_return_target
			}
		}
	}
	else = {
		random_county_in_region = {
			region = world_westeros_the_north_mainland
			limit = { ln_wilderness_county_trigger = yes }
			save_scope_as = ln_return_target
		}
	}
}

# County scope: find an empty county nearby (ring 1, then ring 2). Sets scope:ln_return_target on success.
ln_find_wilderness_neighbor_effect = {
	if = {
		limit = { any_neighboring_county = { ln_wilderness_county_trigger = yes } }
		random_neighboring_county = {
			limit = { ln_wilderness_county_trigger = yes }
			save_scope_as = ln_return_target
		}
	}
	else_if = {
		limit = {
			any_neighboring_county = {
				any_neighboring_county = { ln_wilderness_county_trigger = yes }
			}
		}
		random_neighboring_county = {
			limit = { any_neighboring_county = { ln_wilderness_county_trigger = yes } }
			random_neighboring_county = {
				limit = { ln_wilderness_county_trigger = yes }
				save_scope_as = ln_return_target
			}
		}
	}
}

# REFUGEE PULL — settle where the caravan stands (root = the refugee, standing in a wilderness county).
# Mirrors the native settle decision's core, then clears the refugee state.
ln_refugee_settle_here_effect = {
	location.county = { save_scope_as = ln_settle_county }
	ai_colonization_effect = { WILDERNESS = scope:ln_settle_county ROOT_SCOPE = root }
	# Government mirrors the native settle decision: the culture head's government, feudal fallback.
	#
	# ── 1.1.3: FREE FOLK STAY FREE FOLK, AND BEYOND THE WALL IS TRIBAL LAND (TL / vitor109050, Discord, ────
	# 24 Sep: "the ones that pop up beyond the wall become feudal, which fucks them because of the permafrost
	# modifier"). Not intended. A free folk chief who fled the dead and Returns North to his lost seat copied
	# his culture head's government -- and after a Long Night his culture often has no head at all (wighting
	# set the free folk rulers to frostborn, and the wights die at the Dawn), so he fell to feudal on tribal
	# land. Every free folk culture carries tradition_mpo_northern_tribes, built around tribal holdings
	# (permafrost_unlock_tribal_building, permafrost_cannot_become_feudal), so feudal there is a lord who can
	# build nothing. First branch now: tribal if he is free folk. (1.1.3 review: the work order also made any
	# county beyond the Wall tribal; dropped -- a northern colonist returning to a seat up there is not free
	# folk, his culture has no permafrost_cannot_become_feudal, and AGOT's own settle decision gives him his
	# culture head's government with the permafrost strip below, which is what he now gets.)
	# agot_is_wildling_culture is AGOT's (00_agot_cultural_triggers.txt:62), called through culture = { } as
	# AGOT calls it (00_agot_wildling_effects.txt:472).
	if = {
		limit = {
			culture = { agot_is_wildling_culture = yes }
		}
		change_government = tribal_government
	}
	else_if = {
		limit = {
			exists = culture.culture_head
			culture.culture_head = { has_government = clan_government }
		}
		change_government = clan_government
	}
	else_if = {
		limit = {
			exists = culture.culture_head
			culture.culture_head = { has_government = tribal_government }
		}
		change_government = tribal_government
	}
	else = {
		change_government = feudal_government
	}
	# 1.1.3 -- AND WHAT AGOT DOES FOR A FEUDAL SETTLER, WE DO TOO. "Mirrors the native settle decision" was
	# missing its last step: AGOT strips its permafrost modifier from a feudal settler's counties
	# (00_agot_colonization_decisions.txt:349-356, agot_remove_permafrost_modifiers at
	# 00_agot_wildling_effects.txt:520), because the good variant is a tribal bonus and the bad one is a
	# castle-builder's curse. Same call, same guard.
	if = {
		limit = { government_has_flag = government_is_feudal }
		every_held_county = { agot_remove_permafrost_modifiers = yes }
	}
	scope:ln_settle_county = {
		random_list = {
			33 = { change_development_level = 2 }
			33 = { change_development_level = 3 }
			33 = { change_development_level = 4 }
		}
		if = {
			limit = { root = { has_government = tribal_government } }
			title_province = { set_holding_type = tribal_holding }
		}
		else = {
			title_province = { set_holding_type = castle_holding }
		}
	}
	# The settler folk become the settlement (the native decision destroys them too).
	every_maa_regiment = {
		limit = { is_maa_type = laamp_settler_maa }
		destroy_maa_regiment = yes
	}
	# Reclaiming your own ancestral seat is worth remembering.
	if = {
		limit = {
			has_variable = ln_lost_seat
			var:ln_lost_seat ?= scope:ln_settle_county
		}
		add_prestige = 200
		add_character_modifier = ln_restored_ancestral_seat_modifier
	}
	ln_clear_refugee_state_effect = yes
	correct_wilderness_tracker = yes
}

# Home at last (or the road ends) — the refugee is a refugee no more.
ln_clear_refugee_state_effect = {
	remove_character_flag = dawn_refugee
	if = { limit = { has_character_flag = dawn_refugee_lord } remove_character_flag = dawn_refugee_lord }
	if = { limit = { has_character_flag = dawn_refugee_courtier } remove_character_flag = dawn_refugee_courtier }
	remove_character_modifier = dawn_refugee_modifier
	if = { limit = { has_variable = ln_lost_seat } remove_variable = ln_lost_seat }
	if = { limit = { has_variable = ln_camp_level } remove_variable = ln_camp_level }
	if = { limit = { has_variable = ln_return_destination } remove_variable = ln_return_destination }
}

# ── LIFTED OUT OF THE ON_ACTION ─────────────────────────────────────────────────────────────────────────────
# This logic used to live inline inside ln_aftermath_yearly_pulse's effect block, hanging off vanilla's
# yearly_global_pulse -- which never ran, because vanilla declares that on_action with its own effect block and
# a second declaration of the name competes rather than merges. Now it is a named effect the heartbeat calls.
ln_aftermath_yearly_effect = {
	# Wildling migration waves -- up to two clans a year while zone land remains. The wave effect flips
	# ln_wildling_migration_done when the zone runs dry, stopping this branch for good.
	if = {
		limit = {
			exists = global_var:ln_wildlings_welcomed
			NOT = { exists = global_var:ln_wildling_migration_done }
		}
		ln_wildling_wave_effect = yes
	}
	# Safety net: the Question of the North was scheduled but never delivered (the chosen crown died before the
	# event fired). ln_question_posed auto-expires after 400 days -- repose the question then.
	else_if = {
		limit = {
			exists = global_var:ln_aftermath_started
			NOT = { exists = global_var:ln_nw_dilemma_done }
			NOT = { exists = global_var:ln_question_posed }
		}
		ln_pose_question_of_the_north_effect = yes
	}
	# 1.0.10: the county-memory self-heal -- its own if, NOT part of the else_if chain above, because
	# it must run even in years when a wildling wave or the Question also fires.
	ln_county_memory_sweep_effect = yes
}
