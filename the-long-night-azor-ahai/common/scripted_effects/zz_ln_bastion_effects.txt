# AGOT: The Long Night & Azor Ahai -- raising and lowering the last bastions.
#
# Called once a year from ln_struggle_yearly_work_effect. See zz_ln_bastion_modifiers.txt for why.
#
# WHY A YEARLY LATCH AND NOT A PHASE HOOK. Phase-level on_start exists ONLY on an ending phase -- confirmed
# against vanilla's Iberian and Persian struggles and recorded in ck3-verified-api-notes.md. A hook written on a
# normal phase is silent dead code. So this runs on the yearly work effect and is written to be idempotent:
# it adds only what is missing, every year, which also picks up great houses that rise, fall, or change hands
# mid-war.
#
# COST. every_ruler is a few hundred characters once a year and each one does a single county check. The
# expensive sweep -- every_county, thousands of entries -- runs EXACTLY ONCE, at the Dawn, behind a global
# latch. It is not on the yearly path.
ln_bastion_upkeep_effect = {
	if = {
		limit = {
			exists = struggle:ln_long_night_struggle
			struggle:ln_long_night_struggle = {
				OR = {
					is_struggle_phase = ln_phase_cold_winds
					is_struggle_phase = ln_phase_the_long_night
				}
			}
		}
		# 1.0.15e -- THE WALLS NOW RISE *AND FALL* WITH THE COUNTRY AROUND THEM.
		# This used to add only, never remove, so a great seat stayed fortified all war no matter how much
		# of its region had gone over -- and after 1.0.15d made the capital the war-score objective, that
		# was the difference between "the dead have to work for Winterfell" and "the dead can never take
		# Winterfell". Now the bastion is a live state: raised while under half the seat's own de jure duchy
		# is in dead hands, and dropped the year that stops being true.
		# See ln_seat_region_still_stands_trigger for the measurement and for why this rides the existing
		# bastion system instead of adding a second one.
		# 1.1.3 (W15.6): ONE every_ruler WALK, NOT TWO. The second pass -- the dead do not inherit the
		# walls, Thomas's catch, its note is below -- rode a separate walk of the same list. It is the
		# first branch of the same walk now, asked before the raise/lower question so a converted holder's
		# walls come down before anything could raise them. Same result, one walk.
		every_ruler = {
			limit = {
				OR = {
					ln_holds_a_last_bastion_trigger = yes
					AND = {
						ln_is_one_of_them_trigger = yes
						capital_county ?= { has_county_modifier = ln_last_bastion_modifier }
					}
				}
			}
			# ══ THE DEAD DO NOT INHERIT THE WALLS -- Thomas's catch, and it is a better gate than the one it
			# joins. ln_seat_title_gain (on_title_gain) strips a stranded bastion when a seat CHANGES HANDS.
			# But the nastiest road does not change hands at all: mutate_into_other_effect converts a lord IN
			# PLACE. Same character, same county, same title history -- and he is now one of them, holding a
			# castle at +3 fort level and +50% garrison against everyone trying to take it back. No title-gain
			# hook can ever see that, because no title was gained.
			#
			# So this asks the question by the HOLDER instead of by the event: is the man sitting in this
			# fortified seat one of the dead? It catches conversion in place, and it is also a blanket backstop
			# for every other way a bastion could strand -- title transfers the hook missed, saves made before
			# either guard existed, and anything a future edit forgets.
			if = {
				limit = {
					ln_is_one_of_them_trigger = yes
					capital_county ?= { has_county_modifier = ln_last_bastion_modifier }
				}
				debug_log = "LN_TRACE SEAT: a great seat is held by the dead - the walls do not serve them"
				capital_county = {
					remove_county_modifier = ln_last_bastion_modifier
				}
			}
			else_if = {
				limit = {
					ln_holds_a_last_bastion_trigger = yes
					capital_county = {
						ln_seat_region_still_stands_trigger = yes
						NOT = { has_county_modifier = ln_last_bastion_modifier }
					}
				}
				capital_county = {
					add_county_modifier = ln_last_bastion_modifier
				}
			}
			else_if = {
				limit = {
					ln_holds_a_last_bastion_trigger = yes
					capital_county = {
						ln_seat_region_still_stands_trigger = no
						has_county_modifier = ln_last_bastion_modifier
					}
				}
				debug_log = "LN_TRACE SEAT: the country around a great seat has fallen - its walls stand alone now"
				capital_county = {
					remove_county_modifier = ln_last_bastion_modifier
				}
			}
		}
	}

	# THE DAWN, AND ONLY THE DAWN. Not a bare else: the else branch would also run through every Long Summer
	# year, set the latch decades before the walls were ever raised, and then never clean up after the war.
	else_if = {
		limit = {
			struggle:ln_long_night_struggle ?= { is_struggle_phase = ln_phase_the_dawn }
			NOT = { has_global_variable = ln_bastions_lowered }
		}
		set_global_variable = { name = ln_bastions_lowered  value = yes }
		every_county = {
			limit = { has_county_modifier = ln_last_bastion_modifier }
			remove_county_modifier = ln_last_bastion_modifier
		}
	}
}
