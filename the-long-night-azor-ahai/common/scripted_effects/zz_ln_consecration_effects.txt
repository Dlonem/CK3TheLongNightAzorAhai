# AGOT: The Long Night & Azor Ahai -- CONSECRATION, the second road to Lightbringer.
# Design and the no-rename constraint are written out in common/scripted_triggers/zz_ln_consecration_triggers.txt.

# Root = the champion. scope:ln_the_blade = the artifact being consecrated, saved by the caller.
#
# WHAT THIS DELIBERATELY DOES NOT DO: destroy_artifact + create_artifact. That was the obvious way to get a
# blade called Lightbringer out of a blade called Longclaw, and it is wrong twice over -- the engine cannot
# carry a history entry or a house claim across the swap, so the sword would arrive in the world with no past
# and no owner-house, and AGOT's own Valyrian steel content (the house claim interactions in
# common/character_interactions/00_agot_valyrian_steel_interactions.txt) would stop recognising it.
#
# EVERY EFFECT BELOW IS ONE THE ENGINE ACTUALLY HAS ON AN EXISTING ARTIFACT. The complete list is in
# common/effect_localization/01_ep1_effects.txt: add/remove_artifact_modifier, add/remove_house_artifact_claim,
# add/remove_personal_artifact_claim, copy_artifact_modifiers, create_artifact, destroy_artifact,
# destroy_owned_artifact, equip_artifact_to_owner, equip_artifact_to_owner_replace,
# unequip_artifact_from_owner, set_artifact_rarity.
# (1.0.15e: _replace added to this list -- it is in-tree at zz_ln_household_effects.txt and the_long_night.txt,
#  and this effect now calls it. A future edit reading this list would otherwise call it invalid.)
ln_consecrate_blade_effect = {
	set_global_variable = { name = ln_lightbringer_forged  value = yes }
	set_global_variable = { name = ln_lightbringer_forger  value = root }
	add_character_flag = ln_forged_lightbringer
	# (ln_consecrated_the_blade, a second flag here, is gone -- nothing ever read it, the engine said so in
	# Thomas's log, and the artifact's own ln_was_consecrated variable is the record that matters.)
	create_character_memory = { type = ln_forged_lightbringer_memory }

	scope:ln_the_blade = {
		# The marker the rest of the mod reads. ln_carries_lightbringer_trigger, the reveal chain and the Night
		# King's death all test this variable ON THE ARTIFACT, never the character, so a consecrated blade that
		# is looted, inherited or buried carries its consecration with it exactly as a forged one does.
		set_variable = { name = ln_is_lightbringer  value = yes }
		set_variable = { name = ln_was_consecrated  value = yes }
		add_artifact_modifier = ln_lightbringer_consecrated_modifier
		# Vanilla's own post-creation rarity call, artifact scope -- 05_dlc_bp2_effects.txt:355.
		set_artifact_rarity = illustrious
		# 1.0.15d LIVE-RUN FIX -- ..._replace, not the plain equip. Thomas's 130 AC run:
		#   equip_artifact_to_owner [ No available slot which can hold Cloudbreak! ]
		#
		# CORRECTED 1.0.15e, because the first version of this comment got the mechanism wrong and it is
		# worth recording why. It claimed the blade was left unequipped and therefore invisible to
		# ln_carries_lightbringer_trigger. It cannot have been: ln_pick_the_blade_effect below selects
		# scope:ln_the_blade with random_EQUIPPED_character_artifact in both of its branches, so the blade
		# being consecrated is always the sword the man is already wearing. The slot was occupied by the
		# artifact itself, the call was a no-op, and the error line was the only symptom.
		# _replace re-equips the same artifact into the same slot, which cannot displace anything -- so
		# this is a log-noise fix, not a rescued feature. Kept because a silent error every consecration
		# is exactly how a real one gets missed later.
		equip_artifact_to_owner_replace = yes
	}

	# Same world-news card the forge sends. ln_lightbringer.8 reads global_var:ln_lightbringer_forger and asks
	# whether the hearer believes it, which is the right question either way -- a consecration is a rumour of a
	# red sword exactly as a forging is.
	every_player = {
		limit = { NOT = { this = root } }
		trigger_event = { id = ln_lightbringer.8  days = { 3 30 } }
	}
}

# Root = the champion. Picks the blade the rite will take hold of and saves it as scope:ln_the_blade.
#
# DAWN FIRST, ALWAYS. A Dayne who somehow carries two Valyrian blades should not have the rite land on the
# other one -- Dawn is the case the whole special path exists for.
ln_pick_the_blade_effect = {
	if = {
		limit = {
			ln_carries_dawn_trigger = yes
			has_character_flag = ln_deed_bleeding_star
		}
		random_equipped_character_artifact = {
			limit = {
				ln_blade_is_dawn_trigger = yes
				NOT = { has_variable = ln_is_lightbringer }
			}
			save_scope_as = ln_the_blade
		}
	}
	else = {
		# DAWN IS EXCLUDED FROM THE ORDINARY ROUTE ON PURPOSE. No life will consecrate Dawn: it was quenched in
		# a falling star before House Dayne had a name, and it does not take a lesser payment. A Dayne without
		# the bleeding star who happens to carry a second Valyrian blade consecrates THAT one instead, which is
		# why this is an exclusion here rather than a refusal in the decision.
		random_equipped_character_artifact = {
			limit = {
				ln_blade_can_be_consecrated_trigger = yes
				NOT = { ln_blade_is_dawn_trigger = yes }
			}
			save_scope_as = ln_the_blade
		}
	}
}

# Root = the champion. Picks who the rite would cost him, and saves scope:ln_beloved.
#
# THE ORDER IS THE PRIORITY AND IT IS DELIBERATE, the same way ln_opening.1's station ladder is: spouse first
# because that is the tale, then soulmate, then lover, then best friend, then an adult child last -- a man
# reaches for the person he swore to before he reaches for the child he raised.
#
# NOTHING HERE IS RANDOM IN THE SENSE THAT MATTERS. random_relation with a limit of one is a selector, not a
# dice roll; the branch order decides which category, and only the tie inside a category is arbitrary.
ln_pick_the_beloved_effect = {
	if = {
		limit = { primary_spouse ?= { is_alive = yes  is_adult = yes } }
		primary_spouse = { save_scope_as = ln_beloved }
	}
	else_if = {
		limit = { any_relation = { type = soulmate  is_alive = yes  is_adult = yes } }
		random_relation = {
			type = soulmate
			limit = { is_alive = yes  is_adult = yes }
			save_scope_as = ln_beloved
		}
	}
	else_if = {
		limit = { any_relation = { type = lover  is_alive = yes  is_adult = yes } }
		random_relation = {
			type = lover
			limit = { is_alive = yes  is_adult = yes }
			save_scope_as = ln_beloved
		}
	}
	else_if = {
		limit = { any_relation = { type = best_friend  is_alive = yes  is_adult = yes } }
		random_relation = {
			type = best_friend
			limit = { is_alive = yes  is_adult = yes }
			save_scope_as = ln_beloved
		}
	}
	else_if = {
		limit = { any_child = { is_alive = yes  is_adult = yes } }
		random_child = {
			limit = { is_alive = yes  is_adult = yes }
			save_scope_as = ln_beloved
		}
	}
}

# ── 1.1.3: A GOBLET THAT BECAME LIGHTBRINGER IS UNMADE (Thomas's call D-n) ─────────────────────────────────
# Until 1.1.3 the rite accepted anything carrying AGOT's valyrian_steel variable -- armour, and the Valyrian
# Steel submod's goblet (Xia, Discord, 19 Sep). A save where that happened has a consecrated trinket holding
# ln_is_lightbringer, the world's one-Lightbringer global closed, and no way to consecrate a real blade.
#
# Root = anything; called once a year from ln_struggle_yearly_work_effect. Same shape and the same reasoning
# as ln_lightbringer_restamp_effect (zz_ln_lightbringer_effects.txt): there is no global artifact iterator,
# every_ruler is the population that can reach an artifact cheaply, the whole thing is behind
# ln_lightbringer_forged (a single failed global read in every game where no Lightbringer exists), and the
# latch is set ON THE FIND of any Lightbringer, not on the attempt -- one in a courtier's hands this year
# surfaces on a ruler eventually, and latching early would give up on exactly that save. (1.1.3 review: the
# latch used to wait for a CONSECRATED artifact, so a game where the sword was forged and never consecrated
# walked every ruler's artifacts every year forever; any Lightbringer found closes it now.) The only writers
# of ln_is_lightbringer are the forge (a sword, always a weapon) and the rite, so only a consecrated artifact
# can be wrong, and ln_was_consecrated is the rite's own stamp.
#
# WHAT IS UNMADE: a consecrated artifact in a slot that is plainly not a weapon -- armour, helmet, regalia or
# a miscellaneous slot (the goblet). NOT one standing in a COURT slot: AGOT and the Valyrian Steel submod
# reforge Valyrian blades onto pedestals and back (01_agot_reforge_valyrian_steel.txt), and a real
# Lightbringer on display must not be unmade for where it stands (1.1.3 review). A crossbow consecrated
# before 1.1.3 is kept too: it is a weapon in the hand, even though the rite no longer accepts one.
#
# READ EVERYTHING, THEN REMOVE, THEN ASK THE WORLD. The slot test is read before anything is taken off; the
# globals and the forger's flag go last, and only if no artifact anywhere still carries ln_is_lightbringer,
# so a real blade can be consecrated again. The memory stays: it happened. The rarity stays illustrious: the
# engine cannot tell us what it was before.
ln_unconsecrate_what_is_not_a_blade_effect = {
	if = {
		limit = {
			has_global_variable = ln_lightbringer_forged
			NOT = { has_global_variable = ln_consecration_slot_checked }
		}
		every_ruler = {
			every_character_artifact = {
				limit = { has_variable = ln_is_lightbringer }
				set_global_variable = { name = ln_consecration_slot_checked  value = yes }
				if = {
					limit = {
						has_variable = ln_was_consecrated
						OR = {
							artifact_slot_type = armor
							artifact_slot_type = helmet
							artifact_slot_type = regalia
							artifact_slot_type = miscellaneous
						}
					}
					debug_log = "LN_TRACE CONSECRATION: the consecrated artifact is not a weapon - unmade"
					remove_variable = ln_is_lightbringer
					remove_variable = ln_was_consecrated
					remove_artifact_modifier = ln_lightbringer_consecrated_modifier
					set_global_variable = { name = ln_consecration_was_unmade  value = yes }
				}
				else = {
					debug_log = "LN_TRACE CONSECRATION: Lightbringer is a weapon (or on display) - nothing to repair"
				}
			}
		}
		if = {
			limit = {
				has_global_variable = ln_consecration_was_unmade
				NOT = {
					any_living_character = {
						any_character_artifact = { has_variable = ln_is_lightbringer }
					}
				}
			}
			debug_log = "LN_TRACE CONSECRATION: no Lightbringer stands anywhere - the world may make one again"
			remove_global_variable = ln_consecration_was_unmade
			global_var:ln_lightbringer_forger ?= { remove_character_flag = ln_forged_lightbringer }
			remove_global_variable = ln_lightbringer_forger
			remove_global_variable = ln_lightbringer_forged
		}
	}
}
