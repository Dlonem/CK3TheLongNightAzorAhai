# AGOT: The Long Night & Azor Ahai -- the Others do not conjure a dragon. They take one.
#
# ══════════════════════════════════════════════════════════════════════════════════════════════════════════════
# WHAT WAS WRONG WITH THE INHERITED VERSION. agot_spawn_adult_ice_dragon_effect creates a character called
# `dragon_skroth` out of nothing, borrows the age of whatever the oldest living dragon happens to be, and gives
# it a fixed pale colour scheme. No dragon has to have died. The Others do not have to have been anywhere near
# one. The name is Long Night+'s invention and appears in neither the books nor the show.
#
# Thomas: "the reason the Night King had a dragon was because the living lost a dragon to his army. He was
# present. He resurrected it... when the Greyjoys killed that second dragon outside King's Landing, the Night
# King wouldn't get access to that dragon because he was nowhere near it."
#
# That is the rule, and it is a better rule. So:
#
#   1. When a dragon dies, ln_record_fallen_dragon_effect asks whether the Others could plausibly have been
#      standing over the body. If so the dragon goes into the ln_fallen_dragons global list, and its age at
#      death is stamped onto its own corpse (AGOT keeps dead dragons unpruneable, so the record survives).
#   2. The Others' story cycle then raises ONE OF THOSE. Not a clone -- that dragon. It keeps its name (via
#      change_first_name = { template_character = ... }, vanilla's own namesake mechanism) and all twenty-six
#      of its shape genes, so Vhagar comes back looking like Vhagar. Only the eight COLOUR genes are overwritten,
#      which is what turns her the colour of the thing that took her.
#   3. If no dragon has fallen to them, they do not get one. They wait. That is the whole point.
#
# The legacy behaviour is still available on the rule as `agot_long_night_dragon_on` for anyone who wants the
# old guarantee, and it is now explicitly labelled as the arcade option that it is.
# ══════════════════════════════════════════════════════════════════════════════════════════════════════════════

# Root = a dragon that has just died. Called from ln_death_catalyst_effect.
ln_record_fallen_dragon_effect = {
	if = {
		limit = {
			has_trait = dragon

			# ── IT MUST NOT ALREADY BE ONE OF THEIRS ────────────────────────────────────────────────────────
			# Thomas: "what happens when you kill that dragon that's owned by the Night King? Logically it
			# should stay dead. There shouldn't be an infinite revive loop." Correct, and these three checks are
			# what guarantee it. Living -> undead -> dead, and dead is the end of it.
			#   * house_theothers   -- everything they raise is born into that house
			#   * ln_undead_dragon  -- the trait, in case a raised dragon ever changes hands
			#   * ln_raised_from    -- the variable pointing back at the body it came out of
			# On top of which, a dragon is removed from ln_fallen_dragons the moment it is raised, so even the
			# original corpse cannot be drawn on twice.
			NOT = { house ?= house:house_theothers }
			NOT = { has_trait = ln_undead_dragon }
			NOT = { has_variable = ln_raised_from }

			# THE DARK HAS TO BE HERE. A dragon that dies in the Long Summer is simply a dead dragon.
			exists = struggle:ln_long_night_struggle
			struggle:ln_long_night_struggle = { is_struggle_phase = ln_phase_the_long_night }

			# AND THEY HAVE TO HAVE BEEN THERE -- see zz_ln_ice_dragon_triggers.txt, which asks where the war
			# has reached rather than where it began.
			ln_others_were_here_trigger = yes
		}
		# Stamped on the corpse rather than held in a global, because there may be several and each needs its
		# own number. Dead dragons are unpruneable in AGOT, so this survives as long as the save does.
		set_variable = { name = ln_died_at_age  value = age }
		add_to_global_variable_list = { name = ln_fallen_dragons  target = this }
	}
}

# Root = the Night King. Called from the Long Night story cycle in place of the old spawn.
#
# HE CANNOT RIDE TWO. Thomas: "say he gets two dragons, he can't ride them like a pair of jet skis. Maybe the
# more dragons die, the more get sorted to the Others, so multiple of them can fly."
#
# So the Night King is served first, and every dragon after that goes to a lieutenant -- any Other in his court
# or realm who is not already riding something. That is also why the story cycle keeps calling this every three
# years instead of stopping once he has one: the supply is the number of dragons the living have lost to him,
# and the demand is the number of Others without a mount. Both grow.
#
# There is no cap and there does not need to be one. The living cannot lose more dragons than they have.
ln_others_take_a_dragon_effect = {
	if = {
		limit = {
			has_game_rule = agot_long_night_dragon_raise
			OR = {
				any_in_global_list = { variable = ln_fallen_dragons  count >= 1 }
				# 1.0.7, DOOR TWO: a converted dragon nobody is riding is supply too -- claimed while every
				# captain already had a mount, waiting owned in the Night King's court.
				any_in_global_list = { variable = living_dragons  ln_unridden_undead_trigger = yes }
			}
		}
		# THE NIGHT KING FIRST. He is the story owner and the trigger upstream already checked he has no dragon.
		if = {
			limit = { has_trait = dragonless_dragonrider }
			remove_trait = dragonless_dragonrider
		}
		ln_mount_or_raise_effect = yes

		# AND THEN HIS CAPTAINS, one apiece, for as long as there are bodies to raise
		# -- or, since Door Two, converted dragons waiting unridden.
		ln_arm_the_lieutenants_effect = yes
	}
	else_if = {
		limit = {
			has_game_rule = agot_long_night_dragon_on
			agot_dragon_population_extinct = no
		}
		if = {
			limit = { has_trait = dragonless_dragonrider }
			remove_trait = dragonless_dragonrider
		}
		agot_spawn_adult_ice_dragon_effect = yes
	}
	# agot_long_night_dragon_off, or nothing has fallen to them yet: they wait. Checked again in three years.
}

# Root = anything. Called once a year from ln_struggle_yearly_work_effect, inside the invasion window.
#
# 1.0.12b -- THE UNDO for the bug the guard on mutate_into_other_effect now prevents: Thomas's save holds at
# least one living dragon (Rhaetherion, wild, four years old) wearing wight_trait, frostborn culture and the
# cold gods, because the county sweeps wighted it like any other resident. Left alone it is wrong twice --
# wrong now, and wrong at the dawn, when the wight-collapse sweep (the_long_night.txt:206) would kill a
# living dragon for a trait it should never have carried.
#
# The heal restores what mutate actually changed on a dragon and nothing else: the trait, the identity
# stamp, the gore-variant flags, the culture and the faith (set back to the clone template's own pair,
# zz_ln_ice_dragon_effects.txt create_character above). The purges mutate ran (disease, wounds) cannot be
# un-run and do not need to be. Runs only while the invasion window is open because the bug can only be
# created inside it; once every save has passed through one yearly tick with this live, it finds nothing.
ln_dewight_dragons_effect = {
	if = {
		limit = {
			has_global_variable = ln_invasion_opening
			NOT = { has_global_variable = ln_aftermath_started }
		}
		every_living_character = {
			limit = {
				has_trait = dragon
				has_trait = wight_trait
			}
			remove_trait = wight_trait
			remove_character_flag = ln_one_of_them
			remove_character_flag = wight_v1
			remove_character_flag = wight_v2
			remove_character_flag = wight_v3
			remove_character_flag = wight_pose_loss
			set_culture = culture:dragon_culture
			set_character_faith = faith:valyrian_pan_freehold
		}
	}
}

# Root = the Other. Raises a specific dragon that the Others put in the ground.
ln_raise_fallen_dragon_effect = {
	random_in_global_list = {
		variable = ln_fallen_dragons
		save_scope_as = ln_fallen_dragon
	}

	if = {
		limit = { exists = scope:ln_fallen_dragon }
		remove_list_global_variable = { name = ln_fallen_dragons  target = scope:ln_fallen_dragon }

		# 1.0.12b -- THE AGE IS READ INTO A SCOPE FIRST. Thomas's run produced a raised clone aged ZERO from a
		# corpse that died at ninety-nine: `age = scope:X.var:Y` inside create_character is exactly the kind of
		# chained read that fails quietly at creation time, and a failed read is age 0. A scope value read is
		# not -- so bank it, then hand create_character the banked number.
		scope:ln_fallen_dragon = {
			save_scope_value_as = { name = ln_raise_age  value = var:ln_died_at_age }
		}
		create_character = {
			# Placeholder only. change_first_name in after_creation replaces it with the fallen dragon's own
			# name -- vanilla's namesake mechanism (birth_events.txt:3427), used here for the same reason.
			name = dragon_skroth
			gender = male
			culture = culture:dragon_culture
			faith = faith:valyrian_pan_freehold
			location = location
			age = scope:ln_raise_age
			dynasty_house = house:house_theothers
			trait = immortal
			trait = dragon
			trait = education_dragon_1
			trait = dragon_physique_good_3
			trait = dragon_solitary
			trait = dragon_bloodthirsty
			trait = dragon_voracious
			trait = dragon_defiant
			random_traits = no
			save_scope_as = dragon

			after_creation = {
				# ── IT IS THE SAME DRAGON ────────────────────────────────────────────────────────────────────
				# IT KEEPS ITS NAME. Thomas: "if Vhagar were to get killed by the Night King, I'd like to think
				# the dragon would still be named Vhagar -- but perhaps it should get a nickname called the
				# undead, and a dragon trait to represent that they belong to the Night King."
				# So: same name, new nickname, new trait. Vhagar the Undead.
				change_first_name = { template_character = scope:ln_fallen_dragon }
				give_nickname = nick_the_undead
				add_trait = ln_undead_dragon
				add_character_flag = ln_one_of_them   # 1.0.7: stamped at creation -- see ln_is_one_of_them_trigger
				set_variable = { name = ln_raised_from  value = scope:ln_fallen_dragon }

				set_variable = { name = dragon_gender_hidden  value = 1 }

				# 11 Aug -- A FLOOR FOR THESE TWO, BEFORE THE STORY EXISTS. AGOT's own hatchling template sets
				# both to 0 with the comment "Set to 0 here since they're a baby-the backage effect doesn't
				# update their age" (00_agot_dragon_templates.txt), and it is right: their
				# agot_back_apply_dragon_aging_effect is a `while = { limit = { age > var:dragon_age_counter } }`
				# loop, so at age 0 the body never executes and dragon_size_base is never written.
				#
				# story_dragon_variable_storage.on_setup then reads scope:dragon.var:dragon_size_base UNGUARDED
				# (agot_story_cycle_dragon_variables.txt:22), and gene_dragon_size is applied mode = replace off
				# the result. We can reach age 0 here: the age comes from the corpse's ln_died_at_age, and
				# ln_dragonpit_sweep_effect executes EVERY dragon in a captured pit, hatchlings included.
				# change_dragon_size below re-bases off the corpse, so this is a floor, not the value.
				set_variable = { name = dragon_size       value = 0 }
				set_variable = { name = dragon_size_base  value = 0 }

				create_story = story_dragon_alive
				set_variable = { name = dragon_age  value = age }
				agot_back_apply_dragon_aging_effect = yes

				ln_copy_dragon_shape_effect = yes
				ln_apply_the_cold_effect = yes

				# ── AGOT'S OWN CLOSING STEPS, and leaving them out was a real bug ───────────────────────────
				# The inherited spawn effect ends with four things this one first shipped without. Two of them
				# are load-bearing and the omission would have been invisible until somebody actually got a
				# raised dragon in-game:
				#   * color_grading / color -- the portrait tint flags the ice dragon look reads
				#   * agot_dragon_transfer_vars_to_story_cycle_effect -- pushes the gene variables INTO the
				#     dragon's story cycle, which is where AGOT's portrait system actually reads them from.
				#     Without it every one of the twenty-six copied shape genes would sit on the character
				#     doing nothing and the dragon would render as a default.
				set_variable = { name = color_grading  value = flag:darkest }
				set_variable = { name = color          value = flag:white }
				agot_dragon_transfer_vars_to_story_cycle_effect = { DRAGON = scope:dragon }
			}
		}

		# ── AND IT HAS TO ACTUALLY BE THEIRS TO RIDE ────────────────────────────────────────────────────────
		# The other omission, and the worse one: without agot_tame_dragon the Night King would have been handed
		# a dragon he was not bonded to. It sits in the world, it belongs to his house, and he cannot fly it.
		# Root here is the Other, so TAMER = this rather than root.story_owner -- this effect is called from
		# inside story_owner already.
		hidden_effect = {
			agot_tame_dragon = {
				TAMER = this
				DRAGON = scope:dragon
			}
			# Size follows the body it came out of, so a raised Vhagar is Vhagar-sized. Thomas: "you'd have to
			# make sure the age matches, so it'd be zero" -- age is taken from ln_died_at_age above, and size
			# is the other half of that same point.
			scope:dragon = {
				change_dragon_size = { VALUE = scope:ln_fallen_dragon.dragon_size }
			}
		}
	}
}

# Root = the new dragon. Twenty-six SHAPE genes, copied straight off the body. This is what makes it recognisably
# the same animal -- the skull, the horns, the neck, the tail. AGOT copies dragon genes this way itself
# (`value = scope:story.var:gene_dragon_primary_color_hue` and friends), so the pattern is theirs, not invented.
ln_copy_dragon_shape_effect = {
	# ══ THE SEE-THROUGH DRAGONS ═══════════════════════════════════════════════════════════════════════════
	# Workshop report from MPREG on 1.0.3: "this mod seems to cause issues with dragons, making them see
	# through or causing parts of them to become see through." It is ours, and this is the line that fixes it.
	#
	# WHY IT HAPPENED. AGOT builds a dragon through their scripted_character_templates, every one of which
	# calls agot_gen_appearance_variables -- 1,115 lines that guarantee EVERY dragon gene has a value, taking
	# an inherited one where it exists and rolling a random one where it does not. We do not use their
	# templates: the raised dragon is a create_character, so that generator never ran on it, and the only
	# genes it ever had were the ones copied below.
	#
	# The copies further down are guarded on `has_variable` on the corpse, which was the right instinct and
	# the wrong fix. A history-defined dragon whose look came from DNA rather than variables has no
	# gene_dragon_wings_shading_template to copy -- so the guard skipped it, the new dragon got NOTHING, and
	# AGOT's portrait script values read those templates UNGUARDED:
	#
	#     Error: Failed to fetch variable for 'gene_dragon_wings_shading_template' due to not being set
	#     Error: Failed to fetch variable for 'gene_dragon_body_shading_template' due to not being set
	#     (9,258 lines between them in one session)
	#
	# A shading gene with no value is a shading layer that does not render. That is the transparency.
	#
	# THE FIX IS THEIR OWN GENERATOR, NOT A HAND-WRITTEN DEFAULT LIST. Calling it first fills every gene --
	# including ones we do not copy and ones AGOT may add later -- and the copies below then overwrite the
	# ones we actually have. The ice-white recolour still runs after this effect, so it wins as it always did.
	# A hand-maintained table of fallback values would have gone stale the first time they touched a gene.
	agot_gen_appearance_variables = yes

	# ══ 1.0.9 -- THE OTHER TWENTY-SIX ═════════════════════════════════════════════════════════════════════
	# The eight guarded copies below this list were added in 1.0.7 after the live run. These twenty-six were
	# left UNGUARDED, and that is the see-through dragon bug -- still, after every fix that claimed it.
	#
	# A history-defined dragon gets its look from DNA, not variables. Balerion and Vhagar have no
	# gene_dragon_* variables at all. `value = scope:ln_fallen_dragon.var:gene_dragon_X` on such a corpse
	# does not merely fail to copy -- it leaves the target UNSET, destroying the value
	# agot_gen_appearance_variables wrote one line earlier. The generator ran; the copy undid it.
	#
	# THE PROOF IS THIS EFFECT'S OWN SPLIT. One run of error.log names 26 genes and only 26:
	#
	#     1320 x gene_dragon_main_horn_shape_template   (132 renders x 10 template tests in the modifier)
	#      132 x each of the other 24 shape genes
	#        0 x gene_dragon_body_shading_template / gene_dragon_wings_shading_template / the other 6
	#
	# The zero set is exactly the guarded set. Same effect, same scope, same execution -- the guard is the
	# only difference between a gene that survives and a gene that does not. That is the whole bug.
	#
	# So: inherit where the corpse has it, KEEP THE GENERATOR'S ROLL where it does not. A raised Vhagar
	# whose corpse carried no variables now comes back as a solid dragon of the generator's making rather
	# than a translucent one -- there is nothing to inherit, and a rolled face beats no face.
	# ── 1.0.13: READ THE ENTRY, NOT THE CORPSE ──────────────────────────────────────────────────────────
	# AGOT's own idiom, lifted from change_dragon_size (00_agot_dragon_effects.txt:700): scope the corpse,
	# walk the storage list for its entry, hold it in a temporary scope. Entry first because 0.5.1 strips
	# the character's gene variables at on_setup; character second because during creation, before that
	# runs, the character is the only source there is; and the generator's roll above is the floor when
	# neither has anything -- a rolled face beats no face, which is the see-through-dragon lesson.
	if = {
		limit = { agot_has_dragon_storage_system_global_list = yes }
		every_in_global_list = {
			variable = gl_dragon_variable_storage
			limit = { var:dragon_id ?= scope:ln_fallen_dragon }
			save_temporary_scope_as = ln_fallen_entry
		}
	}
	if = {
		limit = { exists = scope:ln_fallen_entry  scope:ln_fallen_entry = { has_variable = gene_dragon_back_spike_size } }
		set_variable = { name = gene_dragon_back_spike_size                 value = scope:ln_fallen_entry.var:gene_dragon_back_spike_size }
	}
	else_if = {
		limit = { scope:ln_fallen_dragon = { has_variable = gene_dragon_back_spike_size } }
		set_variable = { name = gene_dragon_back_spike_size                 value = scope:ln_fallen_dragon.var:gene_dragon_back_spike_size }
	}
	if = {
		limit = { exists = scope:ln_fallen_entry  scope:ln_fallen_entry = { has_variable = gene_dragon_body_pattern_template } }
		set_variable = { name = gene_dragon_body_pattern_template           value = scope:ln_fallen_entry.var:gene_dragon_body_pattern_template }
	}
	else_if = {
		limit = { scope:ln_fallen_dragon = { has_variable = gene_dragon_body_pattern_template } }
		set_variable = { name = gene_dragon_body_pattern_template           value = scope:ln_fallen_dragon.var:gene_dragon_body_pattern_template }
	}
	if = {
		limit = { exists = scope:ln_fallen_entry  scope:ln_fallen_entry = { has_variable = gene_dragon_body_pattern_value } }
		set_variable = { name = gene_dragon_body_pattern_value              value = scope:ln_fallen_entry.var:gene_dragon_body_pattern_value }
	}
	else_if = {
		limit = { scope:ln_fallen_dragon = { has_variable = gene_dragon_body_pattern_value } }
		set_variable = { name = gene_dragon_body_pattern_value              value = scope:ln_fallen_dragon.var:gene_dragon_body_pattern_value }
	}
	if = {
		limit = { exists = scope:ln_fallen_entry  scope:ln_fallen_entry = { has_variable = gene_dragon_body_shading_template } }
		set_variable = { name = gene_dragon_body_shading_template           value = scope:ln_fallen_entry.var:gene_dragon_body_shading_template }
	}
	else_if = {
		limit = { scope:ln_fallen_dragon = { has_variable = gene_dragon_body_shading_template } }
		set_variable = { name = gene_dragon_body_shading_template           value = scope:ln_fallen_dragon.var:gene_dragon_body_shading_template }
	}
	if = {
		limit = { exists = scope:ln_fallen_entry  scope:ln_fallen_entry = { has_variable = gene_dragon_body_shading_value } }
		set_variable = { name = gene_dragon_body_shading_value              value = scope:ln_fallen_entry.var:gene_dragon_body_shading_value }
	}
	else_if = {
		limit = { scope:ln_fallen_dragon = { has_variable = gene_dragon_body_shading_value } }
		set_variable = { name = gene_dragon_body_shading_value              value = scope:ln_fallen_dragon.var:gene_dragon_body_shading_value }
	}
	if = {
		limit = { exists = scope:ln_fallen_entry  scope:ln_fallen_entry = { has_variable = gene_dragon_brow_width } }
		set_variable = { name = gene_dragon_brow_width                      value = scope:ln_fallen_entry.var:gene_dragon_brow_width }
	}
	else_if = {
		limit = { scope:ln_fallen_dragon = { has_variable = gene_dragon_brow_width } }
		set_variable = { name = gene_dragon_brow_width                      value = scope:ln_fallen_dragon.var:gene_dragon_brow_width }
	}
	if = {
		limit = { exists = scope:ln_fallen_entry  scope:ln_fallen_entry = { has_variable = gene_dragon_center_fin_size } }
		set_variable = { name = gene_dragon_center_fin_size                 value = scope:ln_fallen_entry.var:gene_dragon_center_fin_size }
	}
	else_if = {

		set_variable = { name = gene_dragon_center_fin_size                 value = scope:ln_fallen_dragon.var:gene_dragon_center_fin_size }
	}
	if = {
		limit = { exists = scope:ln_fallen_entry  scope:ln_fallen_entry = { has_variable = gene_dragon_cheek_width } }
		set_variable = { name = gene_dragon_cheek_width                     value = scope:ln_fallen_entry.var:gene_dragon_cheek_width }
	}
	else_if = {
		limit = { scope:ln_fallen_dragon = { has_variable = gene_dragon_cheek_width } }
		set_variable = { name = gene_dragon_cheek_width                     value = scope:ln_fallen_dragon.var:gene_dragon_cheek_width }
	}
	if = {
		limit = { exists = scope:ln_fallen_entry  scope:ln_fallen_entry = { has_variable = gene_dragon_chin_profile } }
		set_variable = { name = gene_dragon_chin_profile                    value = scope:ln_fallen_entry.var:gene_dragon_chin_profile }
	}
	else_if = {
		limit = { scope:ln_fallen_dragon = { has_variable = gene_dragon_chin_profile } }
		set_variable = { name = gene_dragon_chin_profile                    value = scope:ln_fallen_dragon.var:gene_dragon_chin_profile }
	}
	if = {
		limit = { exists = scope:ln_fallen_entry  scope:ln_fallen_entry = { has_variable = gene_dragon_crest_depth } }
		set_variable = { name = gene_dragon_crest_depth                     value = scope:ln_fallen_entry.var:gene_dragon_crest_depth }
	}
	else_if = {
		limit = { scope:ln_fallen_dragon = { has_variable = gene_dragon_crest_depth } }
		set_variable = { name = gene_dragon_crest_depth                     value = scope:ln_fallen_dragon.var:gene_dragon_crest_depth }
	}
	if = {
		limit = { exists = scope:ln_fallen_entry  scope:ln_fallen_entry = { has_variable = gene_dragon_eye_color_hue } }

	if = {

	else_if = {

# THE DRAGONPIT. Thomas's scenario, and it is the best one in the feature:
#
#   "Any dragon trapped in a dragonpit -- not just the one in King's Landing, because you can build your own.
#    Personally I would release all my dragons from the pit before the Night King gets there. Because if he gets
#    there, or I give up my title and they stay locked in the pit, there's a good chance he's going to come by
#    and execute all the dragons in the pit. They're going to get reborn, and he's going to give them to his
#    lieutenants."
#
# That is exactly what this does, and the reason it is worth building is the sentence he started with: it makes
# the player DO something. A dragonpit is a safe place to keep dragons for the entire game right up until the
# moment it becomes the worst place in the world, and nothing warns you. Letting them out costs you the pit's
# bonuses and the convenience; leaving them in costs you the dragons and then hands them to the enemy.
#
# HOW THE PIT IS READ. AGOT stamps var:pitted_dragon_location on a dragon when it is put in one -- the county
# the pit is in (00_agot_dragon_effects.txt, alongside the agot_dragon_in_dragonpit modifier). 63 read sites in
# AGOT, so it is load-bearing there and safe to read here. A dragon with that variable is in a cage.
#
# WHY IT RUNS YEARLY RATHER THAN ON CAPTURE. There is no on_action for "the Others took this county" that covers
# occupation, title transfer and scouring alike, and a hook that only caught one of the three would be the
# silent kind of wrong. Sweeping once a year catches every route, and a year's delay reads as them getting
# round to it.
# ══════════════════════════════════════════════════════════════════════════════════════════════════════════════

# Root = anything. Called once a year from ln_struggle_yearly_work_effect.
ln_dragonpit_sweep_effect = {
	if = {
		limit = {
			exists = struggle:ln_long_night_struggle
			struggle:ln_long_night_struggle = { is_struggle_phase = ln_phase_the_long_night }
		}
		# every_living_dragon rather than every_in_global_list over living_dragons: it is an ENGINE list builder
		# (AGOT uses it to populate their global list in the first place), so iterating it while killing things
		# does not depend on how a script-maintained list behaves mid-loop. Cheap either way -- there are tens of
		# dragons in the world, not thousands.
		every_living_dragon = {
			limit = {
				is_alive = yes
				has_trait = dragon
				NOT = { house ?= house:house_theothers }
				NOT = { has_trait = ln_undead_dragon }

				# It is in a pit, and the pit is theirs now.
				exists = var:pitted_dragon_location
				var:pitted_dragon_location = { ln_county_is_theirs_trigger = yes }
			}
			# Killed in the cage. death_reason is AGOT's own execution reason so the death feed reads correctly,
			# and no killer is named: the recorder does not need one, because the county being theirs already
			# satisfies ln_others_were_here_trigger on its own.
			death = { death_reason = death_execution }
		}
	}
}

# ── THE SEE-THROUGH DRAGONS, FOR SAVES THAT ALREADY HAVE ONE ───────────────────────────────────────────────
# Root = anything. One-shot, self-gating, run at game start.
#
# taylorbb12345 on the Workshop, on 1.0.4a: "great mod but dragons are see through". The 1.0.4 fix for that
# (agot_gen_appearance_variables at the top of ln_copy_dragon_shape_effect) is present, correctly scoped and
# complete -- but it lives inside create_character's after_creation, so it only ever touches dragons raised
# from that point on. A dragon raised on 1.0.3 or earlier has no shading templates and never will, and no
# amount of updating the mod changes that. Reporting a fixed bug from an old save is not the reporter being
# wrong; it is us shipping a fix with no migration.
#
# WHY IT WRITES THE STORY DIRECTLY instead of calling agot_dragon_transfer_vars_to_story_cycle_effect again:
# that effect CREATES A NEW story_dragon_variable_storage and pushes it onto gl_dragon_variable_storage
# (00_agot_dragon_effects.txt:2249-2279). Calling it twice for one dragon leaves two entries fighting over one
# dragon_id. Matching on var:dragon_id is AGOT's own idiom for finding the right one (same file, :2040).
#
# SAFE WHEN IT DOES NOT MATCH. If the storage entry cannot be found the limit simply selects nothing -- the
# character-side variables are still repaired, and nothing is corrupted.
# Root = ONE undead dragon that has not been gene-synced yet. The body of the repair, factored out in
# 1.0.9 so the two sweeps below can share it without sharing an iterator.
#
# ── WHAT WAS ACTUALLY BROKEN, AND WHY THIS EFFECT NEVER FIXED IT ──────────────────────────────────────
# The old gate was `NOT = { has_variable = gene_dragon_wings_shading_template }`. That gene is one of the
# EIGHT GUARDED copies in ln_copy_dragon_shape_effect, so on any dragon raised by 1.0.7 or later it is
# always set: the generator writes it and the guard never destroys it. The gate could therefore never
# open for the dragons this effect exists to repair. A full run bears it out exactly -- ZERO errors
# against wings_shading, 1,320 against gene_dragon_main_horn_shape_template.
#
# The stamp replaces it, because there are two broken populations and only one shows on the character:
#   * raised before 1.0.9 -- the 26 unguarded copies wiped the character's genes AND the storage entry
#     that on_setup built from them.
#   * raised before 1.0.7 and then "repaired" by the old version of this effect -- the generator refilled
#     the CHARACTER, but only eight genes were ever pushed to storage, so the entry the portrait actually
#     reads still holds nothing for the other 26. Nothing on the character reveals this. A character-side
#     gene test cannot find these dragons at all.
# The stamp is absent on every save that predates 1.0.9, so this FAILS OPEN once per dragon and latches.
ln_repair_one_dragon_effect = {
	save_scope_as = ln_repair_dragon
	# 1.0.9b: stamp versioned _1 -> _2. The _1 stamp landed before the branch below and regardless of
	# whether it fired, so every dragon got exactly one repair attempt EVER -- a dragon the narrow gate
	# skipped was written off from all future sweeps. _2 re-admits every dragon 1.0.9 already stamped.
	set_variable = ln_gene_storage_sync_2
	if = {
		limit = { has_variable = ln_gene_storage_sync_1 }
		remove_variable = ln_gene_storage_sync_1
	}

	# REGENERATE ONLY IF ACTUALLY BROKEN, AND ONLY A CLONE. The old effect ran the generator on every
	# dragon that passed the gate; under a one-pass stamp that would re-roll every undead dragon's face
	# on first load, which is a bug of our own making. Two conditions, and both halves matter:
	#
	# ONLY IF: it is a DOOR ONE clone (exists = var:ln_raised_from -- set in ln_raise_fallen_dragon_effect's
	# after_creation, and the same discriminator ln_undead_dragon_upkeep_effect already uses) AND a marker
	# gene is missing. main_horn_shape_template is the loudest of the 26; wings_shading_template covers
	# clones raised before 1.0.7, when the shading copies were unguarded too.
	#
	# NEVER IF: it is a DOOR TWO conversion. That dragon is the same animal it always was -- its own genes,
	# its own storage entry -- and if it is historical its look comes from DNA and it has NO gene variables
	# at all, which would satisfy the missing-gene test and hand it a randomly generated face. Door Two's
	# entire guarantee is that no appearance is ever rebuilt. Do not rebuild it.
	if = {
		limit = {
			# has_variable, NOT `exists = var:ln_raised_from`. The two differ on exactly the dragons that
			# matter here: ln_raised_from points at a CORPSE, and a corpse can be pruned out of the save.
			# After pruning the variable is still there but its target no longer resolves, so `exists` goes
			# false and the oldest clones -- the ones whose corpses have had the longest to be pruned, and
			# the ones most likely to be broken -- would silently fail the provenance test and never be
			# repaired. This is a question about PROVENANCE, not about whether the corpse is still reachable.
			# (ln_undead_dragon_upkeep_effect below is right to use `exists`: it dereferences the variable.)
			OR = {
				# The original clause, unchanged: a Door One clone whose copied genes went missing.
				AND = {
					has_variable = ln_raised_from
					# 1.0.13: these two asked the CHARACTER, and 0.5.1 removes every gene variable from the
					# character once its storage entry exists -- so both were true for every clone alive and
					# this branch would have re-rolled the face of every undead dragon on the next sweep.
					# The entry is the thing that can actually be missing a gene.
					OR = {

					# AND the character cannot supply the genes itself. Caught on the adversarial pass: without

					# face re-rolled by the generator and then synced -- visible, but no longer the dragon the

			limit = { has_trait = ln_undead_dragon }
			ln_apply_the_cold_effect = yes
		}

# Root = anything. THE BLANK-ENTRY SWEEP (1.0.11) -- the first repair in this file allowed to touch a
# dragon that is not ours.
#
# WHAT IT IS FOR. Both sweeps above gate on `has_trait = ln_undead_dragon`, so a living AGOT dragon with a
# blank storage entry -- Dreamfyre in the 16 Aug screenshots -- was unreachable by any repair we have ever
# shipped. This one selects on the DEFECT rather than on provenance, which is why it finds them.
#
# IT WALKS THE LIST, NOT THE WORLD. `living_dragons` is tens of entries; every_living_dragon is declared
# `base = living_character` + `has_trait = dragon` and is the ~191,000-character walk flagged elsewhere in
# this project. Thomas, 14 Aug: "make sure that it's not performance heavy."
#
# IT CREATES NO STORY, AND THAT IS THE SAFETY ARGUMENT. The original design for this feature called
# create_story inside a sweep -- ungated, that spawns a story per dragon per year forever. This repairs
# entries that ALREADY EXIST and never creates one, so that failure mode does not exist here. Dragons with
# no entry at all are still unreached; they are deliberately left until a release that has been booted.
#
# SELF-LATCHING. ln_repair_one_dragon_effect stamps ln_gene_storage_sync_2 as its first act, so each
# dragon is considered exactly once. On a healthy save the limit rejects every entry.
ln_repair_blank_storage_dragons_effect = {
	every_in_global_list = {
		variable = living_dragons
		# ── WHY THE BLANK TEST IS NOT IN THIS LIMIT ──────────────────────────────────────────────────
		# It was, for exactly one live run, and it printed 4,416 errors:
		#     Left side and right side during comparison were of different types
		#     (left was 'character', right was 'story')
		# Scripted-trigger parameters are TEXTUAL SUBSTITUTION, not scope capture. `DRAGON = this`
		# substitutes the literal token `this` into the trigger body, where it is re-evaluated inside
		# ln_storage_entry_is_blank_trigger's own any_in_global_list -- and there the current scope is the
		# STORY being iterated, not the dragon. So it compared the entry's dragon_id against the entry.
		# A limit cannot save a scope, so the dragon is captured in the body instead and passed explicitly.
		# NEVER pass `this` or `prev` into a parameterised trigger that opens a list of its own.
		limit = {
			NOT = { has_variable = ln_gene_storage_sync_2 }
			# 1.0.12b -- THE GATE THIS SWEEP SHIPPED WITHOUT, and the cause of the 17 Aug "glitch dragons"
			# thread (thefoxking554, MPREG: Vermithor fine at age 200, skeletal by 220 -- reported the DAY
			# 1.0.11 shipped this sweep). A HISTORICAL dragon's storage entry is geneless BY AGOT'S DESIGN:
			# story_dragon_variable_storage.on_setup copies genes only for `is_historical_dragon = no`
			# (agot_story_cycle_dragon_variables.txt:14). This sweep read that deliberate blankness as the
			# defect it hunts, "repaired" it by syncing the character's genes into the entry -- PARTIALLY,
			# because every copy is guarded on the character actually having that gene -- and a portrait that
			# then reads the entry finds some layers valued and some missing. A missing shading layer does
			# not render. That is the skeleton. AGOT's own boundary, restored: historical dragons are theirs.
			is_historical_dragon = no
		}
		save_temporary_scope_as = ln_blank_check_dragon
		if = {
			limit = { ln_storage_entry_is_blank_trigger = { DRAGON = scope:ln_blank_check_dragon } }
			ln_repair_one_dragon_effect = yes
		}
	}
}

# DOOR TWO (1.0.7): CONVERT, NEVER CLONE.

# Thomas: "Notice how the wights are handled -- those characters don't die, they get a trait... Does
# that not seem like a more straightforward way to specifically ensure that our mod causes no
# invisible dragons?" Yes. Everything below is that answer for the one death road we can actually
# stand in front of: dragon-vs-dragon combat, intercepted in zz_ln_dragon_combat_overrides.txt
# BEFORE the death lands. The dragon survives as itself -- its own genes, its own storage entry, its
# own Past Riders -- so no appearance is ever rebuilt and the invisible-dragon failure mode cannot
# occur, by construction.

# THE RECORDER AND THE RAISE ABOVE ARE NOW THE FALLBACK, NOT THE MAIN ROAD. Dragons also die to

# attempts (agot_dragon_slaying_events, 2), water wizards, and old age -- none of which funnels

# Root = a dragon that was about to die in dragon combat and passed ln_the_dead_claim_this_trigger.
# Called from the dragon_combat_result_grab override IN PLACE OF the death. Scopes available from
# the combat: scope:dsc_victor (the rider that beat it), scope:dsc_loser (its own rider).
ln_the_dead_claim_dragon_effect = {
	save_scope_as = ln_claimed_dragon

	# ── THE BOND IS CUT, AGOT'S OWN WAY ─────────────────────────────────────────────────────────────
	# agot_untame_dragon, not bare agot_unbond_dragon: untame is the one that pushes var:current_rider
	# onto the dragon's past_riders list (the history Thomas wants preserved) and leaves a surviving
	# rider correctly dragonless_dragonrider. If the rider is about to die in the same combat branch,
	# that trait sits on him for a tick and goes to the grave -- harmless, and his death handling then
	# finds no dragon to mourn, which is right: it is not his any more. The rider is remembered FIRST,
	# because untame is also what removes var:current_rider.
	if = {
		limit = { exists = var:current_rider }
		var:current_rider = { save_scope_as = ln_bereft_rider }
		agot_untame_dragon = {
			OWNER = scope:ln_bereft_rider
			DRAGON = scope:ln_claimed_dragon
		}
	}

	# ── THE CONVERSION ──────────────────────────────────────────────────────────────────────────────
	# Same name, new nickname, new trait -- Vhagar the Undead, and it is still Vhagar. The same marks
	# the raise path leaves, so every downstream guard treats both kinds identically.
	add_trait = ln_undead_dragon
	add_character_flag = ln_one_of_them   # 1.0.7: stamped at conversion -- see ln_is_one_of_them_trigger
	give_nickname = nick_the_undead
	set_variable = { name = color_grading  value = flag:darkest }
	set_variable = { name = color          value = flag:white }

	# The recolour, in BOTH places it has to land. The character-side palette is what any future
	# transfer or creation code reads...
	ln_apply_the_cold_effect = yes
	# ...but a LIVING dragon's portrait reads its gl_dragon_variable_storage entry -- on_setup MOVES
	# the gene variables off the character and into the story (agot_story_cycle_dragon_variables.txt),
	# so the ice palette must be written into the storage as well or the dragon keeps its living
	# colours. Historical dragons carry no storage genes at all (their look is baked DNA); for them
	# this writes nothing and the trait + nickname carry the meaning -- see the session doc, verified
	# vs hypothesis.
	ln_chill_the_storage_effect = yes

	# ── THEIRS NOW ──────────────────────────────────────────────────────────────────────────────────
	# The Night King is served first; his captains after; and a claim with every saddle filled is
	# still OWNED -- it waits in the Night King's court, not in the wild where a living claimant could
	# tame it back, until ln_others_take_a_dragon_effect's next three-year pass finds it a rider.
	if = {
		limit = {
			title:h_the_others.holder ?= {
				is_alive = yes
				NOT = { has_trait = dragonrider }
				NOT = { exists = var:current_dragon }
			}
		}
		title:h_the_others.holder = {
			if = {
				limit = { has_trait = dragonless_dragonrider }
				remove_trait = dragonless_dragonrider
			}
		}
		hidden_effect = {
			agot_tame_dragon = {
				TAMER = title:h_the_others.holder
				DRAGON = scope:ln_claimed_dragon
			}
		}
	}
	else_if = {
		limit = {
			title:h_the_others.holder ?= {
				any_vassal = { ln_wants_an_undead_mount_trigger = yes }
			}
		}
		title:h_the_others.holder = {
			random_vassal = {
				limit = { ln_wants_an_undead_mount_trigger = yes }
				save_scope_as = ln_new_cold_rider
				if = {
					limit = { has_trait = dragonless_dragonrider }
					remove_trait = dragonless_dragonrider
				}
			}
		}
		hidden_effect = {
			agot_tame_dragon = {
				TAMER = scope:ln_new_cold_rider
				DRAGON = scope:ln_claimed_dragon
			}
		}
	}
	else_if = {
		limit = {
			title:h_the_others.holder ?= {
				any_courtier = { ln_wants_an_undead_mount_trigger = yes }
			}
		}
		title:h_the_others.holder = {
			random_courtier = {
				limit = { ln_wants_an_undead_mount_trigger = yes }
				save_scope_as = ln_new_cold_rider
				if = {
					limit = { has_trait = dragonless_dragonrider }
					remove_trait = dragonless_dragonrider
				}
			}
		}
		hidden_effect = {
			agot_tame_dragon = {
				TAMER = scope:ln_new_cold_rider
				DRAGON = scope:ln_claimed_dragon
			}
		}
	}
	else = {
		# Every saddle is filled. Owned, unridden, waiting -- agot_set_as_owned_dragon courts it to the
		# Night King and flags it owned_dragon, which is what keeps the living from taming it back.
		if = {
			limit = { exists = title:h_the_others.holder }
			hidden_effect = {
				agot_set_as_owned_dragon = {
					OWNER = title:h_the_others.holder
					DRAGON = scope:ln_claimed_dragon
				}
			}
		}
	}

	# ── THE RIDER WHO LIVED ─────────────────────────────────────────────────────────────────────────
	# Some branches kill the rider with the dragon; some leave him standing in the snow watching it.
	# If he lives out the combat, he gets the card -- the dragon stamped on him as a variable rather
	# than a saved scope, per the standing rule: never rely on saved scopes in a queued card; rebuild
	# in immediate. Human riders only; an AI lord does not need the scene.
	if = {
		limit = {
			exists = scope:ln_bereft_rider
			scope:ln_bereft_rider = {
				is_alive = yes
				is_ai = no
			}
		}
		scope:ln_bereft_rider = {
			set_variable = { name = ln_lost_to_the_cold  value = scope:ln_claimed_dragon }
			trigger_event = { id = ln_dragonfall.1  days = 2 }
		}

			set_variable = { name = gene_dragon_secondary_value      value = 0.35 }
			set_variable = { name = gene_dragon_eye_color_hue        value = 0.0 }
			set_variable = { name = gene_dragon_eye_color_value      value = 0.0 }
			set_variable = { name = gene_dragon_horn_color_hue       value = 0.55 }
			set_variable = { name = gene_dragon_horn_color_value     value = 0.0 }
			# 1.0.15 -- blue fire, storage side. A LIVING dragon's portrait reads the storage entry, not
			# the character (see this effect's header), so the flame has to land here too. Same knob as

ln_undead_dragon_upkeep_effect = {
	debug_log = "LN_TRACE: yearly dragon repair upkeep fired"
	# 1.0.9b -- THE FULL SWEEP, MOVED TO WHERE IT ACTUALLY RUNS. Proven live in session 26:
	# on_game_start_after_lobby fires on NEW GAME only, never on loading a save, so the game-start
	# sweep in zz_ln_dragon_repair_on_actions.txt has never once run for a mid-campaign player.
	# This heartbeat provably runs on loaded saves (13 observed yearly fires, calendar stamp at the
	# final year). One-shot per stamp version; costs one global check a year once done; a save that
	# predates the global runs it once on the next yearly tick -- fails open, repairs silently.
	if = {
		limit = { NOT = { has_global_variable = ln_full_sweep_done_2 } }
		set_global_variable = { name = ln_full_sweep_done_2  value = yes }
		debug_log = "LN_TRACE: one-shot full repair sweep running from the yearly heartbeat"
		ln_repair_raised_dragons_full_sweep_effect = yes
	}
	ln_repair_raised_dragons_effect = yes
	# 1.0.11: the blank-entry sweep, which reaches dragons the two above cannot -- including AGOT's own.
	ln_repair_blank_storage_dragons_effect = yes
	# 1.0.15 -- BLUE FIRE, RETROACTIVELY. Undead dragons converted before this update carry the ice palette
	# but a living flame. One guarded pass: any undead dragon whose fire template is not yet the custom slot
	# gets the full cold reapplied (both sides -- the constants are fixed, so re-running is a byte-identical
	# no-op everywhere but the two new fire variables). Runs once per old dragon, then the guard holds.
	every_living_dragon = {
		limit = {
			has_trait = ln_undead_dragon
			NOT = { var:gene_dragon_fire_color_template ?= 2 }
		}
		debug_log = "LN_TRACE: retrofitting blue fire onto an undead dragon from an older save"
		ln_apply_the_cold_effect = yes
		ln_chill_the_storage_effect = yes
	}
	every_living_dragon = {
		limit = {
			has_trait = ln_undead_dragon
			exists = var:ln_raised_from
			NOT = { any_in_list = { variable = past_riders  count >= 1 } }
			var:ln_raised_from = { any_in_list = { variable = past_riders  count >= 1 } }
		}
		save_scope_as = ln_clone_dragon
		var:ln_raised_from = {
			every_in_list = {
				variable = past_riders
				scope:ln_clone_dragon = {
					add_to_variable_list = {
						name = past_riders
						target = prev
					}
				}
			}
		}
	}
}

#
# WHAT IT DID: for every storage entry whose dragon is historical, it removed all 42 gene variables, on the
# belief that a historical dragon's entry is BLANK BY DESIGN and that anything in it was pollution.
#
# WHY THAT WAS WRONG: `story_dragon_variable_storage.on_setup` does skip its gene block for historical dragons
# (agot_story_cycle_dragon_variables.txt:14) -- that part was read correctly. But a SECOND effect fills those
# entries instead: `agot_historical_dragon_transfer_vars_to_story_cycle_effect`
# (00_agot_dragon_effects.txt:2302) takes ~40 curated gene values as $PARAM$s and writes them straight onto
# the story -- `scope:story = { set_variable = { name = gene_dragon_primary_color_value ... } }`. It is called
# from agot_historical_dragons_setup at game start, for ~60 named dragons, and the comment in that file reads
# "Balerion, curated dna".

# this effect deleted it. Every Balerion, Caraxes, Vermithor and Vhagar in every running save turned white --

# conclusion contradicted it without ever checking. The standing rule is "a comment asserting precedent is not

# ln_strip_living_regiments_effect -- 1.0.16. THE DEAD DO NOT FIGHT WITH BOWMEN.
#
# Thomas, from a live 27 Aug run, looking at the Night King's realm strength: "the Night King now has bowmen,
# men-at-arms. That's absolutely unacceptable. Wildling Raiders, absolutely unacceptable. He should only have
# access to undead stuff." The tooltip read: Risen Dead 3500, Ice Spiders 500 -- and Wildling Raiders 320,
# Cannibal Bows 240, Bowmen 200.
#
# ── WHERE THEY CAME FROM, AND IT IS NOT THE 1.0.16 CULTURE WORK ───────────────────────────────────────────
# Checked before changing anything, because the traditions and innovations added this release were the
# obvious suspects. They are not the cause:
#   * othertraditions.txt contains ZERO unlock_maa and zero maa references.
#   * none of the seven tribal innovations in history/cultures/frostborn.txt carries unlock_maa.
#   * wildling_raiders and cannibal_bows gate on valid_for_maa_trigger = { PARAMETER = unlock_maa_* }, which
#     resolves to `culture = { has_cultural_parameter = X }` (game/common/scripted_triggers/
#     00_military_triggers.txt:17-26). The parameter is granted by tradition_agot_forestmen
#     (AGOT 00_agot_unique_traditions.txt:2600-2614), which frostborn does not have and never had.
# So these are REGIMENTS THE LORDS ALREADY HAD WHEN THEY WERE RAISED. mutate_into_other_effect converts a
# conquered ruler and sets his culture to frostborn a few lines above this call -- but CK3 does not disband
# regiments on a culture change, so his crossbowmen keep standing, now under a dead man's banner. Vanilla
# `bowmen` needs no cultural parameter at all (its can_recruit only excludes nomadic and herder governments),
# so that one was always available to anybody the dead raised.
#
# ── THE PATTERN IS THE ONE NEXT DOOR ──────────────────────────────────────────────────────────────────────
# This is the same shape as the v0.2.1 dragonglass strip in agot_long_night_scripted_effects.txt -- "the dead
# do not fight with dragonglass", added off an identical live-run report where raised lords kept the living's
# anti-wight kit. Same placement, deliberately OUTSIDE the one-shot guard, so the heartbeat re-runs it on
# existing wights and running saves are cleaned on their next tick rather than needing a new sweep.
#
# ── WHY is_unit_type = Other_infantry IS THE RIGHT FILTER ─────────────────────────────────────────────────
# Both undead units declare `type = Other_infantry` (long_night_maa_types.txt:10, zz_ln_wight_horde.txt:15),
# and that type name appears in NO men-at-arms file in base CK3 or AGOT -- verified. So it identifies our two
# units exactly, with no allowlist to maintain as either mod adds units.
# `is_unit_type` is a real maa_regiment-scope trigger (vanilla uses it in 00_laamp_triggers.txt:11-12) and
# `destroy_maa_regiment` is the effect this mod already uses twice
# (agot_long_night_scripted_effects.txt:4480, zz_long_night_plus_aftermath_effects.txt:377).
#
# SAFE FOR THE NIGHT KING'S OWN HOST. His wight horde is Other_infantry and is kept. His elite core comes
# from spawn_army, which is army-attached rather than an owned MAA regiment, so every_maa_regiment does not
# reach it at all.
ln_strip_living_regiments_effect = {
	every_maa_regiment = {
		limit = { NOT = { is_unit_type = Other_infantry } }
		destroy_maa_regiment = yes
	}
}
