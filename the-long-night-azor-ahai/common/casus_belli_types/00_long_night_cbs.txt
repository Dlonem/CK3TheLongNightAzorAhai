### War for the Dawn CB ###

war_for_the_dawn_cb = {
	group = invasion
	icon = invasion

	allow_hostages = no
	interface_priority = 110

	allowed_for_character = {
	# 1.1.0a -- WAS `NOT = { has_trait = other_trait }`. The trait test is not the identity test: a lord
	# raised by mutate_into_other_effect carries wight_government and, as Thomas's post-war save showed in
	# 1.0.6a, often no trait at all (see ln_is_one_of_them_trigger's own header). One of the dead could
	# therefore declare the War for the Dawn on his own king.
	ln_is_one_of_them_trigger = no
	}

	allowed_against_character = {
		has_trait = nightking
		NOT = { has_character_flag = war_for_dawn_cooldown }
		#has_character_flag = long_night_cooldown 
	}

	target_titles = all
	ignore_effect = change_title_holder

	white_peace_possible = no 

	ai_score = {
		value = 100 # TODO
	}

	should_invalidate = {
		# 1.0.15 -- A WON WAR CANNOT INVALIDATE. The dawn cleanup destroys h_the_others at the moment the
		# Night King dies, and before this guard the engine would read "defender lost his kingdom title"
		# and invalidate a war the attackers had just WON -- no victory screen, no payouts, erine120's
		# report verbatim. The death-time sweep in zz_ln_catalyst_effects.txt ends the war as an attacker
		# victory first; this guard is the belt to that sweep's braces.
		NOT = { has_global_variable = war_for_dawn_victory }
		scope:defender = {
			NOT = { primary_title.tier >= tier_kingdom }
		}
	}

	cost = {
}

	on_declaration = {
		on_declared_war = yes
		#scope:defender = { every_vassal = { limit = { NOT = { has_faith = faith:cold_gods } } scope:war =  { add_attacker = prev } } }
	}

	on_victory_desc = {
		first_valid = {
			desc = war_for_the_dawn_victory_desc
		}
	}

	on_victory = {
		set_global_variable = {
			name = war_for_dawn_victory
			value = yes
		}
		# ══ 1.0.15 -- THE WAR ONLY TRULY ENDS WHEN HE DIES, AND ON SCREEN. ══════════════════════════════
		# Thomas, 24 Aug: "the war for the dawn only ends if the night king dies" + "the destiny scene
		# picks." The old line here killed him instantly, off screen, with killer = scope:attacker -- so a
		# score victory handed Azor Ahai Reborn to the Iron Throne holder in his solar (breakfastenjoyer's
		# report), and no player ever saw the end of the Long Night happen.
		#
		# Now: if he is still breathing when the score is settled, the cornering runs -- the worthiest
		# hand on the attacker side (rulers AND their knights, ln_dawn_worthiness_value) gets the scene,
		# and the death happens there, killer = that champion, two days later. A day-14 failsafe
		# (ln_dawn.12) guarantees the death even if the scene is lost, so this branch can never produce a
		# living Night King after a lost war.
		#
		# If he is ALREADY dead -- the death is what won the war, via the catalyst sweep -- there is
		# nothing left to do here: the reveal, the trait and the cleanup all ran from his death.
		if = {
			limit = { scope:defender = { is_alive = yes } }
			ln_wfd_cornered_effect = yes
		}
	}

	on_defeat_desc = {
		first_valid = {
			desc = war_for_the_dawn_defeat_desc
		}
	}

	on_defeat = {
		#EP2 accolade glory gain for winning against higher ranked enemy
		scope:defender = { accolade_defender_war_end_glory_gain_low_effect = yes }
		scope:defender = { add_character_flag = { flag = war_for_dawn_cooldown years = 5 } }
		# Prestige loss for the attacker
		scope:attacker = {
			add_prestige = -500
		}
		
		# 10 Aug: the create used to sit HERE, outside the loop, while the resolve sat inside
		# every_war_attacker. A title&vassal change is single-use -- created, filled, resolved, gone --
		# so the first attacker consumed scope:change and every attacker after it hit "No transfer type
		# set for title&vassal change": 73 on the setup_invasion_cb line and 73 on the resolve, per run.
		# Now created per attacker, inside the loop, immediately before it is used.
		
	scope:war = { every_war_attacker = {
				create_title_and_vassal_change = {
					type = conquest
					save_scope_as = change
					add_claim_on_loss = yes
				}
				setup_invasion_cb = {
				titles = target_titles
				attacker = scope:defender
				defender = this
				change = scope:change
				take_occupied = yes
			}
			resolve_title_and_vassal_change = scope:change


				}
			}
			
		# Prestige for the defender
		scope:defender = {
			add_prestige = 5000
		}
		
		# Prestige for the attacker's war allies
		add_from_contribution_attackers = {
			prestige = minor_prestige_value
			opinion = {
				modifier = contributed_in_war
			}
		}

		# Prestige for the defender's war allies
		add_from_contribution_defenders = {
			prestige = minor_prestige_value
			opinion = {
				modifier = contributed_in_war
			}
		}

		# Truce
		add_truce_attacker_defeat_effect = yes
	}

	on_white_peace_desc = {
		first_valid = {
			desc = great_ranging_white_peace_desc
		}
	}

	on_invalidated_desc = {
		first_valid = {
			desc = great_ranging_invalidated_desc
		}
	}

	transfer_behavior = transfer

	on_primary_attacker_death = inherit
	on_primary_defender_death = inherit

	attacker_allies_inherit = yes
	defender_allies_inherit = yes
	attacker_capital_gives_war_score = no 				# No warscore from attacker capital
	defender_capital_gives_war_score = no 				# No warscore from defender capital
	imprisonment_by_attacker_give_war_score = no 		# No warscore when the attackers imprisons people
	imprisonment_by_defender_give_war_score = no 		# No warscore when the defenders imprisons people
	# 1.0.16 -- 300 -> 200. THESE NUMBERS HAVE NEVER BEEN PLAYTESTED IN ANY VERSION OF THIS MOD.
	# Upstream LN+ wrote `attacker_score_from_battle_scale` -- SINGULAR "battle" -- which is not a CK3 key:
	# 0 hits across base CK3 against 57 for the plural. Both of upstream's lines were inert, so upstream's
	# "300" never did anything. This fork corrected the spelling, which switched a live 300 scale on for the
	# first time -- DOUBLE vanilla conquest's 150 -- with a 350 cap, no white peace, and the defenders capped
	# at 50 from occupation. That is a large part of why the Others lose wars they should not.
	# 200 keeps the War for the Dawn decisive without making it the easiest war in the game.
	attacker_score_from_battles_scale = 200
	defender_score_from_battles_scale = 200
	attacker_score_from_occupation_scale = 10
	defender_score_from_occupation_scale = 10
	max_attacker_score_from_battles = 350
	max_defender_score_from_occupation = 50
	war_name = WAR_FOR_THE_DAWN_WAR_NAME
	war_name_base = WAR_FOR_THE_DAWN_WAR_NAME_BASE
	cb_name = WAR_FOR_THE_DAWN_WAR_CB

}

			# war settlement moved c_hide_point and c_taldrsek (the legal beyond-Wall goal) AND c_sullen_hills

				defender = this
				change = scope:change

				# North still free. The goal still transfers in full; occupied land outside it now has to be
				# contiguous with ground they already hold. See zz_ln_contiguity_effects.txt.
				take_occupied = no
			}
			}
			}

		# 1.0.10 -- see the twin block in on_defeat: a white peace against the Watch is still the
		# Wall holding, and it refills the stage-two clock the same way.
		if = {
			limit = {
				NOT = { has_global_variable = ln_castle_black_fell }
				scope:defender ?= { capital_province ?= { geographical_region = world_westeros_the_wall } }
				scope:attacker ?= { has_character_flag = ln_wall_gate_started }
			}
			scope:attacker = { add_character_flag = { flag = ln_wall_gate_window  years = 4 } }
		}

		#EP2 accolade glory gain for doing ok against higher ranked enemy

		scope:defender = { accolade_defender_war_end_glory_gain_med_effect = yes }

		# Allies on both sides get full prestige value for helping in the war, based on their war participation.
		modify_allies_of_participants_fame_values = {

	on_defeat = {
		scope:attacker = { show_pow_release_message_effect = yes }
		# FIX (v0.1.0): was the un-braced `add_character_flag = long_night_cooldown years = 5`, which creates a
		# PERMANENT flag and leaves `years = 5` as an orphan token. That flag gates BOTH the 30-day expansion pulse
		# (the_long_night.txt) and allowed_for_character on both invasion CBs -- so the first time the Night King lost
		# any war he froze forever. That is the "sits idle for years" report. Same file uses the braced form
		# correctly for war_for_dawn_cooldown a few lines above.
		# 1.0.10 -- WAS A FLAT FIVE YEARS, and it is the single largest source of the zero-conquest years
		# measured across two runs (eight of them in the 8258-8291 save). A defeat should cost the dead a
		# campaign season, not half a generation of the map standing still -- the living winning a war is
		# supposed to buy them breathing room, not switch the Long Night off. Rolled one to three years,
		# so a defeat is still a real setback and no two defeats read the same.
		scope:attacker = {
			# 1.0.16 -- THE TEMPO DIAL, defeat leg. On Strong and Insane a lost war costs the dead a
			# campaign season or two -- 6-18 months rolled -- not 1-3 years; Weak and Standard keep the
			# 1-3 year freeze unchanged. Measured need (106-131 AC run, Strong): the 22- and 34-month
			# zero-conquest gaps in the falls table are this flag standing. ln_nk_has_been_beaten below
			# is untouched: the endgame still opens the moment the dead lose once, on every tier.
			if = {
				limit = {
					OR = {
						has_game_rule = others_strong_invasion
						has_game_rule = others_insane_invasion
					}
				}
				random_list = {
					30 = { add_character_flag = { flag = long_night_cooldown  months = 6 } }
					45 = { add_character_flag = { flag = long_night_cooldown  months = 12 } }
					25 = { add_character_flag = { flag = long_night_cooldown  months = 18 } }
				}
			}
			else = {
				random_list = {
					30 = { add_character_flag = { flag = long_night_cooldown  years = 1 } }
					45 = { add_character_flag = { flag = long_night_cooldown  years = 2 } }
					25 = { add_character_flag = { flag = long_night_cooldown  years = 3 } }
				}
			}
		}

		# 1.0.16 -- THE DEAD CAN BE BEATEN, AND NOW THE WORLD KNOWS IT. Thomas's own design call:
		# "if the Night King can suffer a defeat, perhaps that is a flag to indicate that the War for the
		# Dawn is doable. My logic: if the Night King can lose a battle or a war, perhaps that's the
		# opportunity for the War for the Dawn."
		#
		# In THIS cb scope:attacker is the Night King -- the Others are the invaders -- so on_defeat here is
		# the one moment in the game that means "the dead lost a war". That is the correct trigger for the
		# endgame becoming available, and it fixes KhanSaru's Hightower report at the ROOT rather than by
		# gating around it: Oldtown declaring the War for the Dawn in year three is absurd BECAUSE NOBODY
		# HAS BEATEN THE DEAD YET. Now nobody can declare it until somebody has.
		#
		# DELIBERATELY A NEW GLOBAL, NOT war_for_dawn_cooldown. That flag already means two things, has
		# three writers and ZERO clearers, and two of those writers stamp it for TEN YEARS at game start
		# (agot_long_night_scripted_effects.txt:336, the_long_night.txt:18). Overloading it a third time
		# would have been the cheap fix and an unreadable one.
		# Set once and left standing for the wave: the living proved it can be done, and that does not stop
		# being true. The Cycle's re-arm clears the wave's other latches and should clear this with them.
		set_global_variable = { name = ln_nk_has_been_beaten  value = yes }
		debug_log = "LN_TRACE ENDGAME: the dead lost a war - the War for the Dawn is now open"
		# 1.0.10 -- THE WALL CAN HOLD (Thomas). If the man who just threw the dead back is seated on
		# the Wall itself, the stage-two clock refills: a Watch actually holding the Wall IS the
		# fantasy, and the escape clock exists to break stalls, not to punish garrisons that win.
		# Fenced on Castle Black standing and on the clock having ever started; both fail open.
		if = {
			limit = {
				NOT = { has_global_variable = ln_castle_black_fell }
				scope:defender ?= { capital_province ?= { geographical_region = world_westeros_the_wall } }
				scope:attacker ?= { has_character_flag = ln_wall_gate_started }
			}
			scope:attacker = { add_character_flag = { flag = ln_wall_gate_window  years = 4 } }
		}
		# Legitimacy
		add_legitimacy_attacker_defeat_effect = yes

		# Influence
		add_influence_attacker_defeat_effect = yes

		#EP2 accolade glory gain for winning against higher ranked enemy
		scope:defender = { accolade_defender_war_end_glory_gain_high_effect = yes }

		# Prestige loss for the attacker
		scope:attacker = {
			pay_short_term_gold_reparations_effect = {
				GOLD_VALUE = 5
			}
			add_prestige = {
				value = massive_prestige_value
				multiply = -1.0
			}
		}
		
		setup_invasion_cb = {
			titles = target_titles
			attacker = scope:attacker
			defender = scope:defender
			claimant = scope:claimant
			victory = no
		}

		# Attacker loses Prestige, all other war participants gain Prestige (Defender gains full prestige, all allies on both sides gain based on war contribution).
		modify_all_participants_fame_values = {
			WINNER = scope:defender
			LOSER = scope:attacker
			FAME_BASE = scope:cb_prestige_factor # Set by 'setup_claim_cb'
			IS_RELIGIOUS_WAR = no
			WINNER_FAME_SCALE = scale_10_war_defender_win
			LOSER_FAME_SCALE = -10
			WINNER_ALLY_FAME_SCALE = 10
			LOSER_ALLY_FAME_SCALE = 10
		}
	
		#add_truce_attacker_defeat_effect = yes
		
		scope:attacker = {
			save_temporary_scope_as = loser
		}
		on_lost_aggression_war_discontent_loss = yes
	}

# THE DUCHY INVASION -- and why this is now the default.
#
# Thomas, watching a live run: the Others' ownership appears at Winterfell and spreads outward, when it should
# be the reverse -- the countryside lost first, the capital encircled, the great seat last.
#
# The cause was not target selection and not army pathing. other_cb is target_title_tier = KINGDOM, and its
