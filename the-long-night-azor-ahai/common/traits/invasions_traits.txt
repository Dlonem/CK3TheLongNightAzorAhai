# ══════════════════════════════════════════════════════════════════════════════════════════════════════════════
# RESURRECTED BY R'HLLOR -- and every time, a little less of you comes back.
#
# THOMAS'S IDEA, and it is the right one: "if you see the opportunity to have a trait track for it developing...
# the more times you get resurrected, you can add trait XP into whatever event handles the resurrection status.
# That way if it occurs, it'll basically be a reapplication but increasing the XP... the same way Beric had lost
# an eye, or one-armed, one-legged kind of thing."
#
# That is exactly Beric Dondarrion. Six deaths in the books, and by the last one he has lost an eye, a great
# deal of his face, and most of his memory -- "every time I come back, I'm a bit less of me." Thoros says it
# plainly: the fire gives him back, but not all of him.
#
# SO THE TRACK GOES DOWN, NOT UP. This is the only progression in the mod where advancing makes you worse at
# being a person and better at frightening people. Dread and piety climb; health, fertility and how anyone
# feels about you fall off a cliff. The base trait keeps the numbers Thomas already tuned -- the track adds
# character, not power.
#
# WHAT STOPS IT BEING FARMED is not a counter, it is ln_priest_can_give_the_last_kiss_trigger: a priest who has
# given the last kiss can never give another. Every return costs a NEW priest, permanently. Which is also why
# Beric needed Thoros and Thoros only ever managed Beric.
# ══════════════════════════════════════════════════════════════════════════════════════════════════════════════
resurrected_by_rhllor = {
	icon = resurrected.dds
	category = fame

	martial = 6
	prowess = 12
	stewardship = 3

	ruler_designer_cost = 300


	negate_health_penalty_add = 0.9
	fertility = -0.35
	health = 1.5
	life_expectancy = 20
	stress_gain_mult = -0.3

	diplomatic_range_mult = 1.2
	same_faith_opinion = 25
	monthly_piety = 1.00
	flag = lower_chance_of_dying_in_combat

	# The name changes with the count, because "Resurrected by R'hllor" stops covering it after the third time.
	name = {
		first_valid = {
			triggered_desc = {
				trigger = { NOT = { exists = this } }
				desc = trait_resurrected_by_rhllor
			}
			triggered_desc = {
				trigger = { has_trait_xp = { trait = resurrected_by_rhllor  track = returned  value >= 100 } }
				desc = trait_resurrected_by_rhllor_4
			}
			triggered_desc = {
				trigger = { has_trait_xp = { trait = resurrected_by_rhllor  track = returned  value >= 75 } }
				desc = trait_resurrected_by_rhllor_3
			}
			triggered_desc = {
				trigger = { has_trait_xp = { trait = resurrected_by_rhllor  track = returned  value >= 50 } }
				desc = trait_resurrected_by_rhllor_2
			}
			desc = trait_resurrected_by_rhllor
		}
	}

	tracks = {
		returned = {
			# ONE. You have a scar that will not close and everybody in the hall has decided not to mention it.
			25 = {
				dread_baseline_add = 5
				monthly_piety = 0.5
				health = -0.25
			}
			# TWO. People stop meeting your eye. The ones who share your faith stop being able to look away.
			50 = {
				dread_baseline_add = 10
				same_faith_opinion = 10
				health = -0.5
				fertility = -0.25
			}
			# THREE. You are frightening now, and the part of you that minded is one of the things that did not
			# come back.
			75 = {
				dread_baseline_add = 15
				general_opinion = -10
				attraction_opinion = -15
				prowess = 2
				health = -1
				fertility = -0.5
			}
			# FOUR. There is not much of the man left. What there is, is very hard to kill.
			100 = {
				dread_baseline_add = 25
				general_opinion = -20
				attraction_opinion = -25
				prowess = 4
				health = -1.5
				fertility = -1
				monthly_piety = 1
			}
		}
	}
}

# ── THE UNDEAD DRAGON ───────────────────────────────────────────────────────────────────────────────────────
# What Vhagar becomes. Thomas asked for both a nickname and a trait so that the thing reads correctly on the
# character sheet: it is still the dragon everyone knew, and it is very obviously not that dragon any more.
#
# It is also LOAD-BEARING, not decoration. ln_record_fallen_dragon_effect refuses to record any dragon carrying
# this trait, which is one of the three guards that stop a raised dragon being raised again -- living, undead,
# dead, and dead is the end of it.
#
# No health penalty and no combat penalty: it is dead already, and the whole point of the thing is that it is a
# siege engine that does not tire. The immortal trait comes with it from create_character.
ln_undead_dragon = {
	# icon: reusing the wight icon rather than shipping a texture that does not exist yet -- a missing trait
	# icon renders as a "?" in the character sheet, which is the exact class of bug that cost a whole pass on
	# the struggle panel. Worth its own art later.
	icon = wight.dds
	category = fame
	shown_in_ruler_designer = no
	random_creation = 0
	birth = 0

	# Nothing gets inherited out of a thing the Others own. Same reasoning as wight_trait.
	inheritance_blocker = all
	claim_inheritance_blocker = all

	# NOTE ON FIELDS: an unknown property inside a trait is silently read as a MODIFIER on the holder, so every
	# line below was checked against the base game's own trait file. `monthly_dread` is NOT a trait field
	# anywhere in vanilla -- dread_baseline_add is the one that exists (43 uses) and it is what is used here.
	dread_baseline_add = 20
	health = 1
	same_opinion = -30
	general_opinion = -20

	# ── 1.0.7, DOOR TWO ─────────────────────────────────────────────────────────────────────────────
	# A CONVERTED dragon never passes through create_character, so nothing hands it the vanilla
	# `immortal` trait the way the clone template did. The trait itself must therefore say what
	# "already dead" means. Diffed against other_trait and taking only what applies to a dragon:
	# no death by old age, no decay with age. immortal_fertility is deliberately NOT taken -- that
	# flag preserves fertility, and nothing breeds out of a thing the Others own.
	immortal = yes
	life_expectancy = 12000
	no_prowess_loss_from_age = yes
	flag = is_immortal
	flag = immortal_visuals
	flag = immortal_incapability
}
