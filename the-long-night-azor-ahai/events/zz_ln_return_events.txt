namespace = ln_return

# AGOT: The Long Night & Azor Ahai -- the ultimatum over your own hall.
# See common/decisions/zz_ln_return_with_fire_decisions.txt for the design note.

ln_return.1 = {
	type = character_event
	window = fullscreen_event
	title = ln_return.1.t
	desc = ln_return.1.desc
	theme = dread
	left_portrait = { character = root  animation = personality_bold }
	right_portrait = { trigger = { exists = scope:ln_squatter }   character = scope:ln_squatter  animation = fear }

	# 1.0.14 DEFENSIVE REBUILD. Rebuilt from a variable rather than relying on the scope saved at the
	# send site. NOTE, corrected 24 Aug: saved scopes DO survive a delayed trigger_event -- verified
	# against vanilla, where court.4751 (90-day delay, one sender, no exists guard) and
	# stewardship_duty.1063 (re-sends itself at 1820 days, never re-saves) would both be dead
	# content otherwise. The boundary scopes do NOT cross is an on_action, not a delay.
	# This pattern is kept anyway because it is strictly more robust: it also survives a sender
	# that never saved the scope on the branch it took, which is the failure that actually bit us.
	# This one is FULLSCREEN. It failing silently means the whole ultimatum never happens.
	trigger = {
		has_variable = ln_lost_seat
		var:ln_lost_seat ?= { exists = holder }
	}

	immediate = {
		debug_log = "LN_TRACE RETURN: the lord who left is at his own gate"
		var:ln_lost_seat ?= { save_scope_as = ln_old_seat }
		var:ln_lost_seat.holder ?= { save_scope_as = ln_squatter }
		var:current_dragon ?= { save_scope_as = ln_dragon }
	}

	# ── CROWN ME, OR I BURN IT ──────────────────────────────────────────────────────────────────────────────
	# The bluff. Whether it works is decided by what he can see: how much of you he already fears, and how much
	# he has to lose. A confident man with soldiers behind him calls it, and then everybody knows you blinked.
	option = {
		name = ln_return.1.a
		ai_chance = {
			base = 35
			modifier = { add = 15  has_trait = just }
			modifier = { add = 15  has_trait = calm }
			modifier = { add = 10  has_trait = generous }
		}
		add_character_flag = ln_return_made_the_demand

		# THE ODDS ARE HEAVILY TOWARDS HIM YIELDING, and Thomas is right that they should be: "if they have a
		# dragon, make sure there's a good opportunity for them to just take it easily, without burning the seat
		# down." A dragon standing in the yard is not a negotiating position, it is a fact. 75/25 before any
		# modifier, and dread or a rung on the prophecy ladder pushes it close to certain.
		#
		# It is not 100 because the refusal scene is the best writing in the chain and because a man who has
		# never once been told no by anybody is a less interesting character. A brave, stubborn squatter facing
		# a lord with no dread can still look up and say it.
		random_list = {
			75 = {
				modifier = { add = 40  dread >= 50 }
				modifier = { add = 25  dread >= 25 }
				modifier = { add = 20  has_trait = azor_ahai_reborn }
				modifier = { add = 15  has_trait = chosen_by_rhllor }
				modifier = { add = 20  scope:ln_squatter = { has_trait = craven } }
				modifier = { add = -25  scope:ln_squatter = { has_trait = brave } }
				modifier = { add = -20  scope:ln_squatter = { has_trait = stubborn } }

				# HE YIELDS.
				create_title_and_vassal_change = {
					type = granted
					save_scope_as = ln_return_change
					add_claim_on_loss = yes
				}
				scope:ln_old_seat = {
					change_title_holder = {
						holder = root
						change = scope:ln_return_change
					}
				}
				resolve_title_and_vassal_change = scope:ln_return_change
				remove_character_flag = dawn_refugee
				add_character_flag = ln_restored_by_fire
				add_dread = 25
				add_prestige = 300

				# THEY REMEMBER THAT YOU LEFT. Thomas: "obviously I abandoned it, so there's gonna be a little
				# -- I have to build up my loyalty again." Right, and it is the detail that keeps the easy road
				# from being a free one. You get the hall back in an afternoon; you get the county back over
				# years. Control, not opinion, because opinion is about the lord and this is about the place.
				scope:ln_old_seat = {
					change_county_control = -40
					add_county_modifier = {
						modifier = ln_the_lord_who_left_modifier
						years = 15
					}
				}
				scope:ln_squatter = { add_opinion = { modifier = hate_opinion  target = root  opinion = -60 } }
				# ⚠ THIS COMMENT WAS WRONG AND THE CODE IS KEPT ANYWAY (corrected 1 Sep). It used to read "the
				# scope will not survive that." Saved scopes DO survive a delayed trigger_event -- verified
				# against vanilla and written up at zz_ln_dragon_wall_events.txt:56-62 (court.4751 delays 90
				# days with no re-save; stewardship_duty.1063 re-sends itself at 1820 days and never
				# re-saves). The boundary scopes do not cross is an on_action, not a delay.
				#
				# Left in place because the variable is still the more robust handle -- it survives a sender
				# that never saved the scope on the branch it took, which is the failure that has actually
				# bitten this project. Two live comments in one codebase disagreeing about an engine rule is
				# how a future session codes against the wrong one, so this one now says which is which.
				# ln_return.3 relies on the scope surviving and is correct to: ln_return.1's own trigger
				# (:25) requires var:ln_lost_seat to have a holder, so ln_squatter is always saved.
				set_variable = { name = ln_yielded_squatter  value = scope:ln_squatter }
				trigger_event = { id = ln_return.2  days = 5 }
			}
			25 = {
				modifier = { add = 30  scope:ln_squatter = { has_trait = brave } }
				modifier = { add = 25  scope:ln_squatter = { has_trait = stubborn } }
				modifier = { add = -35  dread >= 50 }
				modifier = { add = -20  has_trait = azor_ahai_reborn }

				# HE CALLS IT.
				add_character_flag = ln_return_was_refused
				add_pressed_claim = scope:ln_old_seat
				add_prestige = -150
				add_dread = -10
				stress_impact = { base = medium_stress_impact_gain }
				trigger_event = { id = ln_return.3  days = 5 }
			}
		}
	}

	# ── BURN IT ─────────────────────────────────────────────────────────────────────────────────────────────
	# You get nothing. That is the entire point of the option and it is why it is here.
	option = {
		name = ln_return.1.b
		ai_chance = {
			base = 15
			modifier = { add = 25  has_trait = wrathful }
			modifier = { add = 25  has_trait = sadistic }
			modifier = { add = 15  has_trait = callous }
			modifier = { add = -30  has_trait = compassionate }
		}
		add_character_flag = ln_burned_my_own_hall
		add_dread = 40
		add_prestige = 50
		add_piety = -100

		scope:ln_old_seat = {
			add_county_modifier = {
				modifier = ln_dragonfire_scar_modifier
				years = 40
			}
			change_county_control = -75
		}
		scope:ln_squatter = { add_opinion = { modifier = hate_opinion  target = root  opinion = -100 } }
		every_vassal = { add_opinion = { modifier = scared_opinion  target = root  opinion = -15 } }
		stress_impact = { base = major_stress_impact_gain  callous = medium_stress_impact_loss  sadistic = major_stress_impact_loss }
		trigger_event = { id = ln_return.4  days = 10 }
	}

	# ── KILL HIM, AND HIS ────────────────────────────────────────────────────────────────────────────────────
	option = {
		name = ln_return.1.c
		ai_chance = {
			base = 10
			modifier = { add = 30  has_trait = sadistic }
			modifier = { add = 25  has_trait = vengeful }
			modifier = { add = 10  has_trait = paranoid }
			modifier = { add = -40  has_trait = compassionate }
		}
		add_character_flag = ln_return_took_it_in_blood

		create_title_and_vassal_change = {
			type = usurped
			save_scope_as = ln_return_change
			add_claim_on_loss = yes
		}
		scope:ln_old_seat = {
			change_title_holder = {
				holder = root
				change = scope:ln_return_change
			}
		}
		resolve_title_and_vassal_change = scope:ln_return_change
		remove_character_flag = dawn_refugee
		add_character_flag = ln_restored_by_fire

		scope:ln_squatter = {
			every_close_family_member = {
				limit = { is_alive = yes  is_landed = no }
				death = { death_reason = death_burned  killer = root }
			}
			death = { death_reason = death_burned  killer = root }
		}

		add_dread = 60
		add_prestige = 100
		add_piety = -200
		every_vassal = { add_opinion = { modifier = scared_opinion  target = root  opinion = -30 } }
		stress_impact = { base = major_stress_impact_gain  sadistic = major_stress_impact_loss  callous = medium_stress_impact_loss  compassionate = major_stress_impact_gain }
		trigger_event = { id = ln_return.4  days = 10 }
	}

	# ── TRIAL BY COMBAT ─────────────────────────────────────────────────────────────────────────────────────
	# Thomas's addition: "maybe you can do a duel, like a trial by combat." It is the honourable road and the
	# stupid one at the same time -- you have a dragon in the yard and you are choosing to settle it with a
	# sword, which is exactly the sort of thing that makes a man a story afterwards.
	#
	# Prowess decides it, and the resurrected trait's +12 is doing real work here. Losing does NOT kill you --
	# it takes the hall off the table and leaves you standing in front of everyone having been beaten by the
	# man who kept your granary honest.
	option = {
		name = ln_return.1.e
		ai_chance = {
			base = 15
			modifier = { add = 25  has_trait = brave }
			modifier = { add = 10  has_trait = arrogant }
			modifier = { add = 10  has_trait = just }
		}
		add_character_flag = ln_return_fought_for_it

		random_list = {
			50 = {
				modifier = { add = { value = prowess  multiply = 2 } }
				modifier = { add = { value = scope:ln_squatter.prowess  multiply = -2 } }
				modifier = { add = 15  has_trait = resurrected_by_rhllor }
				modifier = { add = 10  ln_has_deed_valyrian_steel_trigger = yes }

				# YOU WIN. He yields the seat in front of the whole yard and nobody had to burn.
				create_title_and_vassal_change = {
					type = granted
					save_scope_as = ln_return_change
					add_claim_on_loss = yes
				}
				scope:ln_old_seat = {
					change_title_holder = {
						holder = root
						change = scope:ln_return_change
					}
					# Far gentler than taking it from the air. They watched you win it.
					change_county_control = -10
				}
				resolve_title_and_vassal_change = scope:ln_return_change
				remove_character_flag = dawn_refugee
				add_character_flag = ln_restored_by_fire
				add_character_flag = ln_won_it_with_a_sword
				add_prestige = 400
				add_piety = 100
				increase_wounds_no_death_effect = { REASON = death_duel }
				scope:ln_squatter = { add_opinion = { modifier = respect_opinion  target = root  opinion = 40 } }
			}
			50 = {
				modifier = { add = { value = scope:ln_squatter.prowess  multiply = 2 } }
				modifier = { add = { value = prowess  multiply = -2 } }
				modifier = { add = -15  has_trait = resurrected_by_rhllor }

				# HE WINS. You keep your life and your dragon and nothing else.
				add_character_flag = ln_return_lost_the_duel
				increase_wounds_no_death_effect = { REASON = death_duel }
				add_prestige = -200
				add_dread = -15
				stress_impact = { base = major_stress_impact_gain  arrogant = major_stress_impact_gain }
				scope:ln_squatter = { add_opinion = { modifier = respect_opinion  target = root  opinion = 25 } }
				trigger_event = { id = ln_return.3  days = 5 }
			}
		}
	}

	# ── LAND, AND ASK ───────────────────────────────────────────────────────────────────────────────────────
	option = {
		name = ln_return.1.d
		ai_chance = {
			base = 25
			modifier = { add = 20  has_trait = just }
			modifier = { add = 20  has_trait = honest }
			modifier = { add = 15  has_trait = humble }
		}
		add_character_flag = ln_return_asked_politely
		add_pressed_claim = scope:ln_old_seat
		add_prestige = 75
		add_piety = 100
		scope:ln_squatter = { add_opinion = { modifier = respect_opinion  target = root  opinion = 20 } }
		stress_impact = { base = minor_stress_impact_loss }
	}
}
