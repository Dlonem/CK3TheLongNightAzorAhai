namespace = ln_favored

# AGOT: The Long Night & Azor Ahai -- the Favored chain.
#
# You have been named, and it was not your idea. Favored is the rung where the claim is still deniable: nobody
# outside the temple has committed to it, and you can still put it down. Everything here is about whether you do.
#
# THE SHAPE, and it is different from the Resurrected chain on purpose. Resurrected is about what was taken from
# you and you have no say in it. Favored is about what people want FROM you, and every single event is somebody
# arriving with an ask. The recurring cost is that agreeing makes the claim more real and refusing makes you
# smaller in the eyes of people who came a long way.
#
# Driven by ln_favored_pulse_effect, one every four years at most, random order.

# ── 1. THE FIRST PILGRIM ────────────────────────────────────────────────────────────────────────────────────
ln_favored.1 = {
	type = character_event

	# STILL QUALIFIES? Thomas renounced the name in ln_favored.3 and then a queued event landed on him
	# anyway, because the pulse checks the trait when it SCHEDULES and these fire up to 250 days later.
	# An event that assumes a trait must re-check it at the moment it opens, or a man who put the name
	# down keeps getting asked about it for a year.
	trigger = {
		OR = {
			has_trait = favored_by_rhllor
			has_trait = chosen_by_rhllor
			has_trait = azor_ahai_reborn
		}
	}
	title = ln_favored.1.t
	desc = ln_favored.1.desc
	theme = faith
	left_portrait = { character = root  animation = worry }
	right_portrait = { character = scope:ln_pilgrim  animation = personality_zealous }

	# HE HAS TO EXIST TO BE LOOKED AT. Volantene by culture and R'hllor by faith because that is where he
	# walked from and why he came; `dynasty = none` because a man who walked for two years has no house left.
	immediate = {
		create_character = {
			location = root.capital_province
			culture = culture:volantene
			faith = faith:rhllor_fc
			gender = male
			age = { 24 55 }
			dynasty = none
			random_traits = yes
			trait = zealous
			save_scope_as = ln_pilgrim
		}
	}

	option = {
		name = ln_favored.1.a          # feed him. hear him. send him home.
		ai_chance = {
			base = 50
			modifier = { add = 20  has_trait = generous }
			modifier = { add = 15  has_trait = compassionate }
			modifier = { add = 10  has_trait = humble }
		}
		add_piety = 50
		add_prestige = 25
		stress_impact = { base = minor_stress_impact_gain  arrogant = minor_stress_impact_loss }
	}
	option = {
		name = ln_favored.1.b          # have the guard turn him around.
		ai_chance = {
			base = 25
			modifier = { add = 20  has_trait = callous }
			modifier = { add = 15  has_trait = paranoid }
			modifier = { add = 15  has_trait = cynical }
		}
		add_character_flag = ln_favored_turned_them_away
		add_piety = -50
		add_prestige = -25
		stress_impact = { base = minor_stress_impact_loss  callous = medium_stress_impact_loss }
	}
	option = {
		name = ln_favored.1.c          # keep him. a man who walked that far is worth having.
		ai_chance = {
			base = 35
			modifier = { add = 30  has_trait = zealous }
			modifier = { add = 10  has_trait = trusting }
			modifier = { add = 10  has_trait = gregarious }
		}
		add_character_flag = ln_favored_kept_the_pilgrim
		add_piety = 75
		add_diplomacy_lifestyle_xp = 40
	}

	# ── 1.0.13: THE SIGHTED OPTION ──────────────────────────────────────────────────────────────────────
	# AGOT ships `greensight` and `dragon_dreams` as real prophetic traits (00_agot_traits.txt) and this mod
	# had never read either. They are the two ways a person in this world can know a thing before being told
	# it, and the prophecy chain -- 81 options across three files, two of them gated before this release --
	# is exactly where that should change what you are able to say.
	#
	# THE GATE IS ITS OWN EXPLANATION. CK3 renders an option's trigger as that option's tooltip and draws the
	# trait icon inline, so the player is shown WHY this door is open to him and nobody else. Vanilla's own
	# dominant option idiom -- 899 trait-gated options in a 220-file sample of game/events/.
	option = {
		name = ln_favored.1.d
		trigger = {
			OR = {
				has_trait = dragon_dreams
				has_trait = greensight
			}
		}
		add_piety = 100
		add_character_flag = ln_favored_recognised_him
		stress_impact = { base = minor_stress_impact_loss }
		ai_chance = {
			base = 70
			modifier = { add = 20  has_trait = compassionate }
		}
	}

}
