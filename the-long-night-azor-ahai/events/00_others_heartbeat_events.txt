namespace = others_heartbeat

others_heartbeat.1 = {
	type = character_event
	hidden = yes

	trigger = {
		has_trait = other_trait
	}

	immediate = {
		save_scope_as = thegreatother

		# 13 Aug (1.0.6) -- SELF-HEAL for saves where the Night King already rolled a random
		# personality (creation now sets random_traits = no, but the mid-save policy says repair,
		# not just prevent). Craven is the one that actively contradicts him -- it drove the
		# "Analytic Craven" AI personality in Thomas's run. Removed once; the trigger keeps this
		# free on every later beat.
		if = {
			limit = {
				has_trait = nightking
				has_trait = craven
			}
			remove_trait = craven
		}

		# Mutate the Night King's direct courtiers if any are not mutated.
		# PERF FIX (v0.2.1, live-log find): these limits checked other_trait only -- but the effect
		# raises WIGHTS (wight_trait), so no converted character ever counted as done, and the whole
		# conversion re-ran on EVERY wight in the realm EVERY 15 days for the rest of the campaign.
		# Thousands of characters, forever -- the hottest loop in the mod and the error-log firehose.
		# Same behavior, one pass per character: done means other_trait OR wight_trait.
		#
		# ── 1.1.3 (W15.5): ONE WALK, NOT THREE. The 1.0.16 note at the bottom of this file said the two
		# realm walks (conversion here, dual-trait cleanup below) iterate the same list and could be merged.
		# They are. Each courtier and each vassal is visited once and asked three things in order: does he
		# need converting (the v0.2.1 limit, unchanged); is he wearing both traits (the LN+ cleanup, now
		# ln_strip_the_stray_wight_trait_effect in zz_ln_raising_effects.txt); and -- new in 1.1.3 -- is he
		# a dead non-ruler still wearing the black or the silent sisterhood, which is the freeze-loop trait
		# (ln_the_dead_keep_no_vows_effect, zz_ln_service_effects.txt). That third question is the mid-save
		# repair for the thousands of wighted brothers already standing in the dead's own halls on a running
		# save: the conversion body only strips at the raise, and this beat does NOT re-run it on existing
		# wights (the limit above is exactly what stops that), so this is where they get stripped, within
		# 30 days of updating. Cost: two trait tests per already-converted character, on a walk that was
		# already visiting him.
		every_courtier_or_guest = {
			if = {
				limit = { NOR = { has_trait = other_trait  has_trait = wight_trait } }
				mutate_into_other_effect = yes
			}
			else_if = {
				limit = { has_trait = other_trait  has_trait = wight_trait }
				ln_strip_the_stray_wight_trait_effect = yes
			}
			if = {
				limit = {
					is_ruler = no
					# 1.1.3 review: only the dead. The mutate just above leaves some living men standing (a
					# dawn_refugee courtier, a noble escape_noble_effect has just moved), and they keep their vows.
					ln_is_one_of_them_trigger = yes
					OR = {
						agot_is_member_of_nights_watch = yes
						has_trait = silent_sister
					}
				}
				ln_the_dead_keep_no_vows_effect = yes
			}
		}

		# Mutate vassals and their courtiers only if they are not already mutated. Restructured with
		# the same fix, plus one more: the old shape ran mutate on an ALREADY-CONVERTED vassal whenever
		# any of his courtiers still needed doing. Now the vassal only mutates if he himself needs it.
		# (1.1.3: the walk is unfiltered and asks per vassal instead -- the old limit visited every vassal
		# to evaluate itself anyway, and the merged questions need to reach the converted ones too.)
		every_vassal_or_below = {
			if = {
				limit = { NOR = { has_trait = other_trait  has_trait = wight_trait } }
				mutate_into_other_effect = yes
			}
			else_if = {
				limit = { has_trait = other_trait  has_trait = wight_trait }
				ln_strip_the_stray_wight_trait_effect = yes
			}
			every_courtier_or_guest = {
				if = {
					limit = { NOR = { has_trait = other_trait  has_trait = wight_trait } }
					mutate_into_other_effect = yes
				}
				else_if = {
					limit = { has_trait = other_trait  has_trait = wight_trait }
					ln_strip_the_stray_wight_trait_effect = yes
				}
				if = {
					limit = {
						is_ruler = no
						ln_is_one_of_them_trigger = yes
						OR = {
							agot_is_member_of_nights_watch = yes
							has_trait = silent_sister
						}
					}
					ln_the_dead_keep_no_vows_effect = yes
				}
			}
		}

		# ONE-TIME REPAIR for saves from before tonight's dragonglass fix: wights raised under the old
		# code kept their preparation modifiers, and the re-run that would have cleaned them is
		# (correctly) gone now. One sweep, once per save, then never again.
		# ...and heal the faith the broken is_head_of_faith guard skipped (see mutate): wights
		# converted while it was live kept their living faith. One sweep, then never again.
		if = {
			limit = { NOT = { has_global_variable = ln_wight_faith_repaired_once } }
			set_global_variable = { name = ln_wight_faith_repaired_once  value = yes }
			every_vassal_or_below = {
				limit = {
					has_trait = wight_trait
					NOT = { faith = faith:cold_gods }
					NOT = { any_held_title = { is_head_of_faith = yes } }
				}
				set_character_faith = faith:cold_gods
				set_culture = culture:frostborn
			}
		}

		if = {
			limit = { NOT = { has_global_variable = ln_wights_disarmed_once } }
			set_global_variable = { name = ln_wights_disarmed_once  value = yes }
			every_vassal_or_below = {
				limit = { has_trait = wight_trait }
				remove_character_modifier = ln_dragonglass_armed_modifier
				remove_character_modifier = ln_dragonglass_knight_modifier
				remove_character_modifier = ln_dragonglass_stores_modifier
				remove_character_modifier = ln_valyrian_armed_modifier
			}
		}
		
		# LN+ cleanup (dual other_trait + wight_trait): merged into the walks above in 1.1.3 -- see the
		# W15.5 note there. The body is ln_strip_the_stray_wight_trait_effect.

		# LN+ — keep the Night King leading his largest host from the FRONT. As a `commanding_army` holder he is
		# shielded from siege capture/death by vanilla's siege_0001_can_be_captured (which excludes any commander /
		# knight from the capture-or-kill pool), so he no longer dies anticlimactically to a siege — yet he can still
		# fall gloriously in open battle. Re-asserted on this 30-day pulse (15 before 1.0.16) if the AI ever leaves him without a command.
		if = {
			limit = {
				has_trait = nightking
				is_commanding_army = no
				any_army = { army_owner = root }
			}
			ordered_army = {
				limit = { army_owner = root }
				order_by = army_size
				position = 0
				assign_commander = root
			}
		}

		# -- 1.0.8: THE DEAD DO NOT TIRE ------------------------------------------------------------
		# WHY THIS EXISTS -- the White Harbor stall, which was never about White Harbor.
		#
		# CK3 scores a siege in DEFENDER MORALE, not in bodies. A holding needs 100 + 75 x fort_level
		# progress (NSiege.BASE_DEFENDER_MORALE / DEFENDER_MORALE_PER_FORT_LEVEL, 00_defines.txt), the
		# daily drain is base 1 + men-at-arms siege_value + 0.01 per 200 men over the garrison, and the
		# whole thing is then multiplied by 0.7 for EACH siege tier the besiegers are short of the fort's
		# threshold (SIEGE_FORT_THRESHOLDS = { 4 6 11 16 31 }), floored at MIN_DAILY_MORALE_LOSS = 0.5.
		#
		# The dead cannot reach the only term that scales. NCombat.LEVY_SIEGE = 0 -- a levy is worth
		# NOTHING to a siege -- and our own live-run notes record the Others fielding 429,171 men of which
		# 392,200 (91%) was accumulated scouring levies. Worse, a levy-only stack is siege tier 0, so
		# against White Harbor's b_new_castle (fort 25, read out of the 13 Aug save; 28 under our own
		# ln_last_bastion_modifier) it also eats 0.7^4 = x0.24. Two thousand two hundred points of progress
		# at half a point a day IS the five-year siege -- and it is universal, not local. Fort 6 is 550
		# points; fort 11 is 925. Every castle in Westeros does this to the dead. White Harbor is only
		# where it became impossible to miss.
		#
		# THE FIX IS THE LEVIES, because the levies are the disease. See zz_ln_siege_modifiers.txt for
		# which of the three values is additive, which two are percentages, and the retuning arithmetic.
		#
		# NOTE ON ROOT -- this event fires for EVERY other_trait character, not only the Night King (see
		# the trigger at the top of this file; the loop is started from 00_others_on_actions.txt on
		# on_title_gain). That is load-bearing here: longnight_create_other_for_scouring builds the
		# scouring Others as INDEPENDENT, LANDLESS dukes with no liege, and longnight_spawn_army_for_
		# scouring spawns their hosts inside THEIR scope -- so those armies are not the Night King's, and
		# a sweep rooted on him would never have seen the hosts that were actually stuck. Each Other
		# services his own.

		# The captain goes on FIRST, so a host that gains one this pulse is buffed in the same pulse.
		#
		# The owner leads his own host. On a scouring Other that is the entire answer: he has no liege, no
		# vassals and no court to draw a captain from, and he is standing idle with brave, logistician and
		# a martial education. Deliberately NOT a vassal or courtier search -- trawling the realm for an
		# idle body can pull a wight lord out of a field host where he is serving as a KNIGHT, and quietly
		# changing the outcome of field battles is exactly what this fix must not do. Rooting it on the
		# owner also means no shared saved scope, so there is no way to hand one character two commands.
		# ONLY IF: he is leading nothing already, the host is his, it is ACTUALLY besieging, and it has
		# nobody at its head.
		# NEVER IF: any of those is false -- then nothing happens at all, and the host simply waits.
		if = {
			limit = { is_commanding_army = no }
			ordered_army = {
				limit = {
					army_owner = root
					is_army_in_siege = yes
					NOT = { exists = army_commander }
				}
				order_by = army_size
				position = 0
				assign_commander = root
			}
		}

		# ONLY IF: the character is leading one of this Other's hosts.
		# NEVER IF: they are not -- siege_morale_loss is a COMMAND modifier and is worth nothing off the
		# front. AGOT's dragon code says it outright ("If I am not a commander, I get no buffs") and strips
		# it the moment the rider stops commanding. The 30-day duration is our removal mechanism: against a
		# 30-day pulse (since 1.0.16) it lapses on its own if the beat ever stops. Remove-then-add cannot
		# stack, and removing an absent modifier is a silent no-op.
		every_army = {
			limit = { army_owner = root }
			army_commander ?= {
				remove_character_modifier = ln_the_dead_do_not_tire_modifier
				add_character_modifier = { modifier = ln_the_dead_do_not_tire_modifier  days = 30 }

				# THE MEASUREMENT (needs -develop). Whether the AI's siege split leaves pure levies on the
				# walls or keeps some of the host's tier-4 other_maa swings the siege length by an order of
				# magnitude, and it cannot be read out of the files. When this line appears, open that siege
				# and divide its progress by its days: that rate against the holding's fort level is the one
				# number levy_siege should be tuned from.
				if = {
					limit = { commanding_army ?= { is_army_in_siege = yes } }
					debug_log = "LN_TRACE: undead siege active - commander buffed. Measure daily siege progress against fort level here."
				}
			}
		}

		# Prevent overlapping loops
		# ── 1.0.16 PERFORMANCE: 15 DAYS -> 30. ─────────────────────────────────────────────────────────────
		# This event self-perpetuates PER other_trait CHARACTER and contains two full every_vassal_or_below
		# walks, the first with a nested any_courtier_or_guest inside its limit -- so its cost is
		# O(vassals x courtiers) per Other, and it SCALES WITH CONQUEST, i.e. with exactly the thing that
		# grows over a Long Night. At a few hundred counties that is thousands of character visits per
		# fortnight per Other.
		# Doubling the interval halves both walks at no functional cost: the only time-sensitive thing this
		# event does is refresh a command modifier that is itself applied with days = 30 (see :200), so a
		# 30-day beat matches what the work actually needs. Conversion latency goes from ~a fortnight to
		# ~a month, which is invisible against a Long Night measured in decades.
		# Corvo reported his machine burning past 300k men; this is the CPU-side half of that answer, and
		# unlike the headcount retune it costs nothing at all.
		#
		# 1.1.3: the two walks ARE merged now (W15.5, see the conversion block above); this beat is one
		# realm walk and one court walk, not two of each. And for the record: this event fires every 30
		# days, not 15 -- the "~15-day" wording elsewhere in this file and in the session notes is from
		# before the 1.0.16 change three lines up.
		if = {
			limit = { NOT = { has_character_flag = others_heartbeat_active } }
			add_character_flag = { flag = others_heartbeat_active days = 29 }
			trigger_event = {
				id = others_heartbeat.1
				days = 30
			}
		}
	}
}
