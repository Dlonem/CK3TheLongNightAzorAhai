# Deliberately has no options beyond closing it and grants nothing.

	type = character_event
	title = ln_prophecy.10.t
	desc = {
		desc = ln_prophecy.10.desc.intro
		first_valid = {
			triggered_desc = { trigger = { has_trait = chosen_by_rhllor }   desc = ln_prophecy.10.desc.is_chosen }
			triggered_desc = { trigger = { has_trait = favored_by_rhllor }  desc = ln_prophecy.10.desc.is_favored }
			triggered_desc = { trigger = { has_character_flag = ln_refused_the_name } desc = ln_prophecy.10.desc.refused }
			desc = ln_prophecy.10.desc.unnamed
		}
		triggered_desc = { trigger = { ln_has_deed_dragon_dreams_trigger = yes }  desc = ln_prophecy.10.deed.dreams }
		triggered_desc = { trigger = { ln_has_deed_greensight_trigger = yes }     desc = ln_prophecy.10.deed.green }
		triggered_desc = { trigger = { ln_has_deed_salt_and_smoke_trigger = yes } desc = ln_prophecy.10.deed.salt }
		triggered_desc = { trigger = { ln_has_deed_bleeding_star_trigger = yes }  desc = ln_prophecy.10.deed.star }
		triggered_desc = { trigger = { ln_has_deed_survived_trigger = yes }       desc = ln_prophecy.10.deed.survived }

		# ── WHAT YOU DID ABOUT IT ───────────────────────────────────────────────────────────────────────────
		# These five were write-only. Every one of them was set by an event option and read by nothing, which

		triggered_desc = { trigger = { has_character_flag = ln_kept_the_name_anyway }   desc = ln_prophecy.10.did.kept }

		triggered_desc = { trigger = { has_character_flag = ln_burned_the_red_priest }  desc = ln_prophecy.10.did.burned }
		triggered_desc = { trigger = { has_character_flag = ln_wishes_it_had_not_worked } desc = ln_prophecy.10.did.wishes }
		triggered_desc = { trigger = { has_character_flag = ln_ended_the_long_night }   desc = ln_prophecy.10.did.ended }

		# and whether anyone is in a position to say it out loud
		first_valid = {
			triggered_desc = { trigger = { ln_has_red_priest_trigger = yes } desc = ln_prophecy.10.desc.has_priest }

			desc = ln_prophecy.10.desc.no_seer
		}
	}
