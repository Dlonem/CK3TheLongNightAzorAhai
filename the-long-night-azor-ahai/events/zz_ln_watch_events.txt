# ── 1. THE CASTLE THEY GAVE YOU ─────────────────────────────────────────────────────────────────────────────
ln_watch.1 = {
	type = character_event
	title = ln_watch.1.t
	desc = ln_watch.1.desc
	theme = realm
	left_portrait = { character = root  animation = thinking }
	right_portrait = { trigger = { exists = scope:ln_lord_commander }   character = scope:ln_lord_commander  animation = personality_honorable }

	trigger = {
		has_government = nights_watch_government
		exists = title:k_the_wall.holder
	}

	# THE PORTRAIT NEEDS THIS AND THE DECISION IS NOT THE ONLY WAY IN. scope:ln_lord_commander is saved right
	# here rather than inherited (the effect that used to set it, ln_take_the_black_effect, is now
	# ln_lay_down_the_realm_effect), and scopes do survive trigger_event -- but a save that reloads mid-chain, or any
	# future caller, would render a portrait pointing at nothing. immediate runs before the window draws, so
	# re-establishing it here is free and removes a whole class of missing-scope errors.
	immediate = {
		title:k_the_wall.holder ?= { save_scope_as = ln_lord_commander }
	}

	option = {
		name = ln_watch.1.a          # start with the walls.
		ai_chance = {
			base = 45
			modifier = { add = 15  has_trait = diligent }
			modifier = { add = 10  has_trait = paranoid }
		}
		add_character_flag = ln_watch_walls_first
		add_martial_lifestyle_xp = 60
		add_prestige = 25
	}
	option = {
		name = ln_watch.1.b          # start with the kitchens. men who eat, stay.
		ai_chance = {
			base = 35
			modifier = { add = 20  has_trait = gluttonous }
			modifier = { add = 10  has_trait = generous }
			modifier = { add = 10  has_trait = gregarious }
		}
		add_character_flag = ln_watch_kitchens_first
		add_stewardship_lifestyle_xp = 60
		add_gold = 50
	}
	option = {
		name = ln_watch.1.c          # start with the library. somebody wrote this down once.
		ai_chance = {
			base = 20
			modifier = { add = 15  has_trait = patient }
			modifier = { add = 10  has_trait = shy }
		}
		add_character_flag = ln_watch_library_first
		add_learning_lifestyle_xp = 80
	}
}

# a more interesting card than the Lord Commander counting turnips.

# ── 9. THE RIDER CAME BACK ──────────────────────────────────────────────────────────────────────────────────
# Fires a week after ln_take_the_black_and_wait_decision, by which time AGOT's own revoked handler has returned
# the castle to the Lord Commander and ln_grant_a_keepership_effect can find it.
ln_watch.9 = {
	type = character_event
	title = ln_watch.9.t
	desc = ln_watch.9.desc
	theme = realm
	left_portrait = { character = root  animation = thinking }
	right_portrait = { trigger = { exists = scope:ln_relieved_brother }   character = scope:ln_relieved_brother  animation = grief }

	trigger = {
		is_alive = yes
		has_trait = nightswatch
		exists = title:k_the_wall.holder
	}

	immediate = {
		# THE WEEK MAY NOT HAVE BEEN ENOUGH -- a revoke can be refused, the man can die, AGOT's handler can
		# take a different route. If no castle came free, say so honestly rather than handing him one out of
		# nowhere; ln_watch.10 is the version of this card where nothing happened.
		if = {
			limit = { ln_wall_has_a_free_keepership_trigger = yes }
			ln_grant_a_keepership_effect = yes
			add_character_flag = ln_took_a_castle_from_a_brother
		}
	}

	option = {
		name = ln_watch.9.a          # he is not to be shamed for it.
		ai_chance = {
			base = 45
			modifier = { add = 30  has_trait = compassionate }
			modifier = { add = 15  has_trait = generous }
			modifier = { add = 10  has_trait = just }
		}
		trigger = { ln_carries_a_keepership_trigger = yes }
		# 1.1.0 (§7): a placeless brother sees what a keepership would let him do.
		show_as_unavailable = { always = yes }
		add_prestige = 50
		# He asked for one thing. You gave it to him, and the Wall noticed that too.
		if = {
			limit = { exists = scope:ln_relieved_brother  scope:ln_relieved_brother = { is_alive = yes } }
			scope:ln_relieved_brother = {
				add_opinion = { modifier = ln_let_me_keep_my_dignity_opinion  target = root  opinion = 40 }
			}
			# 1.0.16 -- `?=`. This option's own trigger is ln_carries_a_keepership_trigger (is_landed +
			# nights_watch_government), which says nothing about whether k_the_wall has a holder -- and it
			# can stand unheld, which is the whole reason ln_lc.10's options carry the same guard and why
			# zz_ln_relieve_effects.txt:34 already uses `?=` for this exact title. Card lands 7 days out, so
			# the window is narrow, but a narrow window is still a window.
			title:k_the_wall.holder ?= {
				every_vassal_or_below = {
					limit = {
						is_alive = yes
						government_has_flag = government_is_nw
						NOT = { this = root }
					}
					# Half the damage from ln_relieve_a_brother_effect comes back. The Wall is small and word
					# travels; they were always going to hear how you did it, not just that you did it.
					add_opinion = { modifier = ln_saw_what_you_did_to_him_opinion  target = root  opinion = 6 }
				}
			}
		}
	}
	option = {
		name = ln_watch.9.b          # he had four years to send a report.
		ai_chance = {
			base = 25
			modifier = { add = 15  has_trait = sadistic }
			modifier = { add = 15  has_trait = callous }
			modifier = { add = 10  has_trait = arbitrary }
		}
		trigger = { ln_carries_a_keepership_trigger = yes }
		# 1.1.0 (§7): as above.
		show_as_unavailable = { always = yes }
		add_dread = 10
		add_prestige = 25
		if = {
			limit = { exists = scope:ln_relieved_brother  scope:ln_relieved_brother = { is_alive = yes } }
			scope:ln_relieved_brother = {
				add_opinion = { modifier = ln_took_my_command_opinion  target = root  opinion = -50 }
			}
		}
	}
	option = {
		name = ln_watch.9.c          # nothing came free. I wait.
		ai_chance = {
			base = 30
			modifier = { add = 25  has_trait = patient }
			modifier = { add = 15  has_trait = humble }
			modifier = { add = 10  has_trait = content }
		}
		# 1.1.0 (§7): no show_as_unavailable here, deliberately -- mirror of 9.a/9.b (keepership = no); one branch always shows.
		trigger = { NOT = { ln_carries_a_keepership_trigger = yes } }
		add_character_flag = ln_still_waiting_for_a_castle
		stress_impact = { base = minor_stress_impact_gain }
	}
}


# ══════════════════════════════════════════════════════════════════════════════════════════════════════════════
# HOW MUCH. The one decision in this mod that is purely a statement of belief, so the player says how loudly.
#
# THREE TIERS, and they are not linear: the Watch remembers a cart of grain, it remembers a season's stores
# differently, and it never forgets a man who emptied his vaults for a wall he will never stand on. The ledger
# (ln_realm_fed_the_watch, read by ln_petition_yield_effect at >= 3 and >= 8) moves by 1, 2 and 3 -- so a
# believer reaches the Watch's gratitude in one gesture where a cautious realm takes a century of carts.
#
# THE AI IS DELIBERATELY STINGY. Thomas: "there should be some level of preventing AI from being able to spam
# that to the detriment of their own realm." The big tiers are weighted to zero unless the AI is genuinely rich
# -- 2,000 spare for the middle gift, 6,000 for the vaults -- so an AI king with money to burn can make the
# grand gesture and nobody else beggars themselves doing it.
# ══════════════════════════════════════════════════════════════════════════════════════════════════════════════
ln_watch.10 = {
	type = character_event
	title = ln_watch.10.t
	desc = ln_watch.10.desc
	theme = diplomacy
	left_portrait = { character = root  animation = thinking }
	right_portrait = { trigger = { exists = scope:ln_lord_commander }   character = scope:ln_lord_commander  animation = personality_honorable }

	# 1.0.12 -- SAME REPAIR AS ln_watch.8, FOR THE OPPOSITE REASON. This card has exactly one door today
	# (zz_ln_donate_to_the_watch_decisions.txt:51 saves the scope on the line before it fires this), so it works
	# -- and it works only because of that one line, in a file that is not this one. Any second caller renders a
	# portrait pointing at nothing. Two lines here and the card is correct on its own terms.
	immediate = {
		title:k_the_wall.holder ?= { save_scope_as = ln_lord_commander }
	}

	# A CART OF GRAIN
	option = {
		name = ln_watch.10.a
		trigger = { gold >= 100 }
		# 1.1.0 (§7): the gold ladder shown, all three rungs.
		show_as_unavailable = { always = yes }
		remove_short_term_gold = 100
		ln_feed_the_wall_effect = { AMOUNT = 100  LEDGER = 1  OPINION = 25  PIETY = 50  PRESTIGE = 25 }
		ai_chance = {
			base = 60
			modifier = { add = 15  has_trait = humble }
			modifier = { add = 10  has_trait = diligent }
		}
	}

	# A SEASON'S STORES
	option = {
		name = ln_watch.10.b
		trigger = { gold >= 500 }
		# 1.1.0 (§7): ladder rung two.
		show_as_unavailable = { always = yes }
		remove_short_term_gold = 500
		ln_feed_the_wall_effect = { AMOUNT = 500  LEDGER = 2  OPINION = 50  PIETY = 120  PRESTIGE = 60 }
		ai_chance = {
			base = 0
			modifier = { add = 40  gold >= 2000 }
			modifier = { add = 20  has_trait = generous }
			modifier = { add = 15  has_character_flag = ln_looked_north }
		}
	}

	# EMPTY THE VAULTS
	option = {
		name = ln_watch.10.c
		trigger = { gold >= 1500 }
		# 1.1.0 (§7): ladder rung three.
		show_as_unavailable = { always = yes }
		remove_short_term_gold = 1500
		ln_feed_the_wall_effect = { AMOUNT = 1500  LEDGER = 3  OPINION = 75  PIETY = 250  PRESTIGE = 120 }
		add_character_flag = ln_emptied_the_vaults
		ai_chance = {
			base = 0
			modifier = { add = 30  gold >= 6000 }
			modifier = { add = 25  has_trait = zealous }
			modifier = { add = 20  ln_is_a_champion_of_rhllor_trigger = yes }
		}
	}

	# NOT THIS YEAR
	option = {
		name = ln_watch.10.d
		ai_chance = { base = 20 }
	}
}
