namespace = ln_prophecy

# The dragonglass arrives.
#
# WAS ln_prophecy.30, AND SO IS THE SEER'S CARD IN zz_ln_entry_events.txt. Two events, one ID. CK3 keeps the
# last one parsed and drops the other without a word, and "zz_ln_preparation" sorts after "zz_ln_entry", so
# this card was quietly eating the one that tells a man what people have started calling him. Renumbered here
# rather than there because this one has a single caller and three localization keys.
# DEVELOPED 9 Aug (Session 10). This is the payoff card of the mod's core decision and it was a single
# paragraph with a nod. Branched: the Watch has no master-at-arms to convince, only a vow that finally makes
# sense; a man who has SEEN one (believed a sighting, or carries the greensight) is not arguing with the
# Citadel, he is arguing with his own memory; everyone else keeps the original scene.
ln_prophecy.31 = {
	type = character_event
	title = ln_prophecy.31.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { has_government = nights_watch_government }
				desc = ln_prophecy.31.desc.watch
			}
			triggered_desc = {
				trigger = {
					OR = {
						has_character_flag = ln_named_it_first
						ln_has_deed_greensight_trigger = yes
					}
				}
				desc = ln_prophecy.31.desc.seen
			}
			desc = ln_prophecy.31.desc
		}
	}
	theme = realm
	override_background = { reference = agot_frozen_forest }

	# Arm everyone, and let the realm laugh until it stops laughing.
	option = {
		name = ln_prophecy.31.a
		ai_chance = {
			base = 60
			modifier = { add = 20  OR = { has_trait = brave  has_trait = zealous  has_government = nights_watch_government } }
		}
	}

	# ── 1.1.0 (§11, rung two): CUT IT PROPERLY. Only a man who studied the glass knows the facets run one
	# way and not the other; only a man with a weapon in hand has an edge to set it into. The modifier
	# rides the ARTIFACT: it travels when the sword is inherited or looted, by design. Greyed to everyone
	# else -- the whole ladder advertised on the card that ends it (§7's idiom).
	option = {
		name = ln_prophecy.31.c
		# ⚠ primary_weapon is NOT a script link in either context -- the boot log's "Unknown trigger:
		# primary_weapon" derailed this whole file. The engine's real surface is the equipped-artifact
		# iterators: any_/every_equipped_character_artifact with artifact_slot_type = primary_armament
		# (vanilla 00_artifact_triggers.txt:526, tournament_actions.txt). A man with no weapon matches
		# nothing on both sides -- a no-op, never a wipe (standing rule 4).
		# ⚠ 1 Sep, Thomas on the live card: "it said that I was missing a modifier." That is CK3's raw
		# fallback line for a bare has_character_modifier gate, and it tells a player nothing about what to
		# go and do. Both clauses are now wrapped so the greyed option says, in words, what the door wants:
		# the vanilla idiom is custom_tooltip = { text = X <triggers> } (00_available_for_events_triggers.txt:22).
		trigger = {
			custom_tooltip = {
				text = ln_knows_the_glass_gate_tt
				has_character_modifier = ln_knows_the_glass_modifier
			}
			custom_tooltip = {
				text = ln_needs_an_unedged_weapon_tt
				any_equipped_character_artifact = {
					artifact_slot_type = primary_armament
					NOT = { has_artifact_modifier = ln_artifact_dragonglass_edged_modifier }
				}
			}
		}
		show_as_unavailable = { always = yes }
		every_equipped_character_artifact = {
			limit = {
				artifact_slot_type = primary_armament
				NOT = { has_artifact_modifier = ln_artifact_dragonglass_edged_modifier }
			}
			add_artifact_modifier = ln_artifact_dragonglass_edged_modifier
		}
		ai_chance = {
			base = 100
		}
	}

	# Arm them quietly. A lord who hands out glass knives with no explanation is a lord people watch.
	option = {
		name = ln_prophecy.31.b
		add_dread = 5
		ai_chance = {
			base = 20
			modifier = { add = 25  OR = { has_trait = paranoid  has_trait = deceitful } }
			modifier = { add = -20  has_government = nights_watch_government }
		}
	}
}
