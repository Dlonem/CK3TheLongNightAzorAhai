namespace = ln_chosen

# AGOT: The Long Night & Azor Ahai -- the Chosen chain.
#
# THE DIFFERENCE FROM FAVORED, and it is the whole design: Favored is people ASKING. Chosen is people ACTING.
# Nobody in this chain requests anything. A king has already decided what you are for. A temple has already
# spent your name. Somebody has already been sent to kill you. By the time you read the event the thing has
# happened and your only choice is what to do about it.
#
# Which is also why there is no "renounce it" option anywhere below. That door was on the Favored rung and it
# closed. The traits are a ladder and this is what the next rung feels like.

# ── 1. A KING WANTS YOU SWORN ───────────────────────────────────────────────────────────────────────────────
ln_chosen.1 = {
	type = character_event

	# STILL QUALIFIES? Thomas renounced the name in ln_favored.3 and then a queued event landed on him
	# anyway, because the pulse checks the trait when it SCHEDULES and these fire up to 250 days later.
	# An event that assumes a trait must re-check it at the moment it opens, or a man who put the name
	# down keeps getting asked about it for a year.
	trigger = {
		OR = {
			has_trait = chosen_by_rhllor
			has_trait = azor_ahai_reborn
		}
	}
	title = ln_chosen.1.t
	desc = ln_chosen.1.desc
	theme = crown

	# 10 Aug -- THE KING WAS NEVER IN THE ROOM. Thomas: "I feel like we need to hard code to show, like, the
	# king actually showing up in that event. I thought this event was more fleshed out." He was right: the
	# card was a monologue with one portrait, about an offer from a man who never appeared.
	#
	# The crown on the right, his envoy below him -- because the text is about the envoy doing the talking
	# while everyone in the room understands who is speaking through him.
	immediate = {
		hidden_effect = {
			# A REAL CROWN IF ONE EXISTS, and in Westeros one always does. Weighted to the Iron Throne
			# first (an empire-tier neighbour is the crown this card is really about), then to anyone
			# whose realm this man already lives inside.
			random_ruler = {
				limit = {
					is_alive = yes
					is_landed = yes
					is_ai = yes
					highest_held_title_tier >= tier_kingdom
					NOT = { this = root }
					NOT = { has_title = title:k_the_wall }
					ln_is_one_of_them_trigger = no
				}
				weight = {
					base = 1
					modifier = { add = 40  highest_held_title_tier >= tier_empire }
					modifier = { add = 25  this = root.top_liege }
					modifier = { add = 15  any_sub_realm_county = { count >= 40 } }
				}
				save_scope_as = ln_chosen_king
			}

			create_character = {
				location = root.capital_province
				culture = root.capital_county.culture
				faith = root.capital_county.faith
				gender = male
				age = { 30 60 }
				dynasty = none
				random_traits = yes
				save_scope_as = ln_chosen_envoy
			}

			# LAST RESORT ONLY -- a world with no crowns left in it, which is a state the endgame can
			# actually reach. A portrait block cannot be conditional, so the scope must always exist.
			if = {
				limit = { NOT = { exists = scope:ln_chosen_king } }
				scope:ln_chosen_envoy = { save_scope_as = ln_chosen_king }
			}
		}
	}

	left_portrait = { character = root  animation = thinking }
	right_portrait = { trigger = { exists = scope:ln_chosen_king }   character = scope:ln_chosen_king  animation = personality_bold }
	lower_right_portrait = { trigger = { exists = scope:ln_chosen_envoy }   character = scope:ln_chosen_envoy  animation = personality_dishonorable }

	option = {
		name = ln_chosen.1.a          # swear. a banner is a banner.
		ai_chance = {
			base = 30
			modifier = { add = 20  has_trait = content }
			modifier = { add = 15  has_trait = craven }
			modifier = { add = 15  has_trait = humble }
		}
		add_prestige = 150
		add_character_flag = ln_chosen_swore_to_a_king
		add_piety = -50
	}
	option = {
		name = ln_chosen.1.b          # I will fight beside him. I will not fight for him.
		ai_chance = {
			base = 45
			modifier = { add = 15  has_trait = brave }
			modifier = { add = 10  has_trait = arrogant }
			modifier = { add = 10  has_trait = just }
		}
		add_character_flag = ln_chosen_kept_the_distance
		add_piety = 100
		add_prestige = -50
		stress_impact = { base = minor_stress_impact_gain }
	}
	option = {
		name = ln_chosen.1.c          # kings are a southern problem.
		ai_chance = {
			base = 25
			modifier = { add = 20  has_trait = arrogant }
			modifier = { add = 15  has_trait = ambitious }
			modifier = { add = 10  has_trait = vengeful }
		}
		add_character_flag = ln_chosen_refused_a_king
		add_piety = 150
		add_prestige = -150
		add_dread = 10
	}
}

# ── 2. THE ENVOY FROM THE OTHER ONE ─────────────────────────────────────────────────────────────────────────
ln_chosen.2 = {
	type = character_event

	# STILL QUALIFIES? Thomas renounced the name in ln_favored.3 and then a queued event landed on him
	# anyway, because the pulse checks the trait when it SCHEDULES and these fire up to 250 days later.
	# An event that assumes a trait must re-check it at the moment it opens, or a man who put the name
	# down keeps getting asked about it for a year.
	trigger = {
		OR = {
			has_trait = chosen_by_rhllor
			has_trait = azor_ahai_reborn
		}
	}
	title = ln_chosen.2.t
	desc = ln_chosen.2.desc
	theme = diplomacy
	left_portrait = { character = root  animation = interested }

	# THE RIVAL CAN BE REAL. ln_chosen_list holds every living Chosen; if there is another one, he is
	# the man behind this envoy, and choosing to meet him should DO something (Thomas: "Can Chosen
	# become allies together?"). Scoped here so the option can read him; absent, the event stays the
	# abstract-rival card it always was.
	immediate = {
		random_in_global_list = {
			variable = ln_chosen_list
			limit = {
				NOT = { this = root }
				is_alive = yes
			}
			save_scope_as = ln_rival_chosen
		}
	}

	option = {
		name = ln_chosen.2.a          # meet him. one of us is wrong and it may be me.
		ai_chance = {
			base = 45
			modifier = { add = 15  has_trait = humble }
			modifier = { add = 15  has_trait = trusting }
			modifier = { add = 10  has_trait = brave }
			modifier = { add = 10  has_trait = compassionate }
		}
		add_character_flag = ln_chosen_met_the_rival
		add_diplomacy_lifestyle_xp = 80
		add_piety = 50
		# Two men the temples cannot tell apart, in one room, deciding not to be each other's problem.
		# Friends always; a full military alliance when both hold land and are not already at war.
		if = {
			limit = { exists = scope:ln_rival_chosen }
			set_relation_friend = { reason = ln_friend_two_chosen  target = scope:ln_rival_chosen }
			if = {
				limit = {
					is_ruler = yes
					scope:ln_rival_chosen = { is_ruler = yes }
					NOT = { is_at_war_with = scope:ln_rival_chosen }
				}
				create_alliance = {
					target = scope:ln_rival_chosen
					allied_through_owner = root
					allied_through_target = scope:ln_rival_chosen
				}
			}
		}
		stress_impact = { base = minor_stress_impact_gain  arrogant = medium_stress_impact_gain }
	}
	option = {
		name = ln_chosen.2.b          # send the envoy back with nothing.
		ai_chance = {
			base = 25
			modifier = { add = 20  has_trait = arrogant }
			modifier = { add = 15  has_trait = paranoid }
		}
		add_prestige = 50
		add_character_flag = ln_chosen_snubbed_the_rival
	}
	option = {
		name = ln_chosen.2.c          # send him back with a name. mine, and where I will be.
		ai_chance = {
			base = 30
			modifier = { add = 25  has_trait = ambitious }
			modifier = { add = 20  has_trait = arrogant }
			modifier = { add = 15  has_trait = wrathful }
		}
		add_character_flag = ln_chosen_challenged_the_rival
		add_dread = 15
		add_prestige = 100
		stress_impact = { base = medium_stress_impact_gain  wrathful = medium_stress_impact_loss }
	}
}

	desc = ln_chosen.5.desc
	theme = murder_scheme
	left_portrait = { character = root  animation = paranoia }
	right_portrait = { trigger = { exists = scope:ln_knifeman }   character = scope:ln_knifeman  animation = shame }

	# The servant with the broken arm and the very strange calm. Local culture and faith -- he got four months
	# into the household on forged references, so he had to pass for one of yours.
	immediate = {
		create_character = {
			location = root.capital_province
			culture = root.capital_county.culture
			faith = root.capital_county.faith
			gender = male
			age = { 19 40 }
			dynasty = none
			random_traits = yes
			save_scope_as = ln_knifeman
		}
		# BACKWARDS UNTIL 10 AUG. `imprison` is executed BY the gaoler ON the target, and this had the
		# knifeman imprisoning the lord he had just failed to kill -- which errored out every time
		# ("Can't imprison someone if you're not a ruler", since a forged-reference servant holds no land)
		# and left the assassin standing free in the hall while the card asked what to do with him.
		# Guarded as well as reversed: a Chosen who holds nothing has nowhere to put him.
		if = {
			limit = { is_ruler = yes }
			imprison = { target = scope:ln_knifeman  type = dungeon }
		}
	}

	option = {
		name = ln_chosen.5.a          # let it be known he failed. that is worth more than he is.

			base = 35
			modifier = { add = 20  has_trait = arrogant }

		add_character_flag = ln_chosen_survived_a_knife
		add_prestige = 150
		add_dread = 15
		add_intrigue_lifestyle_xp = 40
	}
	option = {
		name = ln_chosen.5.b          # find out who sent him, quietly.
		ai_chance = {
			base = 45
			modifier = { add = 30  has_trait = paranoid }
			modifier = { add = 15  has_trait = deceitful }
			modifier = { add = 15  has_trait = patient }
		}
		add_intrigue_lifestyle_xp = 90
		add_character_flag = ln_chosen_hunted_the_hand
		stress_impact = { base = minor_stress_impact_gain  paranoid = medium_stress_impact_gain }

	option = {
