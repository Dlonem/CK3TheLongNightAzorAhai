# AGOT: The Long Night & Azor Ahai -- what it costs to come back.
#
# Thomas has asked for this chain since the beginning: "I don't want just ten events for resurrected, favored
# and chosen. I want ten for each." This is the Resurrected ten, and it went first because resurrected_by_rhllor
# had NO content at all -- the trait has existed in this mod's lineage since Freeholder and nothing in the game
# could grant it until this session, so nothing was ever written for it either.
#
# THE RULE FOR THIS WHOLE CHAIN, and it is Thomas's: "just because it costs you something doesn't mean it has to
# have good outcomes only." Every event below has at least one option that is clearly the wise one and still
# takes something off you, and at least one that is obviously wrong and pays. Beric Dondarrion came back six
# times and each time there was less of him. That is the arc.
#
# Fired from ln_risen_pulse_effect, one every three years at most, in a random order that is different every
# campaign. Each has a one-shot flag so nothing repeats.

# ── 1. WHAT YOU REMEMBER ────────────────────────────────────────────────────────────────────────────────────
ln_risen.1 = {
	type = character_event

	# STILL QUALIFIES? Thomas renounced the name in ln_favored.3 and then a queued event landed on him
	# anyway, because the pulse checks the trait when it SCHEDULES and these fire up to 250 days later.
	# An event that assumes a trait must re-check it at the moment it opens, or a man who put the name
	# down keeps getting asked about it for a year.
	trigger = {
		has_trait = resurrected_by_rhllor
	}
	title = ln_risen.1.t
	desc = ln_risen.1.desc
	theme = death
	left_portrait = { character = root  animation = thinking }

	option = {
		name = ln_risen.1.a          # nothing. there was nothing.
		ai_chance = {
			base = 55
			modifier = { add = 15  has_trait = calm }
			modifier = { add = 10  has_trait = deceitful }
		}
		add_character_flag = ln_risen_saw_nothing
		stress_impact = { base = medium_stress_impact_gain  cynical = minor_stress_impact_loss }
		add_learning_lifestyle_xp = 40
	}
	option = {
		name = ln_risen.1.b          # something. and you will not say what.
		ai_chance = {
			base = 45
			modifier = { add = 20  has_trait = honest }
			modifier = { add = 10  has_trait = paranoid }
		}
		add_character_flag = ln_risen_saw_something
		add_piety = 100
		stress_impact = { base = minor_stress_impact_gain  zealous = medium_stress_impact_loss }
	}

	# ── 1.0.13: THE SIGHTED OPTION ──────────────────────────────────────────────────────────────────────
	# AGOT ships `greensight` and `dragon_dreams` as real prophetic traits (00_agot_traits.txt) and this mod
	# had never read either. They are the two ways a person in this world can know a thing before being told
	# it, and the prophecy chain -- 81 options across three files, two of them gated before this release --
	# is exactly where that should change what you are able to say.
	#
	# THE GATE IS ITS OWN EXPLANATION. CK3 renders an option's trigger as that option's tooltip and draws the
	# trait icon inline, so the player is shown WHY this door is open to him and nobody else. Vanilla's own
	# dominant option idiom -- 899 trait-gated options in a 220-file sample of game/events/.
	option = {
		name = ln_risen.1.d
		trigger = { has_trait = greensight }
		# 1.1.0 (§7): the gift advertised, as in zz_ln_chosen_events.
		show_as_unavailable = { always = yes }
		add_piety = 150
		add_character_flag = ln_risen_named_the_corridor
		stress_impact = { base = medium_stress_impact_loss }
		ai_chance = {
			base = 80
			modifier = { add = 20  has_trait = honest }
		}
	}

}

# ── 12. THE FIRE THAT STAYED ────────────────────────────────────────────────────────────────────────────────
# Not guaranteed (35% per return, rolled in ln_the_fire_takes_something_effect), and not a choice:
# the fire decides, the man finds out. Strength arrives, and the part of him that flinched does not
# come back with it -- a compassionate man wakes callous. The options are only how he carries it.
ln_risen.12 = {
	type = character_event
	title = ln_risen.12.t
	desc = ln_risen.12.desc
	theme = faith
	left_portrait = { character = root  animation = personality_bold }

	trigger = {
		has_trait = resurrected_by_rhllor
		is_alive = yes
		NOT = { has_character_modifier = ln_fire_in_the_blood_modifier }
	}

	immediate = {
		add_character_modifier = ln_fire_in_the_blood_modifier
		# What burned away was the part that flinched. Sadists keep their cruelty; everyone else
		# drifts toward cold. Guarded like every personality swap: opposites block add_trait.
		if = {
			limit = { has_trait = compassionate }
			remove_trait = compassionate
			add_trait = callous
		}
		else_if = {
			limit = {
				NOR = {
					has_trait = callous
					has_trait = sadistic
				}
			}
			add_trait = callous
		}
	}

	option = {
		name = ln_risen.12.a          # then let it burn.
		ai_chance = {
			base = 55
			modifier = { add = 25  has_trait = zealous }
			modifier = { add = 15  has_trait = ambitious }
		}
		add_piety = 50
	}
	option = {
		name = ln_risen.12.b          # tell no one.
		ai_chance = {
			base = 45
			modifier = { add = 25  has_trait = paranoid }
			modifier = { add = 15  has_trait = deceitful }
		}
		add_intrigue_lifestyle_xp = 50
	}
}
