# AGOT: The Long Night & Azor Ahai -- THE KILLING BLOW.
#
# ══════════════════════════════════════════════════════════════════════════════════════════════════════════════
# WHY THIS IS ON on_combat_end_winner AND NOT IN A COMBAT PHASE EVENT.
#
# The store page promises "killing the Night King as a full scene", and design-backlog.md §1 proposed getting it
# by overriding vanilla's `knight_killed` with an is_valid gate, citing Princes of Darkness as precedent.
# common/combat_phase_events/zz_ln_knight_phase_events.txt already contains the research that KILLED that plan,
# and it still holds: AGOT's knight_killed is 993 lines against vanilla's 463, with 587 lines of differences
# including their own `is_valid = { is_current_dragonrider_warfare = no }`. Shipping our copy silently reverts
# every one of those the day AGOT edits it -- dragonriders quietly become knight-killable again, and nothing
# logs it.
#
# THIS IS THE THIRD ROUTE, and it needs no file of theirs and no file of vanilla's. AGOT itself hangs
# agot_others_events.9000 and agot_combat_event.1000 off on_combat_end_winner, which gave us the scope shape for
# free: root is the WINNING combat_side, with side_commander, enemy_side and combat.location all reachable.
# So: you beat his army in the field, and then the scene happens.
#
# WHAT IT DOES NOT DO. It does not intercept his death, because nothing in CK3 can -- on_death is reactive and
# on_natural_death_second_chance is natural-death only, both verified previously. It does not need to. He is
# alive when this fires; whether he stays that way is the point of the card.
# ══════════════════════════════════════════════════════════════════════════════════════════════════════════════

# ── 1. THE DIRECTOR ─────────────────────────────────────────────────────────────────────────────────────────
# Root = the winning combat_side. Hidden, rootless-by-character, fires the scene at the man who won.
ln_dawn.1 = {
	hidden = yes
	scope = combat_side

	trigger = {
		# Somebody human won it, and lived to stand in the field afterwards.
		exists = side_commander
		side_commander = {
			is_alive = yes
			is_ai = no
		}

		# And the thing they beat was HIM. He leads from the front -- the story cycle re-assigns him as army
		# commander every fifteen days -- but primary_participant is checked too so a stand-in cannot hide him.
		exists = enemy_side
		enemy_side = {
			OR = {
				AND = {
					exists = side_commander
					side_commander = { has_trait = nightking }
				}
				AND = {
					exists = side_primary_participant
					side_primary_participant = { has_trait = nightking }
				}
			}
		}

		# Once. The wheel can turn under The Cycle, so this is cleared with the other latches at a new summer
		# rather than being permanent.
		NOT = { has_global_variable = ln_dawn_offered }
	}

	immediate = {
		set_global_variable = { name = ln_dawn_offered  value = yes }

		enemy_side = {
			if = {
				limit = {
					exists = side_commander
					side_commander = { has_trait = nightking }
				}
				side_commander = { save_scope_as = ln_night_king }
			}
			else = {
				side_primary_participant = { save_scope_as = ln_night_king }
			}
		}

		side_commander = { trigger_event = { id = ln_dawn.2  days = 1 } }
	}
}

# ── 3. HE FALLS ─────────────────────────────────────────────────────────────────────────────────────────────
# The death itself, and the ONLY new route to it this mod has ever shipped: before this, open battle or a war
# victory were the only ways, both of which the ancestor mods' own workshop threads complained about.
# `death` routes through on_death -> ln_death_catalyst_effect -> ln_night_king_slain_reveal_effect, which reads
# scope:killer -- so `killer = root` here is what names the slayer on the reveal card, in the roll-call, and in
# the permanent ln_last_dawn_champion record.
ln_dawn.3 = {
	type = character_event
	window = fullscreen_event
	title = ln_dawn.3.t
	desc = ln_dawn.3.desc
	theme = murder_scheme
	override_background = { reference = agot_frozen_forest }

	trigger = {
		exists = scope:ln_night_king
		scope:ln_night_king = { is_alive = yes }
	}

	immediate = {
		play_music_cue = "mood_the_others"
		scope:ln_night_king = {
			death = {
				death_reason = death_the_dawn
				killer = root
			}
		}

		# ── THE DIAGNOSTIC, and it exists because of how this mod debugs ────────────────────────────────
		# nightking, other_trait and wight_trait are all `immortal = yes` (common/traits/invasions_traits.txt,
		# life_expectancy 12000). CK3's own trait docs say immortal blocks death from NATURAL CAUSES only and
		# explicitly not battle or scripted death, so this should simply work -- but it has never been tested
		# in a live game, and if the engine did refuse it the whole feature would do NOTHING, silently, which
		# is the exact failure this project has lost the most time to.
		#
		# So: if he is somehow still breathing one line later, stamp it. Debug saves are plain text and
		# grep-able, which is how every pacing question in this project has actually been answered -- one
		# search for ln_dawn_death_refused tells us instantly whether the death was refused or the event
		# never fired at all. Costs nothing when everything works, because it never gets set.
		if = {
			limit = { scope:ln_night_king = { is_alive = yes } }
			set_global_variable = { name = ln_dawn_death_refused  value = yes }
		}
	}

	option = { name = ln_dawn.3.a }
}

# ── 4. HE DOES NOT ──────────────────────────────────────────────────────────────────────────────────────────
# "More ways to lose well" -- the second thing the store page promises. design-backlog.md: *"make failure
# produce outcomes too... 'You died' is the least interesting branch."* So nobody dies here except somebody who
# chose to. The knight is picked the same way the roll-call picks its three: ordered_knight by prowess, taken
# at the moment it matters.
ln_dawn.4 = {
	type = character_event
	title = ln_dawn.4.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { exists = scope:ln_dawn_shield }
				desc = ln_dawn.4.desc.shield
			}
			desc = ln_dawn.4.desc.alone
		}
	}
	theme = murder_scheme
	override_background = { reference = agot_frozen_forest }
	left_portrait = { character = root  animation = grief }

	immediate = {
		# Somebody sworn to you gets between you and it.
		ordered_knight = {
			order_by = prowess
			position = 0
			limit = { is_alive = yes }
			save_scope_as = ln_dawn_shield
		}
		if = {
			limit = { exists = scope:ln_dawn_shield }
			scope:ln_dawn_shield = {
				add_character_flag = ln_stood_at_the_dawn
				death = { death_reason = death_the_dawn  killer = scope:ln_night_king }
			}
		}
		else = {
			# Nobody left to do it for you. You walk out of the field carrying it yourself.
			increase_wounds_no_death_effect = { REASON = death_wounds }
		}
		add_character_flag = ln_faced_him_and_lived
		# ── 1.0.15c REVIEW FIND: AN UNARMED ATTEMPT MUST NOT SPEND THE WAVE'S ONE SCENE. ────────────
		# ln_dawn.1 offers this once per wave (ln_dawn_offered, cleared only by the Cycle re-arm). With
		# the bane rule in place, a man who beats the host while carrying nothing that can hurt the King
		# of Winter cannot win the roll -- so before this, his first victory burned the mod's flagship
		# scene on a card he could not have won, for the rest of the wave.
		# Re-armed ONLY for the baneless. An armed man who simply lost the roll had his chance and does
		# not get another: that is the gamble, and re-arming it would make the scene farmable.
		if = {
			limit = {
				ln_bears_the_bane_trigger = no
				has_global_variable = ln_dawn_offered
			}
			debug_log = "LN_TRACE BANE: the field scene was spent by a man carrying no bane - re-arming it"
			remove_global_variable = ln_dawn_offered
		}
		# THE LOUDEST THING A CK3 CHARACTER CAN SAY ABOUT ITSELF: it goes in the title bar, the family
		# tree and every other realm diplomacy screen, and it survives into a screenshot. He did not
		# win -- ln_dawn.3 is where winning lives -- but he is the only man in the world who has stood
		# inside its reach and walked off the field, and the realm is going to have a word for that.
		give_nickname = nick_the_undaunted
	}

	option = { name = ln_dawn.4.a }
}

# ══ 1.0.15 -- THE ENDING FUNNEL'S THREE NEW CARDS ═══════════════════════════════════════════════════════════
# ln_dawn.11 : THE LAST HOUR. The destiny-chosen champion ends the Long Night -- fired by the score-victory
#              cornering (war won, he still breathes) and by the captured-king chain (he is in chains).
#              The death is the LAST line of the option, after every read that depends on him being alive.
# ln_dawn.12 : the hidden failsafe. If the scene is lost, the war leader (or captor) finishes it without
#              ceremony on day 14. A lost war can never again leave a living Night King.
# ln_dawn.20 : the cold thing in your cells -- a captured lesser Other or wight, resolved by the captor.

# ── THE LAST HOUR ───────────────────────────────────────────────────────────────────────────────────────────
# Root = the champion the worthiness ordering picked. Rebuilds its scopes from the globals, the handshake
# pattern -- the card is days away from the effect that queued it.
ln_dawn.11 = {
	type = character_event
	window = fullscreen_event
	title = ln_dawn.11.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					has_global_variable = ln_dawn_in_chains
					ln_carries_lightbringer_trigger = yes
				}
				desc = ln_dawn.11.desc.chains_lb
			}
			triggered_desc = {
				trigger = { has_global_variable = ln_dawn_in_chains }
				desc = ln_dawn.11.desc.chains_plain
			}
			triggered_desc = {
				trigger = { ln_carries_lightbringer_trigger = yes }
				desc = ln_dawn.11.desc.field_lb
			}
			desc = ln_dawn.11.desc.field_plain
		}
	}
	theme = murder_scheme
	override_background = { reference = agot_frozen_forest }
	left_portrait = { character = root  animation = personality_bold }
	right_portrait = { trigger = { exists = scope:ln_night_king }  character = scope:ln_night_king  animation = disbelief }

	trigger = {
		is_alive = yes
		exists = global_var:ln_dawn_cornered_king
		global_var:ln_dawn_cornered_king ?= { is_alive = yes }
		# Only the designated hand sees this card -- a stale queue from the deferral cannot double-fire it.
		global_var:ln_dawn_final_champion ?= root
		# And a king who broke his chains is not executed by a card (review find): the chains variant only
		# proceeds while he is actually still held. A re-capture starts a fresh chain via on_imprison.
		trigger_if = {
			limit = { has_global_variable = ln_dawn_in_chains }
			global_var:ln_dawn_cornered_king ?= { is_imprisoned = yes }
		}
	}

	immediate = {
		play_music_cue = "mood_the_others"
		debug_log = "LN_TRACE ENDGAME: the last hour is on screen"
		debug_log_scopes = yes
		global_var:ln_dawn_cornered_king ?= { save_scope_as = ln_night_king }
	}

	# ── END IT. No roll. The war is over and he has nothing left to bargain with; this card is the
	# ceremony, and the ceremony is the point -- the death happens HERE, killer = the man the story chose,
	# and the catalyst chain (reveal, trait, nickname, cleanup) runs off that death exactly as it would for
	# a killing blow in the field.
	option = {
		name = ln_dawn.11.a
		# 1.0.15b -- THE BANE RULE. Even cornered, even in chains, he cannot be ended by ordinary steel.
		# The worthiness ordering weights a bane-bearer +1000 so this option is almost always available to
		# the man the story picked; when nobody in the realm carries one, ln_dawn.11.c below is what is left.
		trigger = { ln_bears_the_bane_trigger = yes }
		# Banked before the blow, per the standing ordering rule: nothing after this line reads him alive.
		scope:ln_night_king = {
			death = {
				death_reason = death_the_dawn
				killer = root
			}
		}
		remove_global_variable = ln_dawn_cornered_king
		if = { limit = { has_global_variable = ln_dawn_final_champion } remove_global_variable = ln_dawn_final_champion }
		if = { limit = { has_global_variable = ln_dawn_second_champion } remove_global_variable = ln_dawn_second_champion }
		if = { limit = { has_global_variable = ln_dawn_in_chains } remove_global_variable = ln_dawn_in_chains }
		if = { limit = { has_global_variable = ln_dawn_deferred } remove_global_variable = ln_dawn_deferred }

		ai_chance = { base = 100 }
	}

	# ── LET ANOTHER HAND DO IT. Once, and only if the ordering produced a second name -- a king who
	# believes the prophecy can hand the moment to the man it names. The story does not wait twice.
	option = {
		name = ln_dawn.11.b
		trigger = {
			exists = global_var:ln_dawn_second_champion
			NOT = { has_global_variable = ln_dawn_deferred }
			NOT = { global_var:ln_dawn_second_champion ?= root }
		}
		set_global_variable = { name = ln_dawn_deferred  value = yes }
		# The this-pattern, same as the reveal's knight stamps: scope in, then write the global off it.
		global_var:ln_dawn_second_champion ?= {
			set_global_variable = { name = ln_dawn_final_champion  value = this }
			trigger_event = { id = ln_dawn.11  days = 1 }
		}
		remove_global_variable = ln_dawn_second_champion

		ai_chance = { base = 0 }
	}

	# ── NOTHING WE CARRY CAN END HIM. The only option a baneless champion has, and the honest one.
	# The war is still won -- the host is broken either way -- but the Long Night is not over, because the
	# Long Night is not over until he dies, and he has not. He goes back into the dark, and the realm now
	# knows exactly what it must find before it can finish this.
	option = {
		name = ln_dawn.11.c
		trigger = { ln_bears_the_bane_trigger = no }
		custom_tooltip = ln_dawn.11.c_tt

		debug_log = "LN_TRACE ENDGAME: the war was won by a champion who carries no bane - he is cast down, not destroyed"
		add_character_flag = ln_faced_him_and_lived
		add_prestige = 500
		stress_impact = { base = medium_stress_impact_gain }

		# ── 1.0.15c REVIEW FIND (BLOCKER): HE HAS TO GO BACK INTO THE WORLD. ───────────────────────────
		# The capture chain stamps keep_imprisoned, which hides AGOT's entire prisoner-interaction
		# surface, and nothing anywhere removes it. Before this, a baneless hall that took him alive left
		# him in a cell forever: un-releasable, un-recapturable (on_imprison cannot re-fire for a man
		# already imprisoned), never in the field again -- the Long Night unendable for the rest of the
		# save. If we cannot destroy him, we do not get to keep him either.
		if = {
			limit = { has_global_variable = ln_dawn_in_chains }
			scope:ln_night_king ?= {
				remove_character_flag = keep_imprisoned
				release_from_prison = yes
			}
		}
		# 1.0.15c REVIEW FIND: and the war that produced no ending must not leave its victory global
		# standing. Left set it permanently disarms should_invalidate for the rest of the wave AND
		# silently degrades others_war_resolution into death-resolution, because the dawn gate reads it.
		if = { limit = { has_global_variable = war_for_dawn_victory } remove_global_variable = war_for_dawn_victory }

		# The handles come down; the struggle stays where it is. A fresh cornering can happen the next
		# time the living beat him in the field or take him alive -- with obsidian, next time.
		remove_global_variable = ln_dawn_cornered_king
		if = { limit = { has_global_variable = ln_dawn_final_champion } remove_global_variable = ln_dawn_final_champion }
		if = { limit = { has_global_variable = ln_dawn_second_champion } remove_global_variable = ln_dawn_second_champion }
		if = { limit = { has_global_variable = ln_dawn_in_chains } remove_global_variable = ln_dawn_in_chains }
		if = { limit = { has_global_variable = ln_dawn_deferred } remove_global_variable = ln_dawn_deferred }

		ai_chance = { base = 100 }
	}
}

# ── THE FAILSAFE ────────────────────────────────────────────────────────────────────────────────────────────
# Root = the war's primary attacker, or the captor. Hidden. If the King of Winter is somehow still breathing
# two weeks after the war was decided, he stops -- without a card, with a log line, by the leader's hand.
ln_dawn.12 = {
	hidden = yes

	trigger = {
		is_alive = yes
		exists = global_var:ln_dawn_cornered_king
		global_var:ln_dawn_cornered_king ?= { is_alive = yes }
		# Same escaped-king guard as ln_dawn.11 (review find): a failsafe must not behead at range a king
		# who is no longer in anybody's cells. The field route carries no such global and is unaffected.
		trigger_if = {
			limit = { has_global_variable = ln_dawn_in_chains }
			global_var:ln_dawn_cornered_king ?= { is_imprisoned = yes }
		}
	}

	immediate = {
		# 1.0.15b -- THE FAILSAFE OBEYS THE BANE RULE TOO. It exists so a won war cannot leave him standing
		# through a lost event queue; it does not exist to hand the kill to a man who could not have made
		# it. No bane, no blow -- the handles come down and the dark keeps its king.
		if = {
			limit = { ln_bears_the_bane_trigger = yes }
			debug_log = "LN_TRACE ENDGAME FAILSAFE: the scene was lost - the leader finishes it"
			global_var:ln_dawn_cornered_king ?= {
				death = {
					death_reason = death_the_dawn
					killer = root
				}
			}
			# 1.0.15c REVIEW FIND: THE TEARDOWN LIVES INSIDE THE ARMED BRANCH NOW. This failsafe is queued
			# at TWO men -- the leader on day 14, the champion on day 15. With the teardown outside the
			# branch, a baneless day-14 holder removed the handles and the day-15 copy failed its own
			# trigger: the unarmed man disarmed the armed one, in exactly the case the failsafe exists
			# for. Leaving the handles standing on the baneless path costs nothing -- the next cornering
			# overwrites them and the Cycle re-arm clears them.
			remove_global_variable = ln_dawn_cornered_king
			if = { limit = { has_global_variable = ln_dawn_final_champion } remove_global_variable = ln_dawn_final_champion }
			if = { limit = { has_global_variable = ln_dawn_second_champion } remove_global_variable = ln_dawn_second_champion }
			if = { limit = { has_global_variable = ln_dawn_in_chains } remove_global_variable = ln_dawn_in_chains }
			if = { limit = { has_global_variable = ln_dawn_deferred } remove_global_variable = ln_dawn_deferred }
		}
		else = {
			debug_log = "LN_TRACE ENDGAME FAILSAFE: scene lost and no bane in the leader's hand - he is cast down, not destroyed"
			# Same blocker as ln_dawn.11.c: a cell we can never open is worse than a king in the field.
			# Only the day-15 copy (the champion) can still act, so the handles deliberately stay up.
			if = {
				limit = { has_global_variable = ln_dawn_in_chains }
				global_var:ln_dawn_cornered_king ?= {
					remove_character_flag = keep_imprisoned
					release_from_prison = yes
				}
			}
			if = { limit = { has_global_variable = war_for_dawn_victory } remove_global_variable = war_for_dawn_victory }
		}
	}
}
