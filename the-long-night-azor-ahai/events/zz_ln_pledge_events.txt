namespace = ln_pledge

# AGOT: The Long Night & Azor Ahai -- the priest at the gate.
#
# The interloper's road into the war. See the header of common/decisions/zz_ln_pledge_decisions.txt for why this
# is shaped as a conversion offer rather than a "join the struggle" button: R'hllor is an involved faith, and
# changing your faith is the only thing a character can actually DO that moves them from interloper to involved
# in CK3's own reckoning. Everything else in this chain is about making that cost what it should cost.

# ── THE ARRIVAL ─────────────────────────────────────────────────────────────────────────────────────────────
ln_pledge.1 = {
	type = character_event
	title = ln_pledge.1.t
	desc = {
		first_valid = {
			# The dark is already here. He did not come to convince you of anything.
			triggered_desc = {
				trigger = {
					struggle:ln_long_night_struggle ?= { is_struggle_phase = ln_phase_the_long_night }
				}
				desc = ln_pledge.1.desc.night
			}
			# The cold has started. He came because you asked, and he is not surprised that you did.
			triggered_desc = {
				trigger = {
					struggle:ln_long_night_struggle ?= { is_struggle_phase = ln_phase_cold_winds }
				}
				desc = ln_pledge.1.desc.cold
			}
			desc = ln_pledge.1.desc.summer
		}
	}
	theme = faith
	left_portrait = {
		character = root
		animation = worry
	}

	# 1.0.9 -- AND NOW HE HAS A FACE. This card previously spoke of a priest who was usually not there, so
	# there was nobody to show. ln_summon_the_red_priest_effect in the immediate below guarantees somebody
	# is standing in the hall by the time this renders -- found, famous, or freshly off a Myrish galley --
	# so put him on the panel. Triggered rather than bare: a portrait pointed at a scope that does not
	# exist is a log line on every render, and this project has paid for that lesson once already.
	right_portrait = {
		trigger = { exists = scope:ln_priest }
		character = scope:ln_priest
		animation = personality_zealous
	}

	immediate = {
		# Somebody has to be standing there. Prefer a red priest already in the hall -- if the player went and
		# got one, this is his moment -- and otherwise the priest is offstage and the event speaks of him.
		if = {
			limit = { ln_has_red_priest_trigger = yes }
			if = {
				limit = { cp:councillor_court_chaplain ?= { has_trait = red_priest } }
				cp:councillor_court_chaplain = { save_scope_as = ln_priest }
			}
			else_if = {
				limit = { court_position:camp_priest_camp_officer ?= { has_trait = red_priest } }
				court_position:camp_priest_camp_officer = { save_scope_as = ln_priest }
			}
		}

		# 1.0.9 -- AND IF NOBODY IS STANDING THERE, ONE ARRIVES. This event has always spoken of a priest
		# and never produced one, which was fine while the decision was only an interloper's door: the
		# point was the faith, not the man. It is open to people who already follow R'hllor now, and for
		# them the man IS the point -- seating him as Court Chaplain is what satisfies
		# ln_has_a_seer_trigger, which is what makes being named Chosen possible at all.
		# No-ops if the branch above already found somebody. See zz_ln_red_priest_effects.txt.
		ln_summon_the_red_priest_effect = yes
	}

	# ── TAKE THE RED. The engine-level transition: R'hllor is an involved faith, so this and only this makes
	# an interloper INVOLVED, with the Struggle Involvement panel giving the true reason.
	option = {
		name = ln_pledge.1.a
		trigger = {
			NOT = { faith = { religion = religion:the_rhllor_religion } }
		}
		# Says the quiet part out loud. The player deserves to know that this option and no other actually
		# changes their standing in the struggle, and why -- guessing at it from the involvement panel later is
		# not a mechanic, it is a puzzle.
		custom_tooltip = ln_pledge.1.a.tt

		set_character_faith = faith:rhllor_fc
		add_character_flag = ln_took_the_red
		add_character_flag = ln_sworn_to_the_dawn
		add_character_modifier = {
			modifier = ln_took_the_red_modifier
			years = 20
		}
		add_piety = -200
		add_prestige = -100

		# The hall empties out. Somebody says something.
		trigger_event = { id = ln_pledge.2  days = { 20 60 } }

		ai_chance = {
			base = 20
			modifier = {
				add = 40
				struggle:ln_long_night_struggle ?= { is_struggle_phase = ln_phase_the_long_night }
			}
			modifier = { add = 25  has_trait = cynical }
			modifier = { factor = 0.1  has_trait = zealous }
		}
	}

	# ── KEEP YOUR GODS AND GO ANYWAY. Stays an interloper on the panel, which is the truth, and gets full
	# access to this mod through ln_participates_in_the_dawn_trigger, which is also the truth.
	option = {
		name = ln_pledge.1.b

		add_character_flag = ln_sworn_to_the_dawn
		add_character_flag = ln_heard_the_warning     # whatever else, you now KNOW, and the dragonglass opens
		add_character_modifier = ln_sworn_to_the_dawn_modifier
		add_prestige = 100

		trigger_event = { id = ln_pledge.3  days = { 30 90 } }

		ai_chance = {
			base = 50
			modifier = { add = 30  has_trait = brave }
			modifier = { add = 20  ai_boldness >= 20 }
			modifier = { add = 25  ln_has_deed_greensight_trigger = yes }
		}
	}

	# ── BURN HIM. The popular answer, the pious answer, and the one that shuts the door.
	option = {
		name = ln_pledge.1.c

		add_character_flag = ln_burned_the_red_priest
		add_character_modifier = ln_burned_the_red_priest_modifier
		add_piety = 150
		add_dread = 10

		if = {
			limit = { exists = scope:ln_priest }
			scope:ln_priest = { death = { death_reason = death_execution  killer = root } }
		}

		trigger_event = { id = ln_pledge.4  days = { 60 200 } }

		ai_chance = {
			base = 30
			modifier = { add = 60  has_trait = zealous }
			modifier = { add = 25  faith = { has_doctrine = doctrine_pluralism_fundamentalist } }
			modifier = {
				factor = 0.2
				struggle:ln_long_night_struggle ?= { is_struggle_phase = ln_phase_the_long_night }
			}
		}
	}
}

# ── AFTER THE CONVERSION ────────────────────────────────────────────────────────────────────────────────────
# The bill for changing your gods in front of your own bannermen. There is no option here that undoes it.
ln_pledge.2 = {
	type = character_event
	title = ln_pledge.2.t
	desc = ln_pledge.2.desc
	theme = faith
	left_portrait = { character = root  animation = disapproval }
	right_portrait = { trigger = { exists = scope:ln_objector }   character = scope:ln_objector  animation = anger }

	trigger = { has_character_flag = ln_took_the_red }

	immediate = {
		# Whoever is angriest and closest. A vassal first, then the heir, then anyone in the court.
		if = {
			limit = { any_vassal = { opinion = { target = root  value < 0 } } }
			random_vassal = {
				limit = { opinion = { target = root  value < 0 } }
				save_scope_as = ln_objector
			}
		}
		else_if = {
			limit = { exists = primary_heir }
			primary_heir = { save_scope_as = ln_objector }
		}
		else = {
			random_courtier = { save_scope_as = ln_objector }
		}
	}

	# Hold the line. Costs you him, gains you the realm knowing you meant it.
	option = {
		name = ln_pledge.2.a
		add_prestige = 100
		add_dread = 5
		if = {
			limit = { exists = scope:ln_objector }
			scope:ln_objector = { add_opinion = { modifier = angry_opinion  target = root  opinion = -30 } }
		}
	}

	# Let him keep the Seven. You did not do this to make anyone else do it.
	option = {
		name = ln_pledge.2.b
		add_piety = -50
		if = {
			limit = { exists = scope:ln_objector }
			scope:ln_objector = { add_opinion = { modifier = respect_opinion  target = root  opinion = 20 } }
		}
		every_vassal = {
			limit = { NOT = { faith = root.faith } }
			add_opinion = { modifier = respect_opinion  target = root  opinion = 10 }
		}
	}
}

# ── AFTER THE PLEDGE ────────────────────────────────────────────────────────────────────────────────────────
# The men go north. This is the one that makes ln_sworn_to_the_dawn feel like it cost something.
ln_pledge.3 = {
	type = character_event
	title = ln_pledge.3.t
	desc = ln_pledge.3.desc
	theme = martial
	left_portrait = { character = root  animation = thinking }

	trigger = { has_character_flag = ln_sworn_to_the_dawn }

	# Send more than you promised.
	option = {
		name = ln_pledge.3.a
		add_prestige = 150
		add_character_modifier = {
			modifier = ln_sworn_to_the_dawn_modifier
			years = 40
		}
		stress_impact = { base = minor_stress_impact_gain }
	}

	# Send what you promised and not a man more. Somebody has to hold the south.
	option = {
		name = ln_pledge.3.b
		add_gold = 50
	}
}
