# The Long Night+ — #12 the Wall, LAYER B cinematic (Option A: middle break + full animation).
# Fired per human player from agot_long_night_wall_collapse_effect.
#
# Built on AGOT's working widget. Tweaks vs the bare AGOT version:
#   - widget = our copy ln_particle_wall_collapse (AGOT's widget byte-for-byte, delay kept at 0.1 — raising it
#     breaks the widget entirely). Music sync is solved on the AUDIO side instead: we play a trimmed copy of the
#     track (LN_wall_collapse_the_others) whose intro is shaved so the boom lands on the ~instant fall.
#   - window = ln_topleft_wall_event (AGOT's character_event window anchored TOP-LEFT so it no longer masks the
#     animation). The window stays VISIBLE/on-screen — hiding it breaks the widget's clock (whole saga).
# The widget spawns the collapse mesh, breaks the Wall via AGOT's agot_wall_fall (building swap to the broken mesh),
# spawns the dust effect and AUTO-CLOSES the event. We only pan the camera and play the collapse music.
long_night_plus_wall.1 = {
	type = character_event
	window = ln_topleft_wall_event
	title = long_night_plus_wall.1.t
	desc = long_night_plus_wall.1.desc
	theme = default

	immediate = {
		pan_camera_to_province = title:b_castle_black.title_province
		play_music_cue = LN_wall_collapse_the_others # trimmed copy, boom synced on the fall (music/in_game/zz_ln_sounds.txt)
	}

	widgets = {
		widget = {
			gui = "ln_particle_wall_collapse"
			container = "custom_widgets_container"
			controller = default
			setup_scope = { }
		}
	}

	option = {
		name = long_night_plus_wall.1.a
	}
}

# Reliable fallback: with no human player to play the widget (AI-only realm / observer), break the Wall directly if it
# is still standing. Mirrors what the widget's agot_wall_fall does (middle break + building swap to force the refresh).
long_night_plus_wall.3 = {
	# FIX (v0.1.0): declared no `type`, so it defaulted to character_event -- but it is fired from the STORY scope
	# in agot_long_night_wall_collapse_effect, a scope mismatch, so in AI-only/observer games the Wall very likely
	# never broke. It needs no root character; the immediate block scopes to the province explicitly. Same shape as
	# others.1 in others_events.txt.
	# No `type` field: vanilla's pattern for a rootless hidden event is scope = none + hidden = yes and
	# nothing else (game/events/, 10 instances). `type = empty` is REJECTED by the parser -- Long Night+
	# used it and it has been erroring since before the fork. character_event is also wrong here: these
	# fire from struggle and story scopes with no character root.
	scope = none
	hidden = yes
	trigger = {
		title:b_castle_black.title_province = {
			NOT = { has_variable = agot_wall_has_fallen_left }
			NOT = { has_variable = agot_wall_has_fallen_middle }
			NOT = { has_variable = agot_wall_has_fallen_right }
		}
	}
	immediate = {
		title:b_castle_black.title_province = {
			# THE VARIABLE IS THE WHOLE BREAK. Deliberately NOT guarded on has_variable = agot_wall_is_normal
			# the way AGOT's agot_wall_fall widget guards its copy (00_agot_scripted_gui.txt:1009-1013).
			# That variable is set by AGOT's own agot_wall_on_actions.txt:90 and by our Long Summer restore
			# (zz_ln_struggle_effects.txt:139,146,153); borrowing their guard would mean a game where that
			# on_action never ran is a game where the Wall can never break -- and this event IS the AI-only
			# fallback, the path nobody is watching. The trigger above already permits exactly one run: it
			# refuses if any of the three fallen variables is set, and this sets one.
			# special_the_wall_broken_middle_graphical (98_special_buildings.txt:109) reads this variable and
			# nothing else, so the break is recorded and the broken mesh is selected on the next refresh.
			set_variable = agot_wall_has_fallen_middle

			# 1.0.16 LIVE-RUN FIX -- BOTH BUILDING CALLS WERE NO-OPS THAT ERRORED. 27 Aug run, 16:58:12:
			#   Error: remove_building effect [ Province 'Castle Black' does not have building 'building_special_the_wall' ]
			#   Error: add_building effect [ No building slots available in province 'Castle Black' ]
			#
			# AND THEY WERE ALWAYS GOING TO. special_the_wall is DEFINED (98_special_buildings.txt:7) and
			# PLACED NOWHERE -- it appears in no province history in AGOT at all. Castle Black's one special
			# slot holds agot_castle_black_01 (AGOT history/provinces/00_k_the_wall_prov.txt:8), which is why
			# the remove finds nothing and the add has nowhere to go.
			#
			# THIS IS AGOT'S, NOT OURS, AND WE ARE NOT FIXING IT HERE. Their own agot_wall_fall widget runs
			# the identical pair unguarded, so the player-facing path no-ops in exactly the same way. Freeing
			# the slot would mean evicting agot_castle_black_01 -- destroying their content to make our
			# cosmetic refresh land. Standing policy: report it upstream, do not repair base AGOT where the
			# repair costs more than the symptom.
			#
			# WHAT IS ACTUALLY LOST, stated plainly rather than glossed: the special_the_wall_damaged
			# modifiers (defender_holding_advantage -50, fort_level -2, garrison_size -0.25) are not applied,
			# and the forced graphical refresh -- 98_special_buildings.txt:2, "GRAPHICAL BACKGROUNDS ARE ONLY
			# REFRESHED WHEN A BUILDING IS CHANGED IN THE HOLDING" -- does not happen, so the broken-Wall
			# mesh waits for the next building change in that holding. Identical to the player path. The
			# guards below change nothing except that the log stops carrying two lines that were never going
			# to be actionable, and they will start working the day AGOT places special_the_wall.
			if = {
				limit = { has_building = special_the_wall }
				remove_building = special_the_wall
				add_building = special_the_wall_damaged
			}
		}
	}
}

# The MONUMENT (v0.2.1, Thomas's live-run note: "should be a monumental, big card of 'the wall has
# fallen'"). The cinematic (.1) stays exactly as it is -- its window is small and top-left because the
# widget saga forbids anything else, and while the Wall is falling the animation IS the card. This is
# what arrives two days later, when the dust has settled and the realm understands what it saw: a
# full-size event against AGOT's own Wall backdrop, the Night King himself on the panel, and prose
# that treats the moment as the hinge of the age. Fired per player from the same collapse effect.
long_night_plus_wall.2 = {
	type = character_event
	title = long_night_plus_wall.2.t
	# 1.1.0 (A1) -- A STATION OPENER, THEN THE MONUMENT. Not a new card: this one already reaches every player

	# day would land on the cinematic's window (the 1.0.9 note above). first_valid is first-match-wins, ordered
	# exactly as ln_opening.6's ladder (watch / free folk / dragon / north / red / lord / nobody) so the two

			}

				desc = long_night_plus_wall.2.desc.north
			}

						faith = faith:rhllor_fots

		# wall fell." This card IS the moment (see above), so a dated memory here reads exactly that on the
		# character page for the rest of his life. Type: zz_ln_melee_memories.txt. No participants.
		create_character_memory = { type = ln_the_wall_fell_memory }
	}

	left_portrait = { trigger = { exists = scope:ln_wall_card_night_king }  
		character = scope:ln_wall_card_night_king
		animation = personality_bold
	}

		name = long_night_plus_wall.2.a
	}
}
