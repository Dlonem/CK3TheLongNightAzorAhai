# AGOT: The Long Night & Azor Ahai -- the Night King has no hair.
# ══ 1.0.15 -- THE TRAIT ALONE IS NO LONGER TRUSTED (the "dead skin on living women" reports). ═══════════════
# Ereska, KngZawd, CinkoSenpai: after AGOT's 0.5.1 update, LIVING characters -- women above all -- rendered
# with the corpse look. Our appliers were keyed on the trait alone, and trait references in a mid-update save
# are exactly the thing that drifts when any mod in the playset adds or removes traits. Every undead-look
# entry now also requires one by-NAME marker that conversion stamps and drift cannot fake:
# ln_one_of_them (stamped at every creation/conversion site since 1.0.7), the Others' house, or an undead
# government. A living character who "gains" the trait through save drift now renders as what she is -- a
# living character. A genuine pre-1.0.7-save wight missing all markers renders alive until the heartbeat
# re-stamps it (mutate_into_other_effect backfill, 1.0.15) -- the mild failure, chosen deliberately.
#
# He shipped with a full head of hair and a beard, which reads as a bearded old king in ice armour rather than
# as an Other. Templates verified against AGOT's own gene files:
#   hairstyles -> no_hairstyles   (common/genes/03_genes_special_accessories_hairstyles.txt)
#   beards     -> no_beard        (common/genes/04_genes_special_accessories_beards.txt, note: singular)
# Same shape AGOT uses in gfx/portraits/portrait_modifiers/00_agot_visible_disfigurement_overrides.txt and that
# Long Night+ uses in zz_wight_nudity.txt.
#
# Gated on `nightking` only, so ordinary Others are untouched for now. Book-accurately every White Walker is
# hairless -- widening this to has_trait = other_trait is a one-line change to the weight block below.

ln_nightking_hairless = {

	usage = game
	selection_behavior = max
	priority = 6

	ln_nightking_bald = {
		dna_modifiers = {
			accessory = {
				mode = add
				gene = hairstyles
				template = no_hairstyles
				range = { 0 1 } # range kept so the randomness selection resolves correctly
			}
			accessory = {
				mode = add
				gene = beards
				template = no_beard
				range = { 0 1 }
			}
		}
		weight = {
			base = 0
			modifier = {
				add = 99999
				has_trait = nightking
				ln_is_marked_as_theirs_trigger = yes
			}
		}
	}
}
