others_armor_overrides = {
# 1.0.15 -- every trait condition below also requires a by-name undead marker; see the header of
# gfx/portraits/portrait_modifiers/others_special.txt for the save-drift reports that forced it.
# ══ 1.0.15 -- THE TRAIT ALONE IS NO LONGER TRUSTED (the "dead skin on living women" reports). ═══════════════
# Ereska, KngZawd, CinkoSenpai: after AGOT's 0.5.1 update, LIVING characters -- women above all -- rendered
# with the corpse look. Our appliers were keyed on the trait alone, and trait references in a mid-update save
# are exactly the thing that drifts when any mod in the playset adds or removes traits. Every undead-look
# entry now also requires one by-NAME marker that conversion stamps and drift cannot fake:
# ln_one_of_them (stamped at every creation/conversion site since 1.0.7), the Others' house, or an undead
# government. A living character who "gains" the trait through save drift now renders as what she is -- a
# living character. A genuine pre-1.0.7-save wight missing all markers renders alive until the heartbeat
# re-stamps it (mutate_into_other_effect backfill, 1.0.15) -- the mild failure, chosen deliberately.

	usage = game
	selection_behavior = max
	priority = 3
	
	others_armor = {
		dna_modifiers = {
			accessory = {
				mode = add
				gene = clothes
				template = no_clothes
				range = { 0 1 } # For the randomness to work correctly
			}
			accessory = {
				mode = add
				gene = others_clothes
				template = others_armor
				range = { 1 1 }
			}
		
			morph = {
                mode = modify_multiply
                gene = gene_bs_bust
                value = 0.2 # Setting this lower to reduce the exaggerated effect of large bust size when wearing clothes
                template = bust_clothes
            }
			morph = {
				mode = modify_multiply
				gene = gene_bs_body_shape
				value = 0.0
				template = body_shape_average_clothed
			}
		}	
		weight = {
			base = 0
			modifier = {
				add = 99999
				has_trait = other_trait
				ln_is_marked_as_theirs_trigger = yes
				is_male = yes
			}
		}		
	}
}
