# The Long Night+ — wight appearance.
# Two independent groups apply together:
#   1) wight_skin  — the common rotting draugr skin (ported EK2 gene ln_wight_skin), on every wight.
#   2) wight_gore  — a per-wight random wound/scar layer (ln_wight_gore) baked in post_skin_color so it shows
#      ON TOP of the draugr skin (fixes the old "wounds never visible" occlusion). The variant is selected by a
#      wight_v* character flag set once at conversion (see scripted_effects mutate_into_other_effect).
# Within a group the FIRST valid entry wins, so the flagged variants are listed before the no-flag fallback.
#
# FIX (10 Aug): mode was `replace` on all five morph blocks. Both ln_wight_skin and ln_wight_gore live in
# special_genes -- deliberately, that wrapper is what killed the ~98,000 "Persistent portrait info missing
# gene" errors -- and the engine REJECTS modify/modify_multiply/replace on a special gene:
#   [E][dnamodifier.cpp:50] "modes modify, modify_multiply and replace only work for regular genes"
# once per block, at lines 14/31/46/61/76 of the pre-fix file. Vanilla's shipping pattern for exactly this
# shape (special morph gene + template + value) is `add` -- 59 uses across game/gfx/portraits, e.g.
# gene_bubonic_plague at trait_portrait_modifiers/00_trait_modifiers.txt:258 and gene_bs_pregnant at
# portrait_modifiers/99_special.txt:173. `replace` appears 20 times in vanilla and NEVER on a special gene.
#
# WATCH THIS IN GAME. If `replace` was being dropped outright then these genes have never applied at all,
# and this is the first run where the draugr skin and the gore layer actually land. Look at a wight before
# trusting the decal priorities/alpha, which the gene file already flags as a first pass.

wight_skin = {
	wight_skin_base = {
		traits = { wight_trait }
		# 1.0.15 -- the trait alone is no longer trusted: see the drift note at the top of
		# gfx/portraits/portrait_modifiers/others_special.txt. The corpse skin needs a by-name marker too.
		trigger = {
			ln_is_marked_as_theirs_trigger = yes
		}
		dna_modifiers = {
			human = {
				morph = {
					mode = add
					gene = ln_wight_skin
					template = draugr_bs
					value = 1.0
				}
			}
		}
	}
}

wight_gore = {
	wight_gore_1 = {
		traits = { wight_trait }
		trigger = { has_character_flag = wight_v1 }
		dna_modifiers = {
			human = {
				morph = {
					mode = add
					gene = ln_wight_gore
					template = gore_1
					value = 1.0
				}
			}
		}
	}

	wight_gore_2 = {
		traits = { wight_trait }
		trigger = { has_character_flag = wight_v2 }
		dna_modifiers = {
			human = {
				morph = {
					mode = add
					gene = ln_wight_gore
					template = gore_2
					value = 1.0
				}
			}
		}
	}

	wight_gore_3 = {
		traits = { wight_trait }
		trigger = { has_character_flag = wight_v3 }
		dna_modifiers = {
			human = {
				morph = {
					mode = add
					gene = ln_wight_gore
					template = gore_3
					value = 1.0
				}
			}
		}
	}

	# Fallback for any wight without a variant flag (e.g. raised before this system existed): use variant 1.
	wight_gore_fallback = {
		traits = { wight_trait }
		# 1.0.15 review find: the flagged variants above are drift-proof by their wight_v* flags, but this
		# fallback was trait-only -- exactly the entry a drift-victim living character lands on. Same
		# by-name gate as the skin.
		trigger = {
			ln_is_marked_as_theirs_trigger = yes
		}
		dna_modifiers = {
			human = {
				morph = {
					mode = add
					gene = ln_wight_gore
					template = gore_1
					value = 1.0
				}
			}
		}
	}
}
